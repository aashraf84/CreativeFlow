   158→// ── Activity icon helper ────────────────────────────────────────────
   159→function getActivityIcon(action: string) {
   160→  const lower = action.toLowerCase();
   161→  if (lower.includes("create") || lower.includes("add") || lower.includes("new"))
   162→    return { icon: Plus, color: "text-emerald-600 dark:text-emerald-400 bg-emerald-100 dark:bg-emerald-900/40" };
   163→  if (lower.includes("edit") || lower.includes("update") || lower.includes("modify"))
   164→    return { icon: Edit3, color: "text-teal-600 dark:text-teal-400 bg-teal-100 dark:bg-teal-900/40" };
   165→  if (lower.includes("delete") || lower.includes("remove"))
   166→    return { icon: Trash2, color: "text-rose-600 dark:text-rose-400 bg-rose-100 dark:bg-rose-900/40" };
   167→  if (lower.includes("approve") || lower.includes("accept") || lower.includes("complete"))
   168→    return { icon: CheckCircle2, color: "text-green-600 dark:text-green-400 bg-green-100 dark:bg-green-900/40" };
   169→  if (lower.includes("review") || lower.includes("view"))
   170→    return { icon: Eye, color: "text-amber-600 dark:text-amber-400 bg-amber-100 dark:bg-amber-900/40" };
   171→  if (lower.includes("comment") || lower.includes("reply"))
   172→    return { icon: MessageSquare, color: "text-violet-600 dark:text-violet-400 bg-violet-100 dark:bg-violet-900/40" };
   173→  if (lower.includes("assign") || lower.includes("member") || lower.includes("join"))
   174→    return { icon: UserPlus, color: "text-cyan-600 dark:text-cyan-400 bg-cyan-100 dark:bg-cyan-900/40" };
   175→  if (lower.includes("schedule") || lower.includes("calendar"))
   176→    return { icon: CalendarCheck, color: "text-indigo-600 dark:text-indigo-400 bg-indigo-100 dark:bg-indigo-900/40" };
   177→  if (lower.includes("publish") || lower.includes("share") || lower.includes("send"))
   178→    return { icon: Share2, color: "text-sky-600 dark:text-sky-400 bg-sky-100 dark:bg-sky-900/40" };
   179→  return { icon: Activity, color: "text-slate-600 dark:text-slate-400 bg-slate-100 dark:bg-slate-900/40" };
   180→}

   182→// ── Time formatting ─────────────────────────────────────────────────
   183→function formatRelativeTime(dateStr: string): string {
   184→  try {
   185→    const date = new Date(dateStr);
   186→    const now = new Date();
   187→    const diffMs = now.getTime() - date.getTime();
   188→    const diffMins = Math.floor(diffMs / 60000);
   189→    const diffHours = Math.floor(diffMs / 3600000);
   190→    const diffDays = Math.floor(diffMs / 86400000);
   191→
   192→    if (diffMins < 1) return "Just now";
   193→    if (diffMins < 60) return `${diffMins}m ago`;
   194→    if (diffHours < 24) return `${diffHours}h ago`;
   195→    if (diffDays < 7) return `${diffDays}d ago`;
   196→    return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   197→  } catch {
   198→    return "";
   199→  }
   200→}

   202→// ── Custom Tooltip ──────────────────────────────────────────────────
   203→function CustomTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string; color: string }>; label?: string }) {
   204→  if (!active || !payload?.length) return null;
   205→  return (
   206→    <div className="rounded-lg border border-border/50 bg-background px-3 py-2 shadow-xl text-xs">
   207→      <p className="font-medium text-foreground mb-1">{label}</p>
   208→      {payload.map((p, i) => (
   209→        <div key={i} className="flex items-center gap-2">
   210→          <div className="h-2 w-2 rounded-sm" style={{ backgroundColor: p.color }} />
   211→          <span className="text-muted-foreground capitalize">{p.dataKey}:</span>
   212→          <span className="font-mono font-medium text-foreground">{p.value}</span>
   213→        </div>
   214→      ))}
   215→    </div>
   216→  );
   217→}

   219→// ── KPI Card Component ──────────────────────────────────────────────
   220→function KPICard({ data, index }: { data: KPICardData; index: number }) {
   221→  const Icon = data.icon;
   222→  return (
   223→    <motion.div variants={itemVariants}>
   224→      <Card className="relative overflow-hidden shadow-sm hover:shadow-md transition-shadow duration-200 border-border/60">
   225→        <CardContent className="p-5">
   226→          <div className="flex items-start justify-between">
   227→            <div className="flex-1 min-w-0">
   228→              <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">
   229→                {data.title}
   230→              </p>
   231→              <div className="mt-2 flex items-baseline gap-2">
   232→                <h3 className="text-2xl font-bold tracking-tight text-foreground">
   233→                  {typeof data.value === "number" ? data.value.toLocaleString() : data.value}
   234→                </h3>
   235→                {data.badge && (
   236→                  <span className={`inline-flex items-center px-1.5 py-0.5 text-[10px] font-semibold rounded-md leading-none ${data.badgeColor}`}>
   237→                    {data.badge}
   238→                  </span>
   239→                )}
   240→              </div>
   241→              {data.change !== undefined && (
   242→                <div className="mt-1.5 flex items-center gap-1">
   243→                  {data.change >= 0 ? (
   244→                    <ArrowUpRight className="h-3 w-3 text-emerald-600 dark:text-emerald-400" />
   245→                  ) : (
   246→                    <ArrowDownRight className="h-3 w-3 text-rose-600 dark:text-rose-400" />
   247→                  )}
   248→                  <span className={`text-xs font-medium ${data.change >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400"}`}>
   249→                    {Math.abs(data.change)}%
   250→                  </span>
   251→                  {data.changeLabel && (
   252→                    <span className="text-[10px] text-muted-foreground">{data.changeLabel}</span>
   253→                  )}
   254→                </div>
   255→              )}
   256→            </div>
   257→            <div className={`h-10 w-10 rounded-xl bg-gradient-to-br ${data.gradient} flex items-center justify-center shadow-lg shadow-emerald-500/20 shrink-0`}>
   258→              <Icon className={`h-5 w-5 ${data.iconColor}`} />
   259→            </div>
   260→          </div>
   261→        </CardContent>
   262→        {/* Subtle decorative gradient overlay at bottom */}
   263→        <div className="absolute bottom-0 left-0 right-0 h-px bg-gradient-to-r from-transparent via-emerald-500/20 to-transparent" />
   264→      </Card>
   265→    </motion.div>
   266→  );
   267→}

   269→// ── Team Productivity Bar ───────────────────────────────────────────
   270→function TeamProductivityBar({ team, completed, pending }: { team: string; completed: number; pending: number }) {
   271→  const total = completed + pending;
   272→  const completedPct = total > 0 ? (completed / total) * 100 : 0;
   273→  const pendingPct = total > 0 ? (pending / total) * 100 : 0;
   274→
   275→  return (
   276→    <div className="space-y-1.5">
   277→      <div className="flex items-center justify-between text-xs">
   278→        <span className="font-medium text-foreground truncate">{team}</span>
   279→        <span className="text-muted-foreground tabular-nums">
   280→          {completed}<span className="text-muted-foreground/60">/{total}</span>
   281→        </span>
   282→      </div>
   283→      <div className="flex h-2.5 w-full overflow-hidden rounded-full bg-muted/60">
   284→        <div
   285→          className="bg-gradient-to-r from-emerald-500 to-emerald-400 transition-all duration-500 rounded-l-full"
   286→          style={{ width: `${completedPct}%` }}
   287→        />
   288→        <div
   289→          className="bg-gradient-to-r from-amber-400 to-amber-300 dark:from-amber-500 dark:to-amber-400 transition-all duration-500"
   290→          style={{ width: `${pendingPct}%` }}
   291→        />
   292→      </div>
   293→    </div>
   294→  );
   295→}
