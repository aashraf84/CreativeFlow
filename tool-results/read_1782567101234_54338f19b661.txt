   100→// ── Format helpers ──────────────────────────────────────────────────
   101→function formatCurrency(amount: number): string {
   102→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" }).format(amount)
   103→}
   104→
   105→function formatDate(dateStr: string | null): string {
   106→  if (!dateStr) return "—"
   107→  return new Date(dateStr).toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" })
   108→}
   109→
   110→function formatFileSize(bytes: number): string {
   111→  if (bytes === 0) return "0 B"
   112→  const k = 1024
   113→  const sizes = ["B", "KB", "MB", "GB"]
   114→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   115→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + " " + sizes[i]
   116→}
