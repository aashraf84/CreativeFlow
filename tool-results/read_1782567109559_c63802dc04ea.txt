    55→function statusBadge(status: string) {
    56→  const map: Record<string, { variant: "default" | "secondary" | "destructive" | "outline"; icon: React.ElementType }> = {
    57→    posted: { variant: "default", icon: CheckCircle2 },
    58→    draft: { variant: "secondary", icon: FileText },
    59→    void: { variant: "destructive", icon: Ban },
    60→  }
    61→  const cfg = map[status] || map.draft
    62→  const Icon = cfg.icon
    63→  return <Badge variant={cfg.variant} className="gap-1"><Icon className="h-3 w-3" />{status.charAt(0).toUpperCase() + status.slice(1)}</Badge>
