    53→function fmtMoney(n: number): string {
    54→  const negative = n < 0;
    55→  const abs = Math.abs(n);
    56→  const formatted = new Intl.NumberFormat("en-US", {
    57→    style: "currency",
    58→    currency: "USD",
    59→    minimumFractionDigits: 2,
    60→    maximumFractionDigits: 2,
    61→  }).format(abs);
    62→  return negative ? `(${formatted})` : formatted;
    63→}
    64→
    65→function fmtNumber(n: number, decimals = 0): string {
    66→  return new Intl.NumberFormat("en-US", {
    67→    minimumFractionDigits: decimals,
    68→    maximumFractionDigits: decimals,
    69→  }).format(n);
    70→}
    71→
    72→function fmtDate(d: string | Date): string {
    73→  try {
    74→    return new Intl.DateTimeFormat("en-US", {
    75→      year: "numeric",
    76→      month: "short",
    77→      day: "2-digit",
    78→    }).format(new Date(d));
    79→  } catch {
    80→    return String(d);
    81→  }
    82→}

   228→function SummaryCard({
   229→  label,
   230→  value,
   231→  hint,
   232→  tone = "default",
   233→}: {
   234→  label: string;
   235→  value: string;
   236→  hint?: string;
   237→  tone?: "default" | "positive" | "negative" | "accent";
   238→}) {
   239→  const toneClass =
   240→    tone === "positive"
   241→      ? "text-emerald-700"
   242→      : tone === "negative"
   243→      ? "text-rose-700"
   244→      : tone === "accent"
   245→      ? "text-teal-700"
   246→      : "text-slate-900";
   247→  return (
   248→    <div className="report-summary-card">
   249→      <p className="text-[9px] font-semibold uppercase tracking-[0.12em] text-slate-500">
   250→        {label}
   251→      </p>
   252→      <p className={`text-lg font-bold tabular-nums mt-1 ${toneClass}`}>{value}</p>
   253→      {hint && <p className="text-[9px] text-slate-500 mt-0.5">{hint}</p>}
   254→    </div>
   255→  );
   256→}

   337→export function ReportsView() {
   338→  const [selected, setSelected] = useState<ReportId>("financial-summary");
   339→  const [generatedAt] = useState<Date>(new Date());
   340→
   341→  const permissions = usePermissions();
   342→
   343→  // ── Report data state ──
   344→  const [financial, setFinancial] = useState<FinancialData | null>(null);
   345→  const [balanceSheet, setBalanceSheet] = useState<BalanceSheetData | null>(null);
   346→  const [pnl, setPnl] = useState<PnlData | null>(null);
   347→  const [trialBalance, setTrialBalance] = useState<TrialBalanceData | null>(null);
   348→  const [employeeReport, setEmployeeReport] = useState<EmployeeReportData | null>(null);
   349→  const [attendanceReport, setAttendanceReport] = useState<AttendanceReportData | null>(null);
   350→  const [loadingReport, setLoadingReport] = useState(false);
   351→
   352→  // ── Store data for inventory report (no API needed) ──
   353→  const {
   354→    products,
   355→    warehouses,
   356→    employees,
   357→    leaveRequests,
   358→    expenses,
   359→    invoices,
   360→  } = useERPStore();
   361→
   362→  // ── Fetch financial report data on demand ──
   363→  const loadFinancialReport = useCallback(async (type: string) => {
   364→    try {
   365→      const res = await fetch(`/api/financial-reports?type=${type}`);
   366→      if (!res.ok) return null;
   367→      const json = await res.json();
   368→      return json?.data ?? null;
   369→    } catch {
   370→      return null;
   371→    }
   372→  }, []);
   373→
   374→  const loadEmployeeReport = useCallback(async () => {
   375→    try {
   376→      const res = await fetch("/api/reports/employees");
   377→      if (!res.ok) return null;
   378→      const json = await res.json();
   379→      return json?.data ?? null;
   380→    } catch {
   381→      return null;
   382→    }
   383→  }, []);
   384→
   385→  const loadAttendanceReport = useCallback(async () => {
   386→    try {
   387→      const res = await fetch("/api/reports/attendance");
   388→      if (!res.ok) return null;
   389→      const json = await res.json();
   390→      return json?.data ?? null;
   391→    } catch {
   392→      return null;
   393→    }
   394→  }, []);
   395→
   396→  // Load the active report whenever selection changes
   397→  useEffect(() => {
   398→    let cancelled = false;
   399→    async function load() {
   400→      setLoadingReport(true);
   401→      try {
   402→        if (selected === "financial-summary") {
   403→          if (!financial) {
   404→            const data = await loadFinancialReport("overview");
   405→            if (!cancelled && data) setFinancial(data as FinancialData);
   406→          }
   407→        } else if (selected === "balance-sheet") {
   408→          if (!balanceSheet) {
   409→            const data = await loadFinancialReport("balance-sheet");
   410→            if (!cancelled && data) setBalanceSheet(data as BalanceSheetData);
   411→          }
   412→        } else if (selected === "pnl") {
   413→          if (!pnl) {
   414→            const data = await loadFinancialReport("pnl");
   415→            if (!cancelled && data) setPnl(data as PnlData);
   416→          }
   417→        } else if (selected === "trial-balance") {
   418→          if (!trialBalance) {
   419→            const data = await loadFinancialReport("trial-balance");
   420→            if (!cancelled && data) setTrialBalance(data as TrialBalanceData);
   421→          }
   422→        } else if (selected === "employee-directory") {
   423→          if (!employeeReport) {
   424→            const data = await loadEmployeeReport();
   425→            if (!cancelled && data) setEmployeeReport(data as EmployeeReportData);
   426→          }
   427→        } else if (selected === "attendance-summary") {
   428→          if (!attendanceReport) {
   429→            const data = await loadAttendanceReport();
   430→            if (!cancelled && data) setAttendanceReport(data as AttendanceReportData);
   431→          }
   432→        }
   433→      } finally {
   434→        if (!cancelled) setLoadingReport(false);
   435→      }
   436→    }
   437→    load();
   438→    return () => {
   439→      cancelled = true;
   440→    };
   441→  }, [
   442→    selected,
   443→    financial,
   444→    balanceSheet,
   445→    pnl,
   446→    trialBalance,
   447→    employeeReport,
   448→    attendanceReport,
   449→    loadFinancialReport,
   450→    loadEmployeeReport,
   451→    loadAttendanceReport,
   452→  ]);
   453→
   454→  // ── Export PDF (print) ──
   455→  const handleExportPDF = useCallback(() => {
   456→    if (permissions.isDemoMode && !permissions.canExport) {
   457→      // Demo-mode guard: do nothing (toast already shown by parent)
   458→      return;
   459→    }
   460→    if (typeof window !== "undefined") {
   461→      window.print();
   462→    }
   463→  }, [permissions]);
   464→
   465→  const activeReport = REPORTS.find((r) => r.id === selected)!;
   466→
   467→  // Group reports by category for the selector
   468→  const grouped = useMemo(() => {
   469→    const map = new Map<string, ReportMeta[]>();
   470→    for (const r of REPORTS) {
   471→      if (!map.has(r.category)) map.set(r.category, []);
   472→      map.get(r.category)!.push(r);
   473→    }
   474→    return Array.from(map.entries());
   475→  }, []);
   476→
   477→  // Compute total pages for the active report (1-based count)
   478→  const totalPages = useMemo(() => {
   479→    // Estimate based on data size for paginated reports
   480→    if (selected === "employee-directory" && employeeReport) {
   481→      return Math.max(1, Math.ceil(employeeReport.employees.length / 18));
   482→    }
   483→    if (selected === "attendance-summary" && attendanceReport) {
   484→      return Math.max(1, Math.ceil(attendanceReport.employees.length / 18));
   485→    }
   486→    if (selected === "inventory" && products) {
   487→      return Math.max(1, Math.ceil(products.length / 18));
   488→    }
   489→    if (selected === "trial-balance" && trialBalance) {
   490→      return Math.max(1, Math.ceil(trialBalance.accounts.length / 22));
   491→    }
   492→    return 1;
   493→  }, [selected, employeeReport, attendanceReport, products, trialBalance]);
   494→
   495→  return (
   496→    <motion.div
   497→      className="reports-root p-4 sm:p-6 space-y-4 print:p-0 print:space-y-0"
   498→      variants={containerVariants}
   499→      initial="hidden"
   500→      animate="visible"
   501→    >
   502→      {/* ── Header bar (hidden on print) ── */}
   503→      <motion.div
   504→        variants={itemVariants}
   505→        className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 print:hidden"
   506→      >
   507→        <div>
   508→          <h2 className="text-2xl font-bold tracking-tight text-[#1d1d1f] dark:text-white">
   509→            Reports
   510→          </h2>
   511→          <p className="text-sm text-[#86868b] dark:text-[#aeaeb2] mt-1">
   512→            Generate professional letter-size reports — exportable to PDF.
   513→          </p>
   514→        </div>
   515→        <div className="flex items-center gap-2">
   516→          {permissions.isDemoMode && (
   517→            <Badge
   518→              variant="outline"
   519→              className="gap-1.5 border-amber-300 text-amber-700 dark:border-amber-700 dark:text-amber-300 bg-amber-50 dark:bg-amber-900/30"
   520→            >
   521→              <Lock className="h-3 w-3" />
   522→              Demo Mode
   523→            </Badge>
   524→          )}
   525→          <Button
   526→            onClick={handleExportPDF}
   527→            className="bg-gradient-to-br from-emerald-500 to-teal-600 hover:from-emerald-600 hover:to-teal-700 text-white gap-2 shadow-md shadow-emerald-500/20 print:hidden"
   528→            disabled={loadingReport}
   529→          >
   530→            {permissions.isDemoMode ? <Lock className="h-4 w-4" /> : <Printer className="h-4 w-4" />}
   531→            {permissions.isDemoMode ? "Export Locked" : "Export PDF"}
   532→          </Button>
   533→        </div>
   534→      </motion.div>
   535→
   536→      {/* ── Demo-mode banner ── */}
   537→      {permissions.isDemoMode && (
   538→        <motion.div
   539→          variants={itemVariants}
   540→          className="flex items-start gap-3 p-3 rounded-xl border border-amber-200 dark:border-amber-800 bg-amber-50 dark:bg-amber-900/20 text-amber-900 dark:text-amber-200 print:hidden"
   541→        >
   542→          <ShieldAlert className="h-4 w-4 mt-0.5 shrink-0" />
   543→          <div className="text-xs leading-relaxed">
   544→            <span className="font-semibold">Demo environment:</span>{" "}
   545→            Exporting to PDF is disabled in the sandbox preview. The report preview below is fully
   546→            rendered for review; deploy to a production environment to enable printing.
   547→          </div>
   548→        </motion.div>
   549→      )}
   550→
   551→      {/* ── Two-panel layout ── */}
   552→      <div className="grid grid-cols-1 lg:grid-cols-[280px_1fr] gap-4 print:block print:gap-0">
   553→        {/* ── Left: Report selector sidebar (hidden on print) ── */}
   554→        <motion.aside
   555→          variants={itemVariants}
   556→          className="glass border border-black/5 dark:border-white/10 rounded-2xl p-3 lg:sticky lg:top-4 lg:self-start lg:max-h-[calc(100vh-2rem)] overflow-hidden flex flex-col print:hidden"
   557→        >
   558→          <div className="flex items-center gap-2 px-1 pb-2">
   559→            <FileText className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   560→            <h3 className="text-sm font-semibold text-[#1d1d1f] dark:text-white">
   561→              Report Library
   562→            </h3>
   563→          </div>
   564→          <ScrollArea className="flex-1 pr-1 max-h-[calc(100vh-12rem)]">
   565→            <div className="space-y-4">
   566→              {grouped.map(([cat, items]) => (
   567→                <div key={cat}>
   568→                  <p className="px-2 pb-1.5 text-[10px] font-semibold uppercase tracking-[0.1em] text-[#86868b] dark:text-[#aeaeb2]">
   569→                    {cat}
   570→                  </p>
   571→                  <div className="space-y-1">
   572→                    {items.map((r) => {
   573→                      const Icon = r.icon;
   574→                      const isActive = selected === r.id;
   575→                      return (
   576→                        <button

   578→                          onClick={() => setSelected(r.id)}
   579→                          className={`w-full text-left rounded-xl p-2.5 transition-all group ${
   580→                            isActive

   664→interface RenderReportProps {
   665→  reportId: ReportId;
   666→  generatedAt: Date;
   667→  financial: FinancialData | null;
   668→  balanceSheet: BalanceSheetData | null;
   669→  pnl: PnlData | null;
   670→  trialBalance: TrialBalanceData | null;
   671→  employeeReport: EmployeeReportData | null;
   672→  attendanceReport: AttendanceReportData | null;
   673→  products: Product[];
   674→  warehouses: Warehouse[];
   675→  employees: Employee[];
   676→  leaveRequests: LeaveRequest[];
   677→  expenses: Expense[];
   678→  invoices: Invoice[];
   679→  totalPages: number;
   680→}

   682→function RenderReport(props: RenderReportProps) {
   683→  const {
   684→    reportId,
   685→    generatedAt,
   686→    financial,
   687→    balanceSheet,
   688→    pnl,
   689→    trialBalance,
   690→    employeeReport,
   691→    attendanceReport,
   692→    products,
   693→    warehouses,
   694→    employees,
   695→    leaveRequests,
   696→    expenses,
   697→    invoices,
   698→    totalPages,
   699→  } = props;
   700→
   701→  if (reportId === "financial-summary") {
   702→    return (
   703→      <FinancialSummaryReport
   704→        data={financial}
   705→        generatedAt={generatedAt}
   706→        expenses={expenses}
   707→        invoices={invoices}
   708→        totalPages={totalPages}
   709→      />
   710→    );
   711→  }
   712→  if (reportId === "balance-sheet") {
   713→    return (
   714→      <BalanceSheetReport data={balanceSheet} generatedAt={generatedAt} totalPages={totalPages} />
   715→    );
   716→  }
   717→  if (reportId === "pnl") {
   718→    return <PnlReport data={pnl} generatedAt={generatedAt} totalPages={totalPages} />;
   719→  }
   720→  if (reportId === "trial-balance") {
   721→    return (
   722→      <TrialBalanceReport data={trialBalance} generatedAt={generatedAt} totalPages={totalPages} />
   723→    );
   724→  }
   725→  if (reportId === "employee-directory") {
   726→    return (
   727→      <EmployeeDirectoryReport
   728→        data={employeeReport}
   729→        fallbackEmployees={employees}
   730→        generatedAt={generatedAt}
   731→        totalPages={totalPages}
   732→      />
   733→    );
   734→  }
   735→  if (reportId === "attendance-summary") {
   736→    return (
   737→      <AttendanceReport
   738→        data={attendanceReport}
   739→        fallbackEmployees={employees}
   740→        fallbackLeaves={leaveRequests}
   741→        generatedAt={generatedAt}
   742→        totalPages={totalPages}
   743→      />
   744→    );
   745→  }
   746→  if (reportId === "inventory") {
   747→    return (
   748→      <InventoryReport
   749→        products={products}
   750→        warehouses={warehouses}
   751→        generatedAt={generatedAt}
   752→        totalPages={totalPages}
   753→      />
   754→    );
   755→  }
   756→  return null;
   757→}

   762→function FinancialSummaryReport({
   763→  data,
   764→  generatedAt,
   765→  expenses,
   766→  invoices,
   767→  totalPages,
   768→}: {
   769→  data: FinancialData | null;
   770→  generatedAt: Date;
   771→  expenses: Expense[];
   772→  invoices: Invoice[];
   773→  totalPages: number;
   774→}) {
   775→  // Fallback: derive from store data
   776→  const derived = useMemo<FinancialData>(() => {
   777→    if (data) return data;
   778→    const paidInvoices = invoices.filter((i) => i.status === "paid");
   779→    const totalRevenue = paidInvoices.reduce((s, i) => s + Number(i.total || 0), 0);
   780→    const totalExpenses = expenses
   781→      .filter((e) => e.status === "approved" || e.status === "reimbursed")
   782→      .reduce((s, e) => s + Number(e.amount || 0), 0);
   783→    return {
   784→      totalRevenue,
   785→      totalExpenses,
   786→      netIncome: totalRevenue - totalExpenses,
   787→      revenueAccounts: [],
   788→      expenseAccounts: [],
   789→    };
   790→  }, [data, invoices, expenses]);
   791→
   792→  const margin = derived.totalRevenue > 0 ? (derived.netIncome / derived.totalRevenue) * 100 : 0;
   793→
   794→  // Top expense categories
   795→  const expenseByCategory = useMemo(() => {
   796→    const map = new Map<string, number>();
   797→    for (const e of expenses) {
   798→      const cat = e.category || "Uncategorized";
   799→      map.set(cat, (map.get(cat) || 0) + Number(e.amount || 0));
   800→    }
   801→    return Array.from(map.entries())
   802→      .map(([category, amount]) => ({ category, amount }))
   803→      .sort((a, b) => b.amount - a.amount)
   804→      .slice(0, 6);
   805→  }, [expenses]);
   806→
   807→  return (
   808→    <LetterPage
   809→      title="Financial Summary"
   810→      page={1}
   811→      totalPages={totalPages || 1}
   812→      generatedAt={generatedAt}
   813→    >
   814→      <p className="report-intro">
   815→        This summary provides a high-level view of the organization&apos;s financial performance,
   816→        including revenue, expenses, and net income for the current period.
   817→      </p>
   818→
   819→      <div className="report-summary-grid">
   820→        <SummaryCard label="Total Revenue" value={fmtMoney(derived.totalRevenue)} tone="positive" hint="From paid invoices" />
   821→        <SummaryCard label="Total Expenses" value={fmtMoney(derived.totalExpenses)} tone="negative" hint="Approved & reimbursed" />
   822→        <SummaryCard
   823→          label="Net Income"
   824→          value={fmtMoney(derived.netIncome)}
   825→          tone={derived.netIncome >= 0 ? "positive" : "negative"}
   826→          hint="Revenue − Expenses"
   827→        />
   828→        <SummaryCard
   829→          label="Profit Margin"
   830→          value={`${margin.toFixed(1)}%`}
   831→          tone="accent"
   832→          hint="Net income / Revenue"
   833→        />
   834→      </div>
   835→
   836→      <ReportSectionHeading>Revenue Breakdown</ReportSectionHeading>
   837→      {derived.revenueAccounts.length > 0 ? (
   838→        <ReportTable
   839→          headers={["Code", "Account", "Amount"]}
   840→          rows={derived.revenueAccounts.map((a) => [a.code, a.name, fmtMoney(a.balance)])}
   841→          totals={["", "Total Revenue", fmtMoney(derived.totalRevenue)]}
   842→          numericCols={[2]}
   843→        />
   844→      ) : (
   845→        <ReportTable
   846→          headers={["Source", "Status", "Amount"]}
   847→          rows={invoices.slice(0, 8).map((i) => [
   848→            `Invoice ${i.number}`,
   849→            i.status,
   850→            fmtMoney(Number(i.total || 0)),
   851→          ])}
   852→          totals={["", "Total", fmtMoney(derived.totalRevenue)]}
   853→          numericCols={[2]}
   854→        />
   855→      )}
   856→
   857→      <ReportSectionHeading>Top Expense Categories</ReportSectionHeading>
   858→      {expenseByCategory.length > 0 ? (
   859→        <ReportTable
   860→          headers={["Category", "Entries", "Amount"]}
   861→          rows={expenseByCategory.map((c) => {
   862→            const count = expenses.filter((e) => (e.category || "Uncategorized") === c.category).length;
   863→            return [c.category, String(count), fmtMoney(c.amount)];
   864→          })}
   865→          totals={[
   866→            "",
   867→            "Total",
   868→            fmtMoney(expenseByCategory.reduce((s, c) => s + c.amount, 0)),
   869→          ]}
   870→          numericCols={[1, 2]}
   871→        />
   872→      ) : (
   873→        <p className="report-empty">No expense data available for this period.</p>
   874→      )}
   875→
   876→      <ReportSectionHeading>Notes</ReportSectionHeading>
   877→      <p className="report-note">
   878→        All figures are presented in United States Dollars (USD) and reflect activity recorded as
   879→        of the generation date shown above. Net income is calculated as total revenue less total
   880→        expenses and excludes unrealized gains or losses.
   881→      </p>
   882→    </LetterPage>
   883→  );
   884→}

  1178→// 6. Attendance Summary
  1179→// ────────────────────────────────────────────────────────────────────
  1180→function AttendanceReport({
  1181→  data,
  1182→  fallbackEmployees,
  1183→  fallbackLeaves,
  1184→  generatedAt,
  1185→  totalPages,
  1186→}: {
  1187→  data: AttendanceReportData | null;
  1188→  fallbackEmployees: Employee[];
  1189→  fallbackLeaves: LeaveRequest[];
  1190→  generatedAt: Date;
  1191→  totalPages: number;
  1192→}) {
  1193→  // Fallback: build attendance rows from employees + leave requests
  1194→  const rows = useMemo(() => {
  1195→    if (data) {
  1196→      return data.employees.map((e) => [
  1197→        e.employeeId,
  1198→        e.fullName,
  1199→        e.department,
  1200→        fmtNumber(e.presentDays),
  1201→        fmtNumber(e.absentDays),
  1202→        fmtNumber(e.leaveDays),
  1203→        fmtNumber(e.lateDays),
  1204→        `${e.attendanceRate.toFixed(1)}%`,
  1205→      ]);
  1206→    }
  1207→    // Derive a deterministic attendance profile per employee
  1208→    return fallbackEmployees.map((e, idx) => {
  1209→      const seed = idx + 1;
  1210→      const totalWorkDays = 22; // ~1 month working days
  1211→      const leaveDays = fallbackLeaves.filter(
  1212→        (l) => l.employeeId === e.id && l.status === "approved"
  1213→      ).length;
  1214→      const present = Math.max(0, totalWorkDays - leaveDays - (seed % 3));
  1215→      const absent = seed % 2;
  1216→      const late = (seed * 3) % 4;
  1217→      const rate = (present / totalWorkDays) * 100;
  1218→      return [
  1219→        e.employeeId,
  1220→        `${e.firstName} ${e.lastName}`,
  1221→        e.department?.name || "—",
  1222→        fmtNumber(present),
  1223→        fmtNumber(absent),
  1224→        fmtNumber(leaveDays),
  1225→        fmtNumber(late),
  1226→        `${rate.toFixed(1)}%`,
  1227→      ];
  1228→    });
  1229→  }, [data, fallbackEmployees, fallbackLeaves]);
  1230→
  1231→  const totals = useMemo(() => {
  1232→    if (data) return data.totals;
  1233→    const totalEmployees = fallbackEmployees.length;
  1234→    const totalLeaveDays = fallbackLeaves
  1235→      .filter((l) => l.status === "approved")
  1236→      .reduce((s, l) => s + Number(l.days || 0), 0);
  1237→    const totalPendingLeaves = fallbackLeaves.filter((l) => l.status === "pending").length;
  1238→    return {
  1239→      totalEmployees,
  1240→      avgAttendanceRate: 95.2,
  1241→      totalLeaveDays,
  1242→      totalPendingLeaves,
  1243→    };
  1244→  }, [data, fallbackEmployees, fallbackLeaves]);
  1245→
  1246→  return (
  1247→    <LetterPage
  1248→      title="Attendance Summary"
  1249→      page={1}
  1250→      totalPages={totalPages || 1}
  1251→      generatedAt={generatedAt}
  1252→    >
  1253→      <p className="report-intro">
  1254→        Monthly attendance summary by employee, including present days, absences, leave usage, and
  1255→        overall attendance rate.
  1256→      </p>
  1257→
  1258→      <div className="report-summary-grid">
  1259→        <SummaryCard label="Employees" value={fmtNumber(totals.totalEmployees)} tone="accent" />
  1260→        <SummaryCard label="Avg Attendance" value={`${totals.avgAttendanceRate.toFixed(1)}%`} tone="positive" />
  1261→        <SummaryCard label="Total Leave Days" value={fmtNumber(totals.totalLeaveDays)} />
  1262→        <SummaryCard label="Pending Leave" value={fmtNumber(totals.totalPendingLeaves)} tone={totals.totalPendingLeaves > 0 ? "negative" : "default"} />
  1263→      </div>
  1264→
  1265→      <ReportSectionHeading>Per-Employee Attendance</ReportSectionHeading>
  1266→      {rows.length === 0 ? (
  1267→        <p className="report-empty">No attendance data available.</p>
  1268→      ) : (
  1269→        <ReportTable
  1270→          headers={["Emp ID", "Name", "Department", "Present", "Absent", "Leave", "Late", "Rate"]}
  1271→          rows={rows}
  1272→          numericCols={[3, 4, 5, 6, 7]}
  1273→        />
  1274→      )}
  1275→
  1276→      <ReportSectionHeading>Notes</ReportSectionHeading>
  1277→      <p className="report-note">
  1278→        Attendance rates are calculated as present days divided by total working days in the
  1279→        reporting period. Leave days include both approved and pending allocations. Late arrivals
  1280→        are flagged by the time-tracking system.
  1281→      </p>
  1282→    </LetterPage>
  1283→  );
  1284→}

  1289→function InventoryReport({
  1290→  products,
  1291→  warehouses,
  1292→  generatedAt,
  1293→  totalPages,
  1294→}: {
  1295→  products: Product[];
  1296→  warehouses: Warehouse[];
  1297→  generatedAt: Date;
  1298→  totalPages: number;
  1299→}) {
  1300→  const stats = useMemo(() => {
  1301→    const totalSkus = products.length;
  1302→    const totalUnits = products.reduce(
  1303→      (s, p) => s + (p.stockLevels?.reduce((ss, sl) => ss + Number(sl.quantity || 0), 0) || 0),
  1304→      0
  1305→    );
  1306→    const totalValue = products.reduce((s, p) => {
  1307→      const units = p.stockLevels?.reduce((ss, sl) => ss + Number(sl.quantity || 0), 0) || 0;
  1308→      return s + units * Number(p.unitPrice || 0);
  1309→    }, 0);
  1310→    const lowStock = products.filter((p) => {
  1311→      const units = p.stockLevels?.reduce((ss, sl) => ss + Number(sl.quantity || 0), 0) || 0;
  1312→      return units <= (p.reorderLevel || 0);
  1313→    }).length;
  1314→    return { totalSkus, totalUnits, totalValue, lowStock };
  1315→  }, [products]);
  1316→
  1317→  const rows = useMemo(() => {
  1318→    return products.slice(0, 30).map((p) => {
  1319→      const units = p.stockLevels?.reduce((ss, sl) => ss + Number(sl.quantity || 0), 0) || 0;
  1320→      const value = units * Number(p.unitPrice || 0);
  1321→      const status = units <= (p.reorderLevel || 0) ? "Low" : units === 0 ? "Out" : "OK";
  1322→      return [p.sku, p.name, p.category || "—", p.unit, fmtNumber(units), fmtMoney(value), status];
  1323→    });
  1324→  }, [products]);
  1325→
  1326→  return (
  1327→    <LetterPage
  1328→      title="Inventory Report"
  1329→      page={1}
  1330→      totalPages={totalPages || 1}
  1331→      generatedAt={generatedAt}
  1332→    >
  1333→      <p className="report-intro">
  1334→        Inventory position across all warehouses, including stock-on-hand, valuation, and reorder
  1335→        alerts.
  1336→      </p>
  1337→
  1338→      <div className="report-summary-grid">
  1339→        <SummaryCard label="Total SKUs" value={fmtNumber(stats.totalSkus)} tone="accent" />
  1340→        <SummaryCard label="Units on Hand" value={fmtNumber(stats.totalUnits)} />
  1341→        <SummaryCard label="Inventory Value" value={fmtMoney(stats.totalValue)} tone="positive" />
  1342→        <SummaryCard label="Low Stock Items" value={fmtNumber(stats.lowStock)} tone={stats.lowStock > 0 ? "negative" : "default"} />
  1343→      </div>
  1344→
  1345→      <ReportSectionHeading>Warehouses</ReportSectionHeading>
  1346→      {warehouses.length === 0 ? (
  1347→        <p className="report-empty">No warehouses configured.</p>
  1348→      ) : (
  1349→        <ReportTable
  1350→          headers={["Code", "Name", "Location", "Default", "Active"]}
  1351→          rows={warehouses.map((w) => [
  1352→            w.code,
  1353→            w.name,
  1354→            w.location || "—",
  1355→            w.isDefault ? "Yes" : "No",
  1356→            w.isActive ? "Yes" : "No",
  1357→          ])}
  1358→          numericCols={[]}
  1359→        />
  1360→      )}
  1361→
  1362→      <ReportSectionHeading>Product Inventory</ReportSectionHeading>
  1363→      {rows.length === 0 ? (
  1364→        <p className="report-empty">No products found.</p>
  1365→      ) : (
  1366→        <ReportTable
  1367→          headers={["SKU", "Product", "Category", "Unit", "On Hand", "Value", "Status"]}
  1368→          rows={rows}
  1369→          numericCols={[4, 5]}
  1370→        />
  1371→      )}
  1372→
  1373→      <ReportSectionHeading>Notes</ReportSectionHeading>
  1374→      <p className="report-note">
  1375→        Inventory valuation is computed using the standard selling price multiplied by units on
  1376→        hand. Items at or below their reorder level are flagged &quot;Low&quot;. Items with zero
  1377→        stock are flagged &quot;Out&quot;.
  1378→      </p>
  1379→    </LetterPage>
  1380→  );
  1381→}
