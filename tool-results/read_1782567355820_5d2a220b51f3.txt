   123→// ── KPI card component ──────────────────────────────────────────────
   124→function KPICard({
   125→  title,
   126→  value,
   127→  change,
   128→  changeLabel,
   129→  icon: Icon,
   130→  color,
   131→}: {
   132→  title: string;
   133→  value: string;
   134→  change: number;
   135→  changeLabel: string;
   136→  icon: React.ComponentType<{ className?: string }>;
   137→  color: string;
   138→}) {
   139→  const isPositive = change >= 0;
   140→  return (
   141→    <Card className="relative overflow-hidden">
   142→      <CardContent className="p-6">
   143→        <div className="flex items-start justify-between">
   144→          <div className="space-y-2">
   145→            <p className="text-sm font-medium text-muted-foreground">{title}</p>
   146→            <p className="text-3xl font-bold tracking-tight">{value}</p>
   147→            <div className="flex items-center gap-1 text-xs">
   148→              {isPositive ? (
   149→                <ArrowUpRight className="h-3 w-3 text-emerald-500" />
   150→              ) : (
   151→                <ArrowDownRight className="h-3 w-3 text-red-500" />
   152→              )}
   153→              <span className={isPositive ? "text-emerald-500 font-medium" : "text-red-500 font-medium"}>
   154→                {Math.abs(change)}%
   155→              </span>
   156→              <span className="text-muted-foreground">{changeLabel}</span>
   157→            </div>
   158→          </div>
   159→          <div
   160→            className="flex h-11 w-11 items-center justify-center rounded-xl"
   161→            style={{ backgroundColor: `${color}15` }}
   162→          >
   163→            <Icon className="h-5 w-5" style={{ color }} />
   164→          </div>
   165→        </div>
   166→      </CardContent>
   167→    </Card>
   168→  );
   169→}

   171→// ── Custom tooltip component ──────────────────────────────────────
   172→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color: string }>; label?: string }) {
   173→  if (!active || !payload) return null;
   174→  return (
   175→    <div className="rounded-lg border bg-background p-3 shadow-md">
   176→      <p className="mb-1.5 text-xs font-medium text-muted-foreground">{label}</p>
   177→      {payload.map((item, i) => (
   178→        <div key={i} className="flex items-center gap-2 text-sm">
   179→          <span className="h-2.5 w-2.5 rounded-full" style={{ backgroundColor: item.color }} />
   180→          <span className="text-muted-foreground">{item.name}:</span>
   181→          <span className="font-semibold">{item.value.toLocaleString()}</span>
   182→        </div>
   183→      ))}
   184→    </div>
   185→  );
   186→}

   201→      data.push({
   202→        date: dateStr,
   203→        likes: Math.round((120 + Math.random() * 80) * factor),
   204→        comments: Math.round((35 + Math.random() * 25) * factor),
   205→        shares: Math.round((18 + Math.random() * 12) * factor),
   206→      });
   207→    }
   208→    return data;
   209→  }, [dateRange]);
   210→
   211→  // ── Content by type ──────────────────────────────────────────────
   212→  const contentByTypeData = useMemo(() => {
   213→    const typeCounts: Record<string, number> = {};
   214→    contents.forEach((c) => {
   215→      typeCounts[c.type] = (typeCounts[c.type] || 0) + 1;
   216→    });

   218→      const types: ContentType[] = ["post", "story", "reel", "carousel", "video", "graphic", "article"];
   219→      types.forEach((t) => {
   220→        typeCounts[t] = Math.floor(Math.random() * 15) + 3;
   221→      });
   222→    }

   224→      name: type.charAt(0).toUpperCase() + type.slice(1),
   225→      value: count,

   230→  // ── Platform comparison data ──────────────────────────────────────

   246→        engagement: (platformData[sa.platform]?.engagement || 0) + eng,

   252→      platform: platform.charAt(0).toUpperCase() + platform.slice(1),

   254→    }));
   255→  }, [socialAccounts]);
