    96→function formatCurrency(amount: number | null) {

   101→function formatDate(dateStr: string | null) {
   102→  if (!dateStr) return "—"

   110→function isWarrantyExpiringSoon(expiry: string | null, days = 30): boolean {
   111→  if (!expiry) return false
   112→  const exp = new Date(expiry)
   113→  const now = new Date()
   114→  const diff = exp.getTime() - now.getTime()
   115→  const daysLeft = diff / (1000 * 60 * 60 * 24)
   116→  return daysLeft > 0 && daysLeft <= days
   117→}
   118→
   119→function isWarrantyExpired(expiry: string | null): boolean {
   120→  if (!expiry) return false
   121→  return new Date(expiry) < new Date()
   122→}
   123→
   124→function getWarrantyDaysLeft(expiry: string | null): number | null {
   125→  if (!expiry) return null
   126→  const diff = new Date(expiry).getTime() - Date.now()
   127→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   128→}
   129→
   130→function isIssuanceOverdue(issuance: AssetIssuance): boolean {
   131→  if (issuance.status !== "issued" || !issuance.expectedReturnDate) return false
   132→  return new Date(issuance.expectedReturnDate) < new Date()
   133→}
   134→
   135→function getEffectiveIssuanceStatus(issuance: AssetIssuance): string {
   136→  if (issuance.status === "returned") return "returned"
   137→  if (issuance.status === "lost") return "lost"
   138→  if (isIssuanceOverdue(issuance)) return "overdue"
   139→  return "issued"
   140→}

   150→function parseSpecifications(spec: string | null): Record<string, string> | null {
   151→  if (!spec) return null
   152→  try {
   153→    const parsed = JSON.parse(spec)
   154→    if (typeof parsed === "object" && parsed !== null) return parsed as Record<string, string>
   155→    return null
   156→  } catch {
   157→    return null

   161→function generateAssetTag(category: string, existingAssets: ITAsset[]): string {
   162→  const prefixMap: Record<string, string> = {
   163→    laptop: "IT-LAP", desktop: "IT-DST", monitor: "IT-MON", keyboard: "IT-KEY",
   164→    mouse: "IT-MOU", headphone: "IT-HDP", usb: "IT-USB", webcam: "IT-WBC",
   165→    dock: "IT-DCK", tablet: "IT-TAB", phone: "IT-PHN", printer: "IT-PRT",
   166→    router: "IT-RTR", other: "IT-OTH",
   167→  }
   168→  const prefix = prefixMap[category] || "IT-OTH"
   169→  const existing = existingAssets.filter(a => a.assetTag.startsWith(prefix))
   170→  const nextNum = existing.length + 1
   171→  return `${prefix}-${String(nextNum).padStart(3, "0")}`
   172→}

   226→function WarrantyIndicator({ expiry }: { expiry: string | null }) {
   227→  if (!expiry) return <span className="text-xs text-muted-foreground">No warranty</span>
   228→  const daysLeft = getWarrantyDaysLeft(expiry)
   229→  if (daysLeft === null) return <span className="text-xs text-muted-foreground">No warranty</span>
   230→  if (daysLeft < 0) return (
   231→    <span className="inline-flex items-center gap-1 text-xs text-rose-600 font-medium">
   232→      <AlertTriangle className="h-3 w-3" /> Expired
   233→    </span>
   234→  )
   235→  if (daysLeft <= 30) return (
   236→    <span className="inline-flex items-center gap-1 text-xs text-amber-600 font-medium">
   237→      <AlertTriangle className="h-3 w-3" /> {daysLeft}d left
   238→    </span>
   239→  )
   240→  return (
   241→    <span className="inline-flex items-center gap-1 text-xs text-emerald-600">
   242→      <Shield className="h-3 w-3" /> {daysLeft}d left
   243→    </span>
   244→  )
   245→}

   259→
   260→  const autoTag = useMemo(() => generateAssetTag(category, itAssets), [category, itAssets])
   261→
   262→  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
   263→    e.preventDefault()
   264→    setLoading(true)
   265→    const form = new FormData(e.currentTarget)
   266→
   267→    let specifications: string | null = null
   268→    if (specType === "json" && specJson.trim()) {
   269→      specifications = specJson.trim()
   270→    } else if (specType === "fields") {
   271→      const specObj: Record<string, string> = {}
   272→      if (specRam) specObj["RAM"] = specRam
   273→      if (specStorage) specObj["Storage"] = specStorage
   274→      if (specCpu) specObj["CPU"] = specCpu
   275→      if (Object.keys(specObj).length > 0) specifications = JSON.stringify(specObj)
   276→    }
   277→
   278→    try {
   279→      const res = await fetch("/api/it-assets", {
   280→        method: "POST",
   281→        headers: { "Content-Type": "application/json" },
   282→        body: JSON.stringify({
   283→          assetTag: form.get("assetTag") || autoTag,
   284→          name: form.get("name"),
   285→          category,
   286→          brand: form.get("brand") || null,
   287→          model: form.get("model") || null,
   288→          serialNumber: form.get("serialNumber") || null,
   289→          specifications,
   290→          purchaseDate: form.get("purchaseDate") || null,

   296→          status: "available",
   297→          location: form.get("location") || null,
   298→          notes: form.get("notes") || null,
   299→        }),
   300→      })

   306→    } catch {

   325→              <Label className="text-xs font-medium">Asset Tag</Label>
   326→              <Input name="assetTag" defaultValue={autoTag} className="h-9" />
