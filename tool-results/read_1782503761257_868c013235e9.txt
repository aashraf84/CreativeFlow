    80→// ─── Helper Functions ─────────────────────────────────────────────────────────
    81→
    82→function getTypeConfig(type: ContentType) {
    83→  return CONTENT_TYPES.find((t) => t.key === type) || CONTENT_TYPES[0];
    84→}
    85→
    86→function getStageConfig(stage: ContentStage) {
    87→  return STAGES.find((s) => s.key === stage) || STAGES[0];
    88→}
    89→
    90→function getNextStage(current: ContentStage): ContentStage | null {
    91→  const idx = STAGE_ORDER.indexOf(current);
    92→  if (idx < 0 || idx >= STAGE_ORDER.length - 1) return null;
    93→  return STAGE_ORDER[idx + 1];
    94→}
    95→
    96→function getInitials(name: string | null | undefined): string {
    97→  if (!name) return "?";
    98→  return name
    99→    .split(" ")
   100→    .map((n) => n[0])
   101→    .join("")
   102→    .toUpperCase()
   103→    .slice(0, 2);
   104→}
   105→
   106→function parseTags(tags: string | null): string[] {
   107→  if (!tags) return [];
   108→  return tags
   109→    .split(",")
   110→    .map((t) => t.trim())
   111→    .filter(Boolean);
   112→}
   113→
   114→function isOverdue(dueDate: string | null): boolean {
   115→  if (!dueDate) return false;
   116→  return new Date(dueDate) < new Date();
   117→}
   118→
   119→function formatDate(dateStr: string | null): string | null {
   120→  if (!dateStr) return null;
   121→  const d = new Date(dateStr);
   122→  const now = new Date();
   123→  const diffMs = d.getTime() - now.getTime();
   124→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   125→
   126→  if (diffDays === 0) return "Today";
   127→  if (diffDays === 1) return "Tomorrow";
   128→  if (diffDays === -1) return "Yesterday";
   129→  if (diffDays > 0 && diffDays <= 7) return `${diffDays}d left`;
   130→  if (diffDays < 0 && diffDays >= -7) return `${Math.abs(diffDays)}d overdue`;
   131→
   132→  return d.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   133→}

   444→// ─── Main Pipeline View ───────────────────────────────────────────────────────
   445→
   446→export function PipelineView() {
   447→  const { contents, teams, setSelectedContent, setShowCreateContent, setContents } = useERPStore();
   448→  const [viewMode, setViewMode] = useState<"kanban" | "list">("kanban");
   449→  const [filterType, setFilterType] = useState<ContentType | "all">("all");
   450→  const [filterPriority, setFilterPriority] = useState<ContentPriority | "all">("all");
   451→  const [filterTeam, setFilterTeam] = useState<string>("all");
   452→  const [movingIds, setMovingIds] = useState<Set<string>>(new Set());
   453→
   454→  // Filter contents
   455→  const filteredContents = useMemo(() => {
   456→    return contents.filter((c) => {
   457→      if (c.stage === "archived") return false;
   458→      if (filterType !== "all" && c.type !== filterType) return false;
   459→      if (filterPriority !== "all" && c.priority !== filterPriority) return false;
   460→      if (filterTeam !== "all" && c.teamId !== filterTeam) return false;
   461→      return true;
   462→    });
   463→  }, [contents, filterType, filterPriority, filterTeam]);
   464→
   465→  // Group by stage
   466→  const groupedByStage = useMemo(() => {
   467→    const grouped: Record<ContentStage, Content[]> = {
   468→      idea: [],
   469→      design: [],
   470→      edit: [],
   471→      review: [],
   472→      approve: [],
   473→      disseminate: [],
   474→    };
   475→    filteredContents.forEach((c) => {
   476→      if (grouped[c.stage]) {
   477→        grouped[c.stage].push(c);
   478→      }
   479→    });
   480→    return grouped;
   481→  }, [filteredContents]);
   482→
   483→  // Move to next stage
   484→  const handleMoveNext = useCallback(
   485→    async (id: string, nextStage: ContentStage) => {
   486→      setMovingIds((prev) => new Set(prev).add(id));
   487→
   488→      // Optimistic update
   489→      const updatedContents = contents.map((c) =>
   490→        c.id === id ? { ...c, stage: nextStage, updatedAt: new Date().toISOString() } : c
   491→      );
   492→      setContents(updatedContents);
   493→
   494→      try {
   495→        const res = await fetch(`/api/contents/${id}`, {
   496→          method: "PUT",
   497→          headers: { "Content-Type": "application/json" },
   498→          body: JSON.stringify({ stage: nextStage }),
   499→        });
   500→        if (!res.ok) {
   501→          // Revert on failure
   502→          setContents(contents);
   503→        }
   504→      } catch {
   505→        // Revert on error
   506→        setContents(contents);
   507→      } finally {
   508→        setMovingIds((prev) => {
   509→          const next = new Set(prev);
   510→          next.delete(id);
   511→          return next;
   512→        });
   513→      }
   514→    },

   734→                  <div

   763→                      <AnimatePresence mode="popLayout">
   764→                        {stageItems.length > 0 ? (

   777→                      </AnimatePresence>

   796→              <span className="w-20"></span>
   797→            </div>
   798→
   799→            {/* List items */}
   800→            <AnimatePresence mode="popLayout">
   801→              {filteredContents.length > 0 ? (
   802→                filteredContents.map((content, idx) => (
   803→                  <ContentListRow

   805→                    content={content}
