   161→   161→
   162→   162→function formatNumber(num: number): string {
   163→   163→  if (num >= 1000000) return (num / 1000000).toFixed(1) + "M";
   164→   164→  if (num >= 1000) return (num / 1000).toFixed(1) + "K";
   165→   165→  return num.toString();
   166→   166→}
   167→   167→
   168→   168→function parseEngagement(engagement: string | null): Record<string, number> | null {
   169→   169→  if (!engagement) return null;
   170→   170→  try {
   171→   171→    if (typeof engagement === "string") return JSON.parse(engagement);
   172→   172→    return engagement as unknown as Record<string, number>;
   173→   173→  } catch {
   174→   174→    return null;
   175→   175→  }
   176→   176→}

   189→   189→function formatScheduleTime(dateStr: string | null): string {
   190→   190→  if (!dateStr) return "No time set";
   191→   191→  const date = new Date(dateStr);
   192→   192→  if (isToday(date)) return `Today at ${format(date, "h:mm a")}`;
   193→   193→  if (isTomorrow(date)) return `Tomorrow at ${format(date, "h:mm a")}`;
   194→   194→  return format(date, "MMM d 'at' h:mm a");
   195→   195→}
