   103→   103→function formatFollowers(count: number): string {
   104→   104→  if (count >= 1_000_000) return `${(count / 1_000_000).toFixed(1)}M`
   105→   105→  if (count >= 1_000) return `${(count / 1_000).toFixed(1)}K`
   106→   106→  return count.toString()
   107→   107→}

   374→   374→      {/* Add Influencer Dialog */}
