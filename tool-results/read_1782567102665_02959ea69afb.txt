    87→function formatFollowers(n: number): string {
    88→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1).replace(/\.0$/, "") + "M"
    89→  if (n >= 1_000) return (n / 1_000).toFixed(1).replace(/\.0$/, "") + "K"
    90→  return n.toString()
    91→}

    93→function getInitials(name: string): string {
    94→  return name.split(" ").map(w => w[0]).join("").toUpperCase().slice(0, 2)

    97→function formatCurrency(n: number | null): string {
    98→  if (n === null || n === undefined) return "—"
    99→  return "$" + n.toLocaleString("en-US", { minimumFractionDigits: 0, maximumFractionDigits: 0 })
   100→}

   578→

   580→      nano: { label: "Nano", count: 0 },
   581→      micro: { label: "Micro", count: 0 },
   582→      midTier: { label: "Mid-Tier", count: 0 },
   583→      macro: { label: "Macro", count: 0 },
   584→      mega: { label: "Mega", count: 0 },
   585→    }
   586→    infs.forEach(i => {
   587→      if (i.followers >= 1000000) tiers.mega.count++
   588→      else if (i.followers >= 500000) tiers.macro.count++
   589→      else if (i.followers >= 100000) tiers.midTier.count++
   590→      else if (i.followers >= 10000) tiers.micro.count++

   596→      const e = platMap.get(i.platform) || { count: 0, totalEng: 0, totalRate: 0, rateCount: 0 }
   597→      e.count++; e.totalEng += i.engagementRate
   598→      if (i.rate) { e.totalRate += i.rate; e.rateCount++ }
   599→      platMap.set(i.platform, e)
   600→    })
   601→
   602→    const nicheMap = new Map<string, number>()
   603→    infs.forEach(i => { if (i.niche) i.niche.split(",").forEach(n => { const t = n.trim(); if (t) nicheMap.set(t, (nicheMap.get(t) || 0) + 1) }) })
   604→
   605→    const statusMap = new Map<string, number>()
   606→    infs.forEach(i => { statusMap.set(i.status, (statusMap.get(i.status) || 0) + 1) })
   607→
   608→    const now = new Date()
   609→    const thirtyDays = new Date(now.getTime() + 30 * 24 * 60 * 60 * 1000)
   610→    const expiring = infs.filter(i => i.contractEnd && new Date(i.contractEnd) >= now && new Date(i.contractEnd) <= thirtyDays)
   611→
   612→    return {
   613→      overview: { totalInfluencers: total, activeCollaborations: active, avgEngagement: Math.round(avgEng * 100) / 100, totalReach, avgRate: Math.round(avgRate * 100) / 100 },

   619→      followerTiers: tiers,
   620→    }
