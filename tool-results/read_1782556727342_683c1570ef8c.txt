   106→   106→function getInitials(name: string) {
   107→   107→  return name
   108→   108→    .split(" ")
   109→   109→    .map((w) => w[0])

   112→   112→    .slice(0, 2)

   115→   115→function parseJsonArray(val: string | null): string[] {
   116→   116→  if (!val) return []
   117→   117→  try {
   118→   118→    return JSON.parse(val)
   119→   119→  } catch {
   120→   120→    return []
   121→   121→  }
   122→   122→}
   123→   123→
   124→   124→function extractVariables(body: string): string[] {
   125→   125→  const matches = body.match(/\{\{(\w+)\}\}/g)
   126→   126→  if (!matches) return []
   127→   127→  return [...new Set(matches.map((m) => m.replace(/\{\{|\}\}/g, "")))]
   128→   128→}
