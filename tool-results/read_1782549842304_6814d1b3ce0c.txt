   162→function formatNumber(num: number): string {
   163→  if (num >= 1000000) return (num / 1000000).toFixed(1) + "M";
   164→  if (num >= 1000) return (num / 1000).toFixed(1) + "K";
   165→  return num.toString();
   166→}
   167→
   168→function parseEngagement(engagement: string | null): Record<string, number> | null {
   169→  if (!engagement) return null;
   170→  try {
   171→    if (typeof engagement === "string") return JSON.parse(engagement);
   172→    return engagement as unknown as Record<string, number>;
   173→  } catch {
   174→    return null;
   175→  }
   176→}
   177→
   178→function getPlatformIcon(platform: Platform, className?: string) {
   179→  const config = PLATFORM_CONFIG[platform];
   180→  if (!config) return <Globe className={className} />;
   181→  const Icon = config.icon;
   182→  return <Icon className={className} />;
   183→}
   184→
   185→function getPlatformColor(platform: Platform): string {
   186→  return PLATFORM_CONFIG[platform]?.color || "#6b7280";
   187→}
   188→
   189→function formatScheduleTime(dateStr: string | null): string {
   190→  if (!dateStr) return "No time set";
   191→  const date = new Date(dateStr);
   192→  if (isToday(date)) return `Today at ${format(date, "h:mm a")}`;
   193→  if (isTomorrow(date)) return `Tomorrow at ${format(date, "h:mm a")}`;
   194→  return format(date, "MMM d 'at' h:mm a");
   195→}

   469→  // ─── Computed Data ──────────────────────────────────────────────────────────

   471→  const publishedPosts = useMemo(() =>
   472→    posts.filter((p) => p.status === "published"),
   473→    [posts]
   474→  );
   475→
   476→  const scheduledPosts = useMemo(() =>
   477→    posts
   478→      .filter((p) => p.status === "scheduled")
   479→      .sort((a, b) => {
   480→        const aTime = a.scheduledAt ? new Date(a.scheduledAt).getTime() : Infinity;
   481→        const bTime = b.scheduledAt ? new Date(b.scheduledAt).getTime() : Infinity;
   482→        return aTime - bTime;
   483→      }),
   484→    [posts]
   485→  );
   486→
   487→  const draftPosts = useMemo(() =>
   488→    posts.filter((p) => p.status === "draft"),
   489→    [posts]
   490→  );
   491→
   492→  const failedPosts = useMemo(() =>
   493→    posts.filter((p) => p.status === "failed"),
   494→    [posts]
   495→  );
   496→
   497→  const publishedToday = useMemo(() =>
   498→    posts.filter((p) => {
   499→      if (p.status !== "published" || !p.publishedAt) return false;
   500→      return isToday(new Date(p.publishedAt));
   501→    }),
   502→    [posts]
   503→  );
   504→
   505→  const nextScheduledPost = useMemo(() => {
   506→    const now = new Date();
   507→    const upcoming = scheduledPosts
   508→      .filter((p) => p.scheduledAt && new Date(p.scheduledAt) > now)
   509→      .sort((a, b) => new Date(a.scheduledAt!).getTime() - new Date(b.scheduledAt!).getTime());
   510→    return upcoming[0] || null;
   511→  }, [scheduledPosts]);
   512→
   513→  const nextScheduledIn = useMemo(() => {
   514→    if (!nextScheduledPost?.scheduledAt) return null;
   515→    const minutes = differenceInMinutes(new Date(nextScheduledPost.scheduledAt), new Date());
   516→    if (minutes < 60) return `${minutes}m`;
   517→    if (minutes < 1440) return `${Math.floor(minutes / 60)}h ${minutes % 60}m`;
   518→    return `${Math.floor(minutes / 1440)}d`;
   519→  }, [nextScheduledPost]);
   520→
   521→  // Platform counts for filter tabs
   522→  const platformCounts = useMemo(() => {
   523→    const counts: Record<string, number> = { all: posts.length };
   524→    posts.forEach((post) => {
   525→      counts[post.platform] = (counts[post.platform] || 0) + 1;
   526→    });
   527→    return counts;
   528→  }, [posts]);
   529→
   530→  // Filtered posts
   531→  const filteredScheduledPosts = useMemo(() =>
   532→    platformFilter === "all"
   533→      ? scheduledPosts
   534→      : scheduledPosts.filter((p) => p.platform === platformFilter),
   535→    [scheduledPosts, platformFilter]
   536→  );
   537→
   538→  const filteredPublishedPosts = useMemo(() => {
   539→    let filtered = publishedPosts;
   540→    if (platformFilter !== "all") {
   541→      filtered = filtered.filter((p) => p.platform === platformFilter);
   542→    }
   543→    return filtered.sort((a, b) => {
   544→      const aTime = a.publishedAt ? new Date(a.publishedAt).getTime() : 0;
   545→      const bTime = b.publishedAt ? new Date(b.publishedAt).getTime() : 0;
   546→      return bTime - aTime;
   547→    });
   548→  }, [publishedPosts, platformFilter]);
   549→
