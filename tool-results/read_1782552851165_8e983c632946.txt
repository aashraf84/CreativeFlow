   102→// ── Helper: format follower count ──
   103→function formatFollowers(count: number): string {
   104→  if (count >= 1_000_000) return `${(count / 1_000_000).toFixed(1)}M`
   105→  if (count >= 1_000) return `${(count / 1_000).toFixed(1)}K`
   106→  return count.toString()
   107→}
   108→
   109→// ── Helper: initials ──
   110→function getInitials(name: string): string {
   111→  return name.split(/\s+/).map(w => w[0]).join("").slice(0, 2).toUpperCase()
   112→}

   114→// ── Main Component ──
   115→export function InfluencersView() {
   116→  const { influencers, clients, campaigns, setInfluencers } = useERPStore()
   117→  const { toast } = useToast()
   118→
   119→  const [search, setSearch] = useState("")
   120→  const [platformFilter, setPlatformFilter] = useState<string>("all")
   121→  const [statusFilter, setStatusFilter] = useState<string>("all")
   122→  const [viewMode, setViewMode] = useState<"grid" | "list">("grid")
   123→  const [selectedInfluencer, setSelectedInfluencer] = useState<Influencer | null>(null)
   124→  const [showAddDialog, setShowAddDialog] = useState(false)
   125→
   126→  // Filter influencers
   127→  const filteredInfluencers = useMemo(() => {
   128→    return influencers.filter(inf => {
   129→      const matchesSearch = !search ||
   130→        inf.name.toLowerCase().includes(search.toLowerCase()) ||
   131→        inf.username.toLowerCase().includes(search.toLowerCase()) ||
   132→        inf.niche?.toLowerCase().includes(search.toLowerCase())
   133→      const matchesPlatform = platformFilter === "all" || inf.platform === platformFilter
   134→      const matchesStatus = statusFilter === "all" || inf.status === statusFilter
   135→      return matchesSearch && matchesPlatform && matchesStatus
   136→    })
   137→  }, [influencers, search, platformFilter, statusFilter])
   138→
   139→  // KPIs
   140→  const totalInfluencers = influencers.length
   141→  const activeCollaborations = influencers.filter(i => i.status === "active").length
   142→  const avgEngagement = influencers.length > 0
   143→    ? influencers.reduce((sum, i) => sum + i.engagementRate, 0) / influencers.length
   144→    : 0
   145→  const totalReach = influencers.reduce((sum, i) => sum + i.followers, 0)
   146→
   147→  // Status counts
   148→  const statusCounts = useMemo(() => {
   149→    const counts: Record<string, number> = {}
   150→    Object.keys(statusConfig).forEach(s => { counts[s] = influencers.filter(i => i.status === s).length })
   151→    return counts
   152→  }, [influencers])
   153→
   154→  // API helpers
   155→  const createInfluencer = useCallback(async (data: Record<string, unknown>) => {
   156→    try {
   157→      const res = await fetch("/api/influencers", {
   158→        method: "POST",
   159→        headers: { "Content-Type": "application/json" },
   160→        body: JSON.stringify(data),
   161→      })
   162→      const json = await res.json()
   163→      if (json.influencer) {
   164→        setInfluencers([json.influencer, ...influencers])
   165→        toast({ title: "Influencer added", description: `${data.name} has been added to your roster.` })
   166→      }
   167→    } catch {
   168→      toast({ title: "Error", description: "Failed to add influencer.", variant: "destructive" })
   169→    }
   170→  }, [influencers, setInfluencers, toast])
   171→
   172→  const deleteInfluencer = useCallback(async (id: string) => {
   173→    try {
   174→      await fetch(`/api/influencers/${id}`, { method: "DELETE" })
   175→      setInfluencers(influencers.filter(i => i.id !== id))
   176→      if (selectedInfluencer?.id === id) setSelectedInfluencer(null)
   177→      toast({ title: "Influencer removed", description: "The influencer has been deleted." })
   178→    } catch {
   179→      toast({ title: "Error", description: "Failed to delete influencer.", variant: "destructive" })
   180→    }
   181→  }, [influencers, setInfluencers, toast, selectedInfluencer])
   182→
   183→  const updateInfluencer = useCallback(async (id: string, data: Record<string, unknown>) => {
   184→    try {
   185→      const res = await fetch(`/api/influencers/${id}`, {
   186→        method: "PUT",
   187→        headers: { "Content-Type": "application/json" },
   188→        body: JSON.stringify(data),
   189→      })
   190→      const json = await res.json()
   191→      if (json.influencer) {
   192→        setInfluencers(influencers.map(i => i.id === id ? json.influencer : i))
   193→        if (selectedInfluencer?.id === id) setSelectedInfluencer(json.influencer)
   194→        toast({ title: "Influencer updated", description: "Changes have been saved." })
   195→      }
   196→    } catch {
   197→      toast({ title: "Error", description: "Failed to update influencer.", variant: "destructive" })
   198→    }
   199→  }, [influencers, setInfluencers, toast, selectedInfluencer])
   200→
   201→  return (
   202→    <motion.div className="p-4 lg:p-6 space-y-6" variants={containerVariants} initial="hidden" animate="visible">
   203→      {/* Header */}
   204→      <motion.div variants={itemVariants} className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   205→        <div>
   206→          <h2 className="text-xl font-bold tracking-tight">Influencer CRM</h2>
   207→          <p className="text-sm text-muted-foreground mt-0.5">Manage influencers, collaborations, and partnerships</p>
   208→        </div>
   209→        <Button className="gap-1.5 bg-emerald-600 hover:bg-emerald-700 text-white" onClick={() => setShowAddDialog(true)}>
   210→          <UserPlus className="h-4 w-4" /> Add Influencer
   211→        </Button>
   212→      </motion.div>
   213→
   214→      {/* KPI Row */}
   215→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   216→        {[
   217→          { label: "Total Influencers", value: totalInfluencers, icon: Users, gradient: "from-emerald-500 to-emerald-600" },
   218→          { label: "Active Collaborations", value: activeCollaborations, icon: UserCheck, gradient: "from-teal-500 to-teal-600" },
   219→          { label: "Avg Engagement", value: `${avgEngagement.toFixed(1)}%`, icon: TrendingUp, gradient: "from-cyan-500 to-cyan-600" },
   220→          { label: "Total Reach", value: formatFollowers(totalReach), icon: BarChart3, gradient: "from-emerald-600 to-teal-600" },
   221→        ].map(kpi => {
   222→          const Icon = kpi.icon
   223→          return (
   224→            <motion.div key={kpi.label} variants={itemVariants}>
   225→              <Card className="shadow-sm border-border/60">
   226→                <CardContent className="p-4 flex items-center gap-3">
   227→                  <div className={cn("h-10 w-10 rounded-xl bg-gradient-to-br flex items-center justify-center shadow-sm", kpi.gradient)}>
   228→                    <Icon className="h-5 w-5 text-white" />
   229→                  </div>
   230→                  <div>
   231→                    <p className="text-[10px] font-medium text-muted-foreground uppercase tracking-wide">{kpi.label}</p>
   232→                    <p className="text-xl font-bold tracking-tight">{kpi.value}</p>
   233→                  </div>
   234→                </CardContent>
   235→              </Card>
   236→            </motion.div>
   237→          )
   238→        })}
   239→      </div>
   240→
   241→      {/* Filters Row */}
   242→      <motion.div variants={itemVariants} className="space-y-3">
   243→        <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   244→          {/* Search */}
   245→          <div className="relative flex-1 w-full sm:max-w-xs">
   246→            <Search className="absolute left-2.5 top-2.5 h-3.5 w-3.5 text-muted-foreground" />
   247→            <Input placeholder="Search influencers..." className="pl-8 h-9" value={search} onChange={e => setSearch(e.target.value)} />
   248→          </div>
   249→
   250→          {/* View mode toggle */}
   251→          <div className="flex items-center gap-1 bg-muted/50 p-1 rounded-lg">
   252→            <Button variant={viewMode === "grid" ? "default" : "ghost"} size="icon" className="h-7 w-7" onClick={() => setViewMode("grid")}>
   253→              <Grid3X3 className="h-3.5 w-3.5" />
   254→            </Button>
   255→            <Button variant={viewMode === "list" ? "default" : "ghost"} size="icon" className="h-7 w-7" onClick={() => setViewMode("list")}>
   256→              <List className="h-3.5 w-3.5" />
   257→            </Button>
   258→          </div>
   259→        </div>
   260→
   261→        {/* Platform filter */}
   262→        <div className="flex items-center gap-1.5 overflow-x-auto pb-1">
   263→          <Button
   264→            variant={platformFilter === "all" ? "default" : "outline"}
   265→            size="sm"
   266→            className={cn("h-7 text-xs gap-1 shrink-0", platformFilter === "all" && "bg-emerald-600 hover:bg-emerald-700 text-white")}
   267→            onClick={() => setPlatformFilter("all")}
   268→          >
   269→            All Platforms
   270→          </Button>
   271→          {(Object.keys(platformConfig) as Platform[]).map(platform => {
   272→            const cfg = platformConfig[platform]
   273→            const Icon = cfg.icon
   274→            return (
   275→              <Button
   276→                key={platform}
   277→                variant={platformFilter === platform ? "default" : "outline"}
   278→                size="sm"
   279→                className={cn("h-7 text-xs gap-1 shrink-0", platformFilter === platform && "bg-emerald-600 hover:bg-emerald-700 text-white")}
   280→                onClick={() => setPlatformFilter(platform)}
   281→              >
   282→                <Icon className="h-3 w-3" />
   283→                {cfg.label}
   284→              </Button>
   285→            )
   286→          })}
   287→        </div>
   288→
   289→        {/* Status filter tabs */}
   290→        <Tabs value={statusFilter} onValueChange={setStatusFilter}>
   291→          <TabsList className="h-8 flex-wrap">
   292→            <TabsTrigger value="all" className="text-xs">All ({influencers.length})</TabsTrigger>
   293→            {Object.entries(statusConfig).map(([key, cfg]) => (
   294→              <TabsTrigger key={key} value={key} className="text-xs">
   295→                {cfg.label} ({statusCounts[key] || 0})
   296→              </TabsTrigger>
   297→            ))}
   298→          </TabsList>
   299→        </Tabs>
   300→      </motion.div>
   301→
   302→      {/* Main Content - Split View */}
   303→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   304→        {/* Influencer Cards */}
   305→        <div className={cn(selectedInfluencer ? "lg:col-span-2" : "lg:col-span-3")}>
   306→          <AnimatePresence mode="popLayout">
   307→            {viewMode === "grid" ? (
   308→              <motion.div key="grid" className="grid grid-cols-1 sm:grid-cols-2 gap-3" layout>
   309→                {filteredInfluencers.map(inf => (
   310→                  <InfluencerCard
   311→                    key={inf.id}
   312→                    influencer={inf}
   313→                    isSelected={selectedInfluencer?.id === inf.id}
   314→                    onSelect={() => setSelectedInfluencer(inf)}
   315→                    onDelete={() => deleteInfluencer(inf.id)}
   316→                    onUpdateStatus={(status) => updateInfluencer(inf.id, { status })}
   317→                  />
   318→                ))}
   319→              </motion.div>
   320→            ) : (
   321→              <motion.div key="list" className="space-y-2" layout>
   322→                {filteredInfluencers.map(inf => (
   323→                  <InfluencerListItem
   324→                    key={inf.id}
   325→                    influencer={inf}
   326→                    isSelected={selectedInfluencer?.id === inf.id}
   327→                    onSelect={() => setSelectedInfluencer(inf)}
   328→                    onDelete={() => deleteInfluencer(inf.id)}
   329→                    onUpdateStatus={(status) => updateInfluencer(inf.id, { status })}
   330→                  />
   331→                ))}
   332→              </motion.div>
   333→            )}
   334→          </AnimatePresence>
   335→
   336→          {filteredInfluencers.length === 0 && (
   337→            <Card className="shadow-sm border-dashed">
   338→              <CardContent className="py-12 text-center">
   339→                <Star className="h-10 w-10 mx-auto text-muted-foreground/30 mb-3" />
   340→                <p className="text-sm font-medium text-muted-foreground">No influencers found</p>
   341→                <p className="text-xs text-muted-foreground/70 mt-1">
   342→                  {search || platformFilter !== "all" || statusFilter !== "all"
   343→                    ? "Try adjusting your filters"
   344→                    : "Add your first influencer to get started"}
   345→                </p>
   346→                {!search && platformFilter === "all" && statusFilter === "all" && (
   347→                  <Button size="sm" variant="outline" className="mt-3 gap-1.5" onClick={() => setShowAddDialog(true)}>
   348→                    <UserPlus className="h-3.5 w-3.5" /> Add Influencer
   349→                  </Button>
   350→                )}
   351→              </CardContent>
   352→            </Card>
   353→          )}
   354→        </div>
   355→
   356→        {/* Detail Panel */}
   357→        {selectedInfluencer && (
   358→          <motion.div
   359→            initial={{ opacity: 0, x: 20 }}
   360→            animate={{ opacity: 1, x: 0 }}
   361→            exit={{ opacity: 0, x: 20 }}
   362→            className="space-y-3"
   363→          >
   364→            <InfluencerDetailPanel
   365→              influencer={selectedInfluencer}
   366→              onUpdateStatus={(status) => updateInfluencer(selectedInfluencer.id, { status })}
   367→              onDelete={() => deleteInfluencer(selectedInfluencer.id)}
   368→              onClose={() => setSelectedInfluencer(null)}
   369→            />
   370→          </motion.div>
   371→        )}
   372→      </div>
   373→
   374→      {/* Add Influencer Dialog */}
   375→      <AddInfluencerDialog
   376→        open={showAddDialog}
   377→        onOpenChange={setShowAddDialog}
   378→        onSubmit={createInfluencer}
   379→        clients={clients}
   380→        campaigns={campaigns}
   381→      />
   382→    </motion.div>
   383→  )
   384→}

   386→// ── Influencer Card (Grid View) ──
   387→function InfluencerCard({ influencer, isSelected, onSelect, onDelete, onUpdateStatus }: {
   388→  influencer: Influencer
   389→  isSelected: boolean
   390→  onSelect: () => void
   391→  onDelete: () => void
   392→  onUpdateStatus: (status: string) => void
   393→}) {
   394→  const platCfg = platformConfig[influencer.platform]
   395→  const statCfg = statusConfig[influencer.status] || statusConfig.prospect
   396→  const StatusIcon = statCfg.icon
   397→  const PlatformIcon = platCfg.icon
   398→
   399→  return (
   400→    <motion.div variants={itemVariants} layout>
   401→      <Card
   402→        className={cn(
   403→          "shadow-sm cursor-pointer transition-all hover:shadow-md group",
   404→          isSelected ? "border-emerald-400 dark:border-emerald-600 ring-1 ring-emerald-400/30" : "border-border/60"
   405→        )}
   406→        onClick={onSelect}
   407→      >
   408→        <CardContent className="p-4">
   409→          {/* Header */}
   410→          <div className="flex items-start gap-3 mb-3">
   411→            <Avatar className="h-11 w-11 rounded-xl shrink-0">
   412→              <AvatarFallback className={cn("font-bold text-sm", platCfg.color)}>
   413→                {getInitials(influencer.name)}
   414→              </AvatarFallback>
   415→            </Avatar>
   416→            <div className="flex-1 min-w-0">
   417→              <div className="flex items-center gap-2">
   418→                <h4 className="text-sm font-semibold truncate">{influencer.name}</h4>
   419→                <div className={cn("h-5 w-5 rounded-full flex items-center justify-center shrink-0", platCfg.color)}>
   420→                  <PlatformIcon className="h-2.5 w-2.5" />
   421→                </div>
   422→              </div>
   423→              <p className="text-[11px] text-muted-foreground">@{influencer.username}</p>
   424→            </div>
   425→            <Badge className={cn("text-[9px] shrink-0", statCfg.color)}>
   426→              <StatusIcon className="h-2.5 w-2.5 mr-0.5" />
   427→              {statCfg.label}
   428→            </Badge>
   429→          </div>
   430→
   431→          {/* Stats */}
   432→          <div className="grid grid-cols-3 gap-2 mb-3">
   433→            <div className="text-center p-2 rounded-lg bg-muted/50">
   434→              <p className="text-sm font-bold">{formatFollowers(influencer.followers)}</p>
   435→              <p className="text-[9px] text-muted-foreground">Followers</p>
   436→            </div>
   437→            <div className="text-center p-2 rounded-lg bg-muted/50">
   438→              <p className="text-sm font-bold">{influencer.engagementRate.toFixed(1)}%</p>
   439→              <p className="text-[9px] text-muted-foreground">Engagement</p>
   440→            </div>
   441→            <div className="text-center p-2 rounded-lg bg-muted/50">
   442→              <p className="text-sm font-bold">{influencer.rate ? `$${influencer.rate.toLocaleString()}` : "—"}</p>
   443→              <p className="text-[9px] text-muted-foreground">Rate</p>
   444→            </div>
   445→          </div>
   446→
   447→          {/* Engagement bar */}
   448→          <div className="mb-3">
   449→            <div className="flex items-center justify-between mb-1">
   450→              <span className="text-[9px] text-muted-foreground">Engagement Rate</span>
   451→              <span className="text-[9px] font-semibold text-emerald-600 dark:text-emerald-400">{influencer.engagementRate.toFixed(1)}%</span>
   452→            </div>
   453→            <Progress value={Math.min(influencer.engagementRate * 10, 100)} className="h-1.5" />
   454→          </div>
   455→
   456→          {/* Niche & Assignments */}
   457→          <div className="flex flex-wrap items-center gap-1.5">
   458→            {influencer.niche && (
   459→              <Badge variant="secondary" className="text-[9px]">{influencer.niche}</Badge>
   460→            )}
   461→            {influencer.client && (
   462→              <Badge variant="outline" className="text-[9px] gap-1">
   463→                <Briefcase className="h-2.5 w-2.5" /> {influencer.client.name}
   464→              </Badge>
   465→            )}
   466→          </div>
