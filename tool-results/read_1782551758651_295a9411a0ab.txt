   215→// ── Activity icon helper ────────────────────────────────────────────
   216→function getActivityIcon(action: string, entity?: string) {
   217→  const lower = action.toLowerCase();
   218→  const entityLower = (entity ?? "").toLowerCase();
   219→
   220→  // Approval-specific icons
   221→  if (lower.includes("approve") || lower.includes("accept") || lower.includes("complete"))
   222→    return { icon: CheckCircle2, color: "text-green-600 dark:text-green-400 bg-green-100 dark:bg-green-900/40" };
   223→  if (lower.includes("reject") || lower.includes("deny"))
   224→    return { icon: ShieldCheck, color: "text-red-600 dark:text-red-400 bg-red-100 dark:bg-red-900/40" };
   225→  if (lower.includes("review"))
   226→    return { icon: Eye, color: "text-amber-600 dark:text-amber-400 bg-amber-100 dark:bg-amber-900/40" };
   227→
   228→  // Invoice-specific icons
   229→  if (entityLower.includes("invoice") || lower.includes("invoice") || lower.includes("payment"))
   230→    return { icon: Receipt, color: "text-emerald-600 dark:text-emerald-400 bg-emerald-100 dark:bg-emerald-900/40" };
   231→
   232→  if (lower.includes("create") || lower.includes("add") || lower.includes("new"))
   233→    return { icon: Plus, color: "text-emerald-600 dark:text-emerald-400 bg-emerald-100 dark:bg-emerald-900/40" };
   234→  if (lower.includes("edit") || lower.includes("update") || lower.includes("modify"))
   235→    return { icon: Edit3, color: "text-teal-600 dark:text-teal-400 bg-teal-100 dark:bg-teal-900/40" };
   236→  if (lower.includes("delete") || lower.includes("remove"))
   237→    return { icon: Trash2, color: "text-rose-600 dark:text-rose-400 bg-rose-100 dark:bg-rose-900/40" };
   238→  if (lower.includes("comment") || lower.includes("reply"))
   239→    return { icon: MessageSquare, color: "text-violet-600 dark:text-violet-400 bg-violet-100 dark:bg-violet-900/40" };
   240→  if (lower.includes("assign") || lower.includes("member") || lower.includes("join"))
   241→    return { icon: UserPlus, color: "text-cyan-600 dark:text-cyan-400 bg-cyan-100 dark:bg-cyan-900/40" };
   242→  if (lower.includes("schedule") || lower.includes("calendar"))
   243→    return { icon: CalendarCheck, color: "text-indigo-600 dark:text-indigo-400 bg-indigo-100 dark:bg-indigo-900/40" };
   244→  if (lower.includes("publish") || lower.includes("share") || lower.includes("send"))
   245→    return { icon: Share2, color: "text-sky-600 dark:text-sky-400 bg-sky-100 dark:bg-sky-900/40" };
   246→  return { icon: Activity, color: "text-slate-600 dark:text-slate-400 bg-slate-100 dark:bg-slate-900/40" };
   247→}

   249→// ── Entity type badge helper ────────────────────────────────────────
   250→function getEntityTypeBadge(entity: string | null | undefined): { label: string; color: string } | null {
   251→  if (!entity) return null;
   252→  const lower = entity.toLowerCase();
   253→  if (lower.includes("content")) return { label: "Content", color: "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300" };
   254→  if (lower.includes("post")) return { label: "Post", color: "bg-teal-100 text-teal-700 dark:bg-teal-900/40 dark:text-teal-300" };
   255→  if (lower.includes("campaign")) return { label: "Campaign", color: "bg-cyan-100 text-cyan-700 dark:bg-cyan-900/40 dark:text-cyan-300" };
   256→  if (lower.includes("approval")) return { label: "Approval", color: "bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300" };
   257→  if (lower.includes("invoice") || lower.includes("payment")) return { label: "Invoice", color: "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300" };
   258→  if (lower.includes("team")) return { label: "Team", color: "bg-violet-100 text-violet-700 dark:bg-violet-900/40 dark:text-violet-300" };
   259→  if (lower.includes("client")) return { label: "Client", color: "bg-sky-100 text-sky-700 dark:bg-sky-900/40 dark:text-sky-300" };
   260→  if (lower.includes("asset")) return { label: "Asset", color: "bg-rose-100 text-rose-700 dark:bg-rose-900/40 dark:text-rose-300" };
   261→  return null;
   262→}

   264→// ── Time formatting ─────────────────────────────────────────────────
   265→function formatRelativeTime(dateStr: string): string {
   266→  try {
   267→    const date = new Date(dateStr);
   268→    const now = new Date();
   269→    const diffMs = now.getTime() - date.getTime();
   270→    const diffMins = Math.floor(diffMs / 60000);
   271→    const diffHours = Math.floor(diffMs / 3600000);
   272→    const diffDays = Math.floor(diffMs / 86400000);
   273→
   274→    if (diffMins < 1) return "Just now";
   275→    if (diffMins < 60) return `${diffMins}m ago`;
   276→    if (diffHours < 24) return `${diffHours}h ago`;
   277→    if (diffDays < 7) return `${diffDays}d ago`;
   278→    return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   279→  } catch {
   280→    return "";
   281→  }
   282→}

   329→              </p>
   330→              <div className="mt-2 flex items-baseline gap-2">
   331→                <h3 className="text-2xl font-bold tracking-tight text-foreground">
   332→                  {typeof data.value === "number" ? data.value.toLocaleString() : data.value}
   333→                </h3>
   334→                {data.badge && (
   335→                  <span className={`inline-flex items-center px-1.5 py-0.5 text-[10px] font-semibold rounded-md leading-none ${data.badgeColor}`}>
   336→                    {data.badge}
   337→                  </span>
   338→                )}
   339→              </div>
   340→              {data.change !== undefined && (
   341→                <div className="mt-1.5 flex items-center gap-1">
   342→                  {data.change >= 0 ? (
   343→                    <ArrowUpRight className="h-3 w-3 text-emerald-600 dark:text-emerald-400" />
   344→                  ) : (
   345→                    <ArrowDownRight className="h-3 w-3 text-rose-600 dark:text-rose-400" />
   346→                  )}
   347→                  <span className={`text-xs font-medium ${data.change >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400"}`}>
   348→                    {Math.abs(data.change)}%
   349→                  </span>
   350→                  {data.changeLabel && (
   351→                    <span className="text-[10px] text-muted-foreground">{data.changeLabel}</span>
   352→                  )}
   353→                </div>
   354→              )}
   355→            </div>
   356→            <div className={`h-10 w-10 rounded-xl bg-gradient-to-br ${data.gradient} flex items-center justify-center shadow-lg shadow-emerald-500/20 shrink-0`}>
   357→              <Icon className={`h-5 w-5 ${data.iconColor}`} />
   358→            </div>
   359→          </div>
   360→        </CardContent>
   361→        {/* Subtle decorative gradient overlay at bottom */}
   362→        <div className="absolute bottom-0 left-0 right-0 h-px bg-gradient-to-r from-transparent via-emerald-500/20 to-transparent" />
   363→      </Card>
   364→    </motion.div>
   365→  );
   366→}

   369→function TeamProductivityBar({ team, completed, pending }: { team: string; completed: number; pending: number }) {

   377→        <span className="font-medium text-foreground truncate">{team}</span>
   378→        <span className="text-muted-foreground tabular-nums">
   379→          {completed}<span className="text-muted-foreground/60">/{total}</span>
   380→        </span>
   381→      </div>
   382→      <div className="flex h-2.5 w-full overflow-hidden rounded-full bg-muted/60">
