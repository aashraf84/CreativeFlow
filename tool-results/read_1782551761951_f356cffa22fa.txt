   215→   215→// ── Activity icon helper ────────────────────────────────────────────
   216→   216→function getActivityIcon(action: string, entity?: string) {
   217→   217→  const lower = action.toLowerCase();
   218→   218→  const entityLower = (entity ?? "").toLowerCase();
   219→   219→
   220→   220→  // Approval-specific icons
   221→   221→  if (lower.includes("approve") || lower.includes("accept") || lower.includes("complete"))
   222→   222→    return { icon: CheckCircle2, color: "text-green-600 dark:text-green-400 bg-green-100 dark:bg-green-900/40" };
   223→   223→  if (lower.includes("reject") || lower.includes("deny"))
   224→   224→    return { icon: ShieldCheck, color: "text-red-600 dark:text-red-400 bg-red-100 dark:bg-red-900/40" };
   225→   225→  if (lower.includes("review"))
   226→   226→    return { icon: Eye, color: "text-amber-600 dark:text-amber-400 bg-amber-100 dark:bg-amber-900/40" };
   227→   227→
   228→   228→  // Invoice-specific icons
   229→   229→  if (entityLower.includes("invoice") || lower.includes("invoice") || lower.includes("payment"))
   230→   230→    return { icon: Receipt, color: "text-emerald-600 dark:text-emerald-400 bg-emerald-100 dark:bg-emerald-900/40" };
   231→   231→
   232→   232→  if (lower.includes("create") || lower.includes("add") || lower.includes("new"))
   233→   233→    return { icon: Plus, color: "text-emerald-600 dark:text-emerald-400 bg-emerald-100 dark:bg-emerald-900/40" };
   234→   234→  if (lower.includes("edit") || lower.includes("update") || lower.includes("modify"))
   235→   235→    return { icon: Edit3, color: "text-teal-600 dark:text-teal-400 bg-teal-100 dark:bg-teal-900/40" };
   236→   236→  if (lower.includes("delete") || lower.includes("remove"))
   237→   237→    return { icon: Trash2, color: "text-rose-600 dark:text-rose-400 bg-rose-100 dark:bg-rose-900/40" };
   238→   238→  if (lower.includes("comment") || lower.includes("reply"))
   239→   239→    return { icon: MessageSquare, color: "text-violet-600 dark:text-violet-400 bg-violet-100 dark:bg-violet-900/40" };
   240→   240→  if (lower.includes("assign") || lower.includes("member") || lower.includes("join"))
   241→   241→    return { icon: UserPlus, color: "text-cyan-600 dark:text-cyan-400 bg-cyan-100 dark:bg-cyan-900/40" };
   242→   242→  if (lower.includes("schedule") || lower.includes("calendar"))
   243→   243→    return { icon: CalendarCheck, color: "text-indigo-600 dark:text-indigo-400 bg-indigo-100 dark:bg-indigo-900/40" };
   244→   244→  if (lower.includes("publish") || lower.includes("share") || lower.includes("send"))
   245→   245→    return { icon: Share2, color: "text-sky-600 dark:text-sky-400 bg-sky-100 dark:bg-sky-900/40" };
   246→   246→  return { icon: Activity, color: "text-slate-600 dark:text-slate-400 bg-slate-100 dark:bg-slate-900/40" };
   247→   247→}

   264→   264→// ── Time formatting ─────────────────────────────────────────────────
   265→   265→function formatRelativeTime(dateStr: string): string {
   266→   266→  try {
   267→   267→    const date = new Date(dateStr);
   268→   268→    const now = new Date();
   269→   269→    const diffMs = now.getTime() - date.getTime();
   270→   270→    const diffMins = Math.floor(diffMs / 60000);
   271→   271→    const diffHours = Math.floor(diffMs / 3600000);
   272→   272→    const diffDays = Math.floor(diffMs / 86400000);
   273→   273→
   274→   274→    if (diffMins < 1) return "Just now";
   275→   275→    if (diffMins < 60) return `${diffMins}m ago`;
   276→   276→    if (diffHours < 24) return `${diffHours}h ago`;
   277→   277→    if (diffDays < 7) return `${diffDays}d ago`;
   278→   278→    return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   279→   279→  } catch {
   280→   280→    return "";
   281→   281→  }
   282→   282→}
