    79→// ─── Helpers ────────────────────────────────────────────────────────
    80→function parseJsonSafe(val: string | null | undefined): string[] {
    81→  if (!val) return []
    82→  try {
    83→    const parsed = JSON.parse(val)
    84→    return Array.isArray(parsed) ? parsed : []
    85→  } catch {
    86→    if (typeof val === "string" && val.length > 0) {
    87→      return val.split(",").map(s => s.trim()).filter(Boolean)
    88→    }
    89→    return []
    90→  }
    91→}

    93→function detectVariables(text: string): string[] {
    94→  const regex = /\{\{(\w+)\}\}/g
    95→  const vars: string[] = []
    96→  let match: RegExpExecArray | null
    97→  while ((match = regex.exec(text)) !== null) {
    98→    if (!vars.includes(match[1])) vars.push(match[1])
    99→  }
   100→  return vars
   101→}

   103→function substituteVariables(text: string, vars: Record<string, string>): string {
   104→  return text.replace(/\{\{(\w+)\}\}/g, (_, key) => vars[key] || `{{${key}}}`)
   105→}

   234→function InboxTab({ conversations, currentUser }: { conversations: Conversation[]; currentUser: { id: string; name: string; email: string } | null }) {
   235→  const [selectedId, setSelectedId] = useState<string | null>(null)
   236→  const [search, setSearch] = useState("")
   237→  const [channelFilter, setChannelFilter] = useState("all")
   238→  const [statusFilter, setStatusFilter] = useState("active")
   239→
   240→  const filtered = useMemo(() => {
   241→    return conversations.filter(c => {
   242→      if (channelFilter !== "all" && c.type !== channelFilter) return false
   243→      if (statusFilter !== "all" && c.status !== statusFilter) return false
   244→      if (search) {
   245→        const s = search.toLowerCase()
   246→        const matchSubject = c.subject?.toLowerCase().includes(s)
   247→        const matchPreview = c.lastMessagePreview?.toLowerCase().includes(s)
   248→        const matchClient = c.client?.name?.toLowerCase().includes(s)
   249→        const matchInfluencer = c.influencer?.name?.toLowerCase().includes(s)
   250→        return matchSubject || matchPreview || matchClient || matchInfluencer
   251→      }
   252→      return true
   253→    })
   254→  }, [conversations, search, channelFilter, statusFilter])
   255→
   256→  const selectedConversation = useMemo(() => {
   257→    if (!selectedId) return null
   258→    return conversations.find(c => c.id === selectedId) || null
   259→  }, [conversations, selectedId])
   260→
   261→  return (
   262→    <div className="h-full flex flex-col lg:flex-row gap-4">
   263→      {/* Left Panel - Conversation List */}
   264→      <div className="w-full lg:w-[380px] xl:w-[420px] flex-shrink-0 flex flex-col border rounded-xl bg-card overflow-hidden">

   267→            <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />

   278→                key={ch}
   279→                variant={channelFilter === ch ? "default" : "ghost"}

   281→                className={`h-7 text-xs px-2.5 ${channelFilter === ch ? "bg-emerald-500 hover:bg-emerald-600 text-white" : ""}`}
   282→                onClick={() => setChannelFilter(ch)}

   291→              <Button
   292→                key={st}
   293→                variant={statusFilter === st ? "default" : "ghost"}
   294→                size="sm"
   295→                className={`h-7 text-xs px-2.5 ${statusFilter === st ? "bg-emerald-500 hover:bg-emerald-600 text-white" : ""}`}

   319→              ))}

   322→        </ScrollArea>

   325→      {/* Right Panel - Conversation Detail */}

   328→          <ConversationDetail conversation={selectedConversation} currentUser={currentUser} />

   331→            <div className="text-center">

   335→            </div>
   336→          </div>
   337→        )}
   338→      </div>
   339→    </div>
   340→  )
   341→}

   414→function ConversationDetail({ conversation, currentUser }: {
   415→  conversation: Conversation; currentUser: { id: string; name: string; email: string } | null
   416→}) {
   417→  const [messages, setMessages] = useState<CommunicationMessage[]>([])
   418→  const [loading, setLoading] = useState(false)
   419→  const [composeType, setComposeType] = useState<string>(conversation.type === "whatsapp" ? "whatsapp" : "email")
   420→  const [composeBody, setComposeBody] = useState("")
   421→  const [composeSubject, setComposeSubject] = useState("")
   422→  const [sending, setSending] = useState(false)
   423→  const [priority, setPriority] = useState(conversation.priority)
   424→  const [status, setStatus] = useState(conversation.status)
   425→  const scrollRef = useRef<HTMLDivElement>(null)
   426→
   427→  useEffect(() => {
   428→    loadMessages()
   429→  }, [conversation.id])
   430→
   431→  useEffect(() => {
   432→    if (scrollRef.current) {
   433→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   434→    }
   435→  }, [messages])
   436→
   437→  async function loadMessages() {
   438→    setLoading(true)
   439→    try {
   440→      const res = await fetch(`/api/communications/conversations/${conversation.id}`)
   441→      const json = await res.json()
   442→      if (json.conversation?.messages) {
   443→        setMessages(json.conversation.messages)
   444→      }
   445→    } catch (e) {
   446→      console.error("Load messages error:", e)
   447→    } finally {
   448→      setLoading(false)
   449→    }
   450→  }
   451→
   452→  async function handleSend() {
   453→    if (!composeBody.trim()) return
   454→    setSending(true)
   455→    try {
   456→      const payload: Record<string, unknown> = {
   457→        body: composeBody,
   458→        type: composeType,
   459→        senderId: currentUser?.id || "user-admin",
   460→        senderName: currentUser?.name || "Sarah Chen",
   461→      }
   462→      if (composeType === "email") {
   463→        payload.recipientEmail = conversation.messages?.[0]?.recipientEmail || conversation.messages?.[0]?.senderEmail || ""
   464→        payload.subject = composeSubject || conversation.subject || ""
   465→        payload.recipientName = conversation.messages?.[0]?.recipientName || conversation.messages?.[0]?.senderName || ""
   466→      } else if (composeType === "whatsapp") {
   467→        payload.recipientPhone = conversation.messages?.find(m => m.recipientPhone)?.recipientPhone || conversation.messages?.find(m => m.senderPhone)?.senderPhone || ""
   468→        payload.recipientName = conversation.messages?.[0]?.recipientName || conversation.messages?.[0]?.senderName || ""
   469→      }
   470→
   471→      await fetch(`/api/communications/conversations/${conversation.id}/messages`, {
   472→        method: "POST",
   473→        headers: { "Content-Type": "application/json" },
   474→        body: JSON.stringify(payload),
   475→      })
   476→
   477→      setComposeBody("")
   478→      setComposeSubject("")
   479→      await loadMessages()
   480→      await refreshConversations()
   481→    } catch (e) {
   482→      console.error("Send message error:", e)
   483→    } finally {
   484→      setSending(false)
   485→    }
   486→  }

   655→function MessageBubble({ message, onToggleStar }: {
   656→  message: CommunicationMessage; onToggleStar: () => void
   657→}) {
   658→  const isOutbound = message.direction === "outbound"
   659→  const isNote = message.type === "note"
   660→  const isWA = message.type === "whatsapp"
   661→
   662→  let bubbleClass = ""
   663→  if (isNote) {
   664→    bubbleClass = "bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-800"
   665→  } else if (isWA && isOutbound) {
   666→    bubbleClass = "bg-[#DCF8C6] dark:bg-emerald-950/40 border border-[#25D366]/20"
   667→  } else if (isWA && !isOutbound) {
   668→    bubbleClass = "bg-white dark:bg-muted/50 border border-[#25D366]/10"
   669→  } else if (isOutbound) {
   670→    bubbleClass = "bg-emerald-50 dark:bg-emerald-950/30 border border-emerald-200 dark:border-emerald-800"
   671→  } else {
   672→    bubbleClass = "bg-white dark:bg-muted/50 border"
   673→  }
   674→
   675→  return (
   676→    <motion.div
   677→      initial={{ opacity: 0, y: 10 }}
   678→      animate={{ opacity: 1, y: 0 }}
   679→      className={`flex ${isOutbound || isNote ? "justify-end" : "justify-start"}`}
   680→    >
   681→      <div className={`max-w-[80%] rounded-xl p-3 ${bubbleClass}`}>
   682→        <div className="flex items-center gap-1.5 mb-1">
   683→          {isWA ? (
   684→            <MessageCircle className="h-3.5 w-3.5 text-[#25D366]" />
   685→          ) : isNote ? (
   686→            <StickyNote className="h-3.5 w-3.5 text-amber-500" />
   687→          ) : (
   688→            <Mail className="h-3.5 w-3.5 text-emerald-500" />
   689→          )}
   690→          <span className="text-xs font-medium">{message.senderName}</span>
   691→          <span className="text-[10px] text-muted-foreground">
   692→            {message.createdAt ? formatDistanceToNow(new Date(message.createdAt), { addSuffix: true }) : ""}
   693→          </span>
   694→          <button onClick={onToggleStar} className="ml-1 hover:scale-110 transition-transform">
   695→            {message.isStarred ? (
   696→              <Star className="h-3 w-3 text-amber-400 fill-amber-400" />
   697→            ) : (
   698→              <Star className="h-3 w-3 text-muted-foreground/30 hover:text-amber-400" />
   699→            )}
   700→          </button>
   701→        </div>
   702→        {message.subject && (
   703→          <p className="text-xs font-medium text-muted-foreground mb-1">{message.subject}</p>
   704→        )}
   705→        <p className="text-sm whitespace-pre-wrap">{message.body}</p>
   706→        <div className="flex items-center gap-1.5 mt-1.5 justify-end">
   707→          {getStatusIcon(message.status)}
   708→          <span className="text-[10px] text-muted-foreground/60 capitalize">{message.status}</span>
   709→        </div>
   710→      </div>
   711→    </motion.div>
   712→  )

   734→function ComposeTab({ templates, contacts, currentUser }: {
   735→  templates: CommunicationTemplate[]; contacts: CommunicationContact[]; currentUser: { id: string; name: string; email: string } | null
   736→}) {
   737→  const [channel, setChannel] = useState<"email" | "whatsapp">("email")
   738→  const [to, setTo] = useState("")
   739→  const [subject, setSubject] = useState("")
   740→  const [body, setBody] = useState("")
   741→  const [selectedTemplateId, setSelectedTemplateId] = useState<string>("")
   742→  const [variableValues, setVariableValues] = useState<Record<string, string>>({})
   743→  const [sending, setSending] = useState(false)
   744→  const [toSearch, setToSearch] = useState("")
   745→  const [showToDropdown, setShowToDropdown] = useState(false)
   746→  const [sent, setSent] = useState(false)
   747→
   748→  const selectedTemplate = templates.find(t => t.id === selectedTemplateId)
   749→
   750→  const detectedVars = useMemo(() => {
   751→    if (!selectedTemplate) return []
   752→    const allText = (selectedTemplate.subject || "") + " " + selectedTemplate.body
   753→    return detectVariables(allText)
   754→  }, [selectedTemplate])
   755→
   756→  useEffect(() => {
   757→    if (selectedTemplate) {
   758→      const subbedSubject = substituteVariables(selectedTemplate.subject || "", variableValues)
   759→      const subbedBody = substituteVariables(selectedTemplate.body, variableValues)
   760→      setSubject(subbedSubject)
   761→      setBody(subbedBody)
   762→    }
   763→  }, [selectedTemplate, variableValues])
   764→
   765→  const filteredContacts = useMemo(() => {
   766→    if (!toSearch) return contacts.slice(0, 10)
   767→    const s = toSearch.toLowerCase()
   768→    return contacts.filter(c =>
   769→      c.name.toLowerCase().includes(s) ||
   770→      (c.email?.toLowerCase().includes(s)) ||
   771→      (c.phone?.toLowerCase().includes(s)) ||
   772→      (c.company?.toLowerCase().includes(s))
   773→    ).slice(0, 10)
   774→  }, [contacts, toSearch])
   775→
   776→  function selectContact(contact: CommunicationContact) {
   777→    if (channel === "email" && contact.email) {
   778→      setTo(contact.email)
   779→    } else if (channel === "whatsapp" && contact.whatsappNumber) {
   780→      setTo(contact.whatsappNumber)
   781→    } else if (channel === "whatsapp" && contact.phone) {
   782→      setTo(contact.phone)
   783→    } else if (contact.email) {
   784→      setTo(contact.email)
   785→    }
   786→    setShowToDropdown(false)
   787→    setToSearch("")
   788→  }
   789→
   790→  async function handleSend() {
   791→    if (!to.trim() || !body.trim()) return
   792→    if (channel === "email" && !subject.trim()) return
   793→    setSending(true)
   794→    try {
   795→      const endpoint = channel === "email" ? "/api/communications/email/send" : "/api/communications/whatsapp/send"
   796→      const payload: Record<string, unknown> = {
   797→        to: to.trim(),
   798→        body,
   799→        senderId: currentUser?.id || "user-admin",
   800→        senderName: currentUser?.name || "Sarah Chen",
   801→        templateId: selectedTemplateId || undefined,
   802→      }
   803→      if (channel === "email") {
   804→        payload.subject = subject
   805→        payload.senderEmail = currentUser?.email || "sarah@vortexdigital.com"
   806→      }
   807→      const res = await fetch(endpoint, {
   808→        method: "POST",
   809→        headers: { "Content-Type": "application/json" },
   810→        body: JSON.stringify(payload),
   811→      })
   812→      if (res.ok) {
   813→        setSent(true)
   814→        setTimeout(() => setSent(false), 3000)
   815→        setTo("")
   816→        setSubject("")
   817→        setBody("")
   818→        setSelectedTemplateId("")
   819→        setVariableValues({})
   820→        await refreshConversations()
   821→      }
   822→    } catch (e) {
   823→      console.error("Compose send error:", e)
   824→    } finally {
   825→      setSending(false)
   826→    }
   827→  }
   828→
   829→  return (
   830→    <div className="max-w-3xl mx-auto">
   831→      <Card className="border-0 shadow-sm">
   832→        <CardHeader className="pb-4">
   833→          <CardTitle className="flex items-center gap-2 text-lg">
   834→            <PenSquare className="h-5 w-5 text-emerald-500" />
   835→            Compose Message
   836→          </CardTitle>
   837→        </CardHeader>
   838→        <CardContent className="space-y-4">
   839→          {/* Channel Selector */}
   840→          <div className="flex gap-2">
   841→            <Button
   842→              variant={channel === "email" ? "default" : "outline"}
   843→              className={channel === "email" ? "bg-emerald-500 hover:bg-emerald-600 text-white" : ""}
   844→              onClick={() => setChannel("email")}

   846→              <Mail className="h-4 w-4 mr-2" />Email
   847→            </Button>
   848→            <Button
   849→              variant={channel === "whatsapp" ? "default" : "outline"}

  1659→function AnalyticsTab({ conversations, templates, contacts }: {
  1660→  conversations: Conversation[]; templates: CommunicationTemplate[]; contacts: CommunicationContact[]
  1661→}) {
  1662→  // Gather all messages from conversations
  1663→  const allMessages = useMemo(() => {
  1664→    const msgs: CommunicationMessage[] = []
  1665→    conversations.forEach(c => {
  1666→      if (c.messages) msgs.push(...c.messages)
  1667→    })
  1668→    return msgs
  1669→  }, [conversations])
  1670→
  1671→  // KPIs
  1672→  const totalSent = useMemo(() => allMessages.filter(m => m.direction === "outbound").length, [allMessages])
  1673→  const emailSent = useMemo(() => allMessages.filter(m => m.direction === "outbound" && m.type === "email").length, [allMessages])
  1674→  const emailOpened = useMemo(() => allMessages.filter(m => m.direction === "outbound" && m.type === "email" && (m.status === "read" || m.openedAt)).length, [allMessages])
  1675→  const emailOpenRate = emailSent > 0 ? Math.round((emailOpened / emailSent) * 100) : 0
  1676→  const repliedCount = useMemo(() => allMessages.filter(m => m.direction === "outbound" && m.repliedAt).length, [allMessages])
  1677→  const replyRate = totalSent > 0 ? Math.round((repliedCount / totalSent) * 100) : 0
  1678→
  1679→  // Channel breakdown
  1680→  const emailCount = allMessages.filter(m => m.type === "email").length
  1681→  const waCount = allMessages.filter(m => m.type === "whatsapp").length
  1682→  const noteCount = allMessages.filter(m => m.type === "note").length
  1683→  const channelTotal = emailCount + waCount + noteCount || 1
  1684→
  1685→  // 7-day volume
  1686→  const sevenDayData = useMemo(() => {
  1687→    const days: { date: string; count: number; emailCount: number; waCount: number }[] = []
  1688→    for (let i = 6; i >= 0; i--) {
  1689→      const d = new Date()
  1690→      d.setDate(d.getDate() - i)
  1691→      const dateStr = format(d, "yyyy-MM-dd")
  1692→      const dayMsgs = allMessages.filter(m => {
  1693→        try { return format(new Date(m.createdAt), "yyyy-MM-dd") === dateStr } catch { return false }
  1694→      })
  1695→      days.push({
  1696→        date: format(d, "EEE"),
  1697→        count: dayMsgs.length,
  1698→        emailCount: dayMsgs.filter(m => m.type === "email").length,
  1699→        waCount: dayMsgs.filter(m => m.type === "whatsapp").length,
  1700→      })
  1701→    }
  1702→    return days
  1703→  }, [allMessages])
  1704→
  1705→  const maxDayCount = Math.max(...sevenDayData.map(d => d.count), 1)
  1706→
  1707→  // Top contacts by message count
  1708→  const topContacts = useMemo(() => {
  1709→    const contactCounts: Record<string, { name: string; count: number }> = {}
  1710→    allMessages.forEach(m => {
  1711→      const name = m.direction === "outbound" ? m.recipientName : m.senderName
  1712→      if (!contactCounts[name]) contactCounts[name] = { name, count: 0 }
  1713→      contactCounts[name].count++
  1714→    })
  1715→    return Object.values(contactCounts).sort((a, b) => b.count - a.count).slice(0, 5)
  1716→  }, [allMessages])
  1717→
  1718→  // Template usage
  1719→  const templateUsage = useMemo(() => {
  1720→    return templates.map(t => {
  1721→      const count = (t as Record<string, unknown>)._count
  1722→        ? ((t as Record<string, unknown>)._count as Record<string, number>)?.messages || 0
  1723→        : 0
  1724→      return { name: t.name, type: t.type, count }
  1725→    }).filter(t => t.count > 0).sort((a, b) => b.count - a.count).slice(0, 5)
  1726→  }, [templates])
  1727→
  1728→  return (
  1729→    <div className="space-y-6">
  1730→      {/* KPI Cards */}
  1731→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
  1732→        <KPICard icon={<Send className="h-5 w-5" />} label="Total Messages Sent" value={totalSent} color="emerald" />
  1733→        <KPICard icon={<MailCheck className="h-5 w-5" />} label="Email Open Rate" value={`${emailOpenRate}%`} color="emerald" />
  1734→        <KPICard icon={<MessageSquare className="h-5 w-5" />} label="Reply Rate" value={`${replyRate}%`} color="teal" />
  1735→        <KPICard icon={<Timer className="h-5 w-5" />} label="Avg Response Time" value="2.4h" color="amber" />
  1736→      </div>
  1737→
  1738→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
  1739→        {/* Channel Breakdown */}
  1740→        <Card>
  1741→          <CardHeader className="pb-3">
  1742→            <CardTitle className="text-sm font-medium flex items-center gap-2">
  1743→              <Globe className="h-4 w-4 text-emerald-500" />Channel Breakdown
  1744→            </CardTitle>

  1746→          <CardContent className="space-y-4">
