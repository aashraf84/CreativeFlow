   106→function getInitials(name: string) {
   107→  return name
   108→    .split(" ")
   109→    .map((w) => w[0])
   110→    .join("")
   111→    .toUpperCase()
   112→    .slice(0, 2)
   113→}

   115→function parseJsonArray(val: string | null): string[] {
   116→  if (!val) return []
   117→  try {
   118→    return JSON.parse(val)
   119→  } catch {
   120→    return []
   121→  }
   122→}
   123→
   124→function extractVariables(body: string): string[] {
   125→  const matches = body.match(/\{\{(\w+)\}\}/g)
   126→  if (!matches) return []
   127→  return [...new Set(matches.map((m) => m.replace(/\{\{|\}\}/g, "")))]
   128→}
