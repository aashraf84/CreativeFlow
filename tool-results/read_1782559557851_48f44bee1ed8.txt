   105→// ── Financial Report Data Type ──────────────────────────────────────
   106→interface FinancialReportData {
   107→  totalRevenue: number
   108→  totalExpenses: number
   109→  netIncome: number
   110→  expensesByCategory: Record<string, number>
   111→  accountBalancesByType: Record<string, { count: number; totalBalance: number }>
   112→  revenueAccounts: { id: string; code: string; name: string; balance: number }[]
   113→  expenseAccounts: { id: string; code: string; name: string; balance: number }[]

   115→
   116→export function FinanceAccountsView() {
   117→  const { financeAccounts, setFinanceAccounts, taxRates, setTaxRates, expenses } = useERPStore()
   118→  const [activeTab, setActiveTab] = useState("chart-of-accounts")
   119→  const [accountTypeFilter, setAccountTypeFilter] = useState("All")
   120→  const [searchQuery, setSearchQuery] = useState("")
   121→  const [showAddAccountDialog, setShowAddAccountDialog] = useState(false)
   122→  const [showAddTaxDialog, setShowAddTaxDialog] = useState(false)
   123→  const [submitting, setSubmitting] = useState(false)
   124→  const [reportData, setReportData] = useState<FinancialReportData | null>(null)
   125→  const [loadingReport, setLoadingReport] = useState(false)
   126→
   127→  // Account form state
   128→  const [accCode, setAccCode] = useState("")
   129→  const [accName, setAccName] = useState("")
   130→  const [accType, setAccType] = useState("")
   131→  const [accCategory, setAccCategory] = useState("")
   132→  const [accBalance, setAccBalance] = useState("")
   133→
   134→  // Tax rate form state
   135→  const [taxName, setTaxName] = useState("")
   136→  const [taxRate, setTaxRate] = useState("")
   137→  const [taxType, setTaxType] = useState("")
   138→  const [taxActive, setTaxActive] = useState(true)
   139→
   140→  // ── Fetch financial report data ───────────────────────────────────
   141→  const fetchReportData = useCallback(async () => {
   142→    setLoadingReport(true)
   143→    try {
   144→      const res = await fetch("/api/financial-reports")
   145→      if (res.ok) {
   146→        const json = await res.json()
   147→        setReportData(json.data as FinancialReportData)
   148→      }
   149→    } catch (err) {
   150→      console.error("Failed to fetch financial report:", err)
   151→    } finally {
   152→      setLoadingReport(false)
   153→    }
   154→  }, [])
   155→
   156→  useEffect(() => {
   157→    if (activeTab === "financial-reports" && !reportData) {
   158→      fetchReportData()
   159→    }
   160→  }, [activeTab, reportData, fetchReportData])
   161→
   162→  // ── Filtered accounts ─────────────────────────────────────────────
   163→  const filteredAccounts = useMemo(() => {
   164→    return financeAccounts.filter(a => {
   165→      if (accountTypeFilter !== "All" && a.type !== accountTypeFilter.toLowerCase()) return false
   166→      if (searchQuery && !a.name.toLowerCase().includes(searchQuery.toLowerCase()) && !a.code.toLowerCase().includes(searchQuery.toLowerCase())) return false
   167→      return true
   168→    })
   169→  }, [financeAccounts, accountTypeFilter, searchQuery])
   170→
   171→  // ── Expense breakdown data for report ─────────────────────────────
   172→  const expenseBreakdown = useMemo(() => {
   173→    if (!reportData?.expensesByCategory) return []
   174→    return Object.entries(reportData.expensesByCategory).map(([category, amount], i) => ({
   175→      category: category.charAt(0).toUpperCase() + category.slice(1),
   176→      amount: Math.round(amount * 100) / 100,
   177→      fill: CHART_COLORS[i % CHART_COLORS.length],
   178→    }))
   179→  }, [reportData])
   180→
   181→  // ── Account balances by type ──────────────────────────────────────
   182→  const accountBalances = useMemo(() => {
   183→    if (!reportData?.accountBalancesByType) return []
   184→    return Object.entries(reportData.accountBalancesByType).map(([type, data]) => ({
   185→      type: type.charAt(0).toUpperCase() + type.slice(1),
   186→      balance: Math.round(data.totalBalance * 100) / 100,
   187→      count: data.count,
   188→    }))
   189→  }, [reportData])
   190→
   191→  // ── Monthly trend (synthesized from expenses & accounts) ──────────
   192→  const monthlyTrend = useMemo(() => {
   193→    if (!reportData) return []
   194→    // Build a monthly trend from expenses data
   195→    const revenueMap = new Map<string, number>()
   196→    const expenseMap = new Map<string, number>()
   197→    expenses.forEach(e => {
   198→      const month = e.date ? format(new Date(e.date), "MMM") : "Unknown"
   199→      if (e.category === "revenue" || e.status === "approved") {
   200→        expenseMap.set(month, (expenseMap.get(month) || 0) + e.amount)
   201→      }
   202→    })
   203→    // Create a fixed set of months
   204→    const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"]
   205→    const revBase = reportData.totalRevenue / 12
   206→    const expBase = reportData.totalExpenses / 12
   207→    // Use random but deterministic variation for demo
   208→    return months.slice(0, 6).map((month, i) => ({
   209→      month,
   210→      revenue: Math.round((revBase * (0.8 + (i * 0.07))) * 100) / 100,
   211→      expenses: Math.round((expBase * (0.7 + (i * 0.1))) * 100) / 100,
   212→    }))
   213→  }, [reportData, expenses])
   214→
   215→  // ── Add Account ───────────────────────────────────────────────────
   216→  const handleAddAccount = useCallback(async () => {
   217→    if (!accCode || !accName || !accType) return
   218→    setSubmitting(true)
   219→    try {
   220→      const res = await fetch("/api/accounts", {
   221→        method: "POST",
   222→        headers: { "Content-Type": "application/json" },
   223→        body: JSON.stringify({
   224→          code: accCode,
   225→          name: accName,
   226→          type: accType,
   227→          category: accCategory || null,
   228→          balance: parseFloat(accBalance) || 0,
   229→          isActive: true,
   230→        }),
   231→      })
   232→      if (res.ok) {
   233→        const newAccount = await res.json()
   234→        setFinanceAccounts([...financeAccounts, newAccount as Account])
   235→        setShowAddAccountDialog(false)
   236→        resetAccountForm()
   237→      }
   238→    } catch (err) {
   239→      console.error("Failed to create account:", err)
   240→    } finally {
   241→      setSubmitting(false)
   242→    }
   243→  }, [accCode, accName, accType, accCategory, accBalance, financeAccounts, setFinanceAccounts])
   244→
   245→  // ── Add Tax Rate ──────────────────────────────────────────────────
   246→  const handleAddTaxRate = useCallback(async () => {
   247→    if (!taxName || !taxRate || !taxType) return
   248→    setSubmitting(true)
   249→    try {
   250→      const res = await fetch("/api/tax-rates", {
   251→        method: "POST",
   252→        headers: { "Content-Type": "application/json" },
   253→        body: JSON.stringify({
   254→          name: taxName,
   255→          rate: parseFloat(taxRate),
   256→          type: taxType,
   257→          isActive: taxActive,
   258→        }),
   259→      })
   260→      if (res.ok) {
   261→        const newTaxRate = await res.json()
   262→        setTaxRates([...taxRates, newTaxRate as TaxRate])
   263→        setShowAddTaxDialog(false)
   264→        resetTaxForm()
   265→      }
   266→    } catch (err) {
   267→      console.error("Failed to create tax rate:", err)
   268→    } finally {
   269→      setSubmitting(false)
   270→    }
   271→  }, [taxName, taxRate, taxType, taxActive, taxRates, setTaxRates])
   272→
   273→  const resetAccountForm = () => {
   274→    setAccCode("")
   275→    setAccName("")
   276→    setAccType("")
   277→    setAccCategory("")
   278→    setAccBalance("")
   279→  }
   280→
   281→  const resetTaxForm = () => {
   282→    setTaxName("")
   283→    setTaxRate("")
   284→    setTaxType("")
   285→    setTaxActive(true)
   286→  }
   287→
   288→  // ── Render type badge ─────────────────────────────────────────────
   289→  const renderTypeBadge = (type: string) => {
   290→    const config = ACCOUNT_TYPE_CONFIG[type]
   291→    if (!config) return <Badge variant="outline" className="text-xs">{type}</Badge>
   292→    const Icon = config.icon
   293→    return (
   294→      <Badge variant="outline" className={`gap-1 text-xs ${config.bg} ${config.color} border-current/20`}>
   295→        <Icon className="h-3 w-3" />
   296→        {config.label}
   297→      </Badge>
   298→    )
   299→  }
   300→
   301→  return (
   302→    <div className="p-4 md:p-6 space-y-6">
   303→      {/* ── Header ────────────────────────────────────────────────── */}
   304→      <motion.div
   305→        initial={{ opacity: 0, y: -10 }}
   306→        animate={{ opacity: 1, y: 0 }}
   307→        className="flex flex-col sm:flex-row sm:items-center justify-between gap-4"
   308→      >
   309→        <div>
   310→          <h1 className="text-2xl font-bold tracking-tight">Finance & Accounts</h1>
   311→          <p className="text-muted-foreground text-sm">Chart of accounts, financial reports, and tax management</p>
   312→        </div>
   313→      </motion.div>
   314→
   315→      {/* ── Main Tabs ─────────────────────────────────────────────── */}
   316→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   317→        <TabsList className="grid w-full max-w-lg grid-cols-3">
   318→          <TabsTrigger value="chart-of-accounts" className="gap-1.5 text-xs sm:text-sm">
   319→            <Landmark className="h-3.5 w-3.5" />
   320→            <span className="hidden sm:inline">Chart of Accounts</span>
   321→            <span className="sm:hidden">Accounts</span>
   322→          </TabsTrigger>
   323→          <TabsTrigger value="financial-reports" className="gap-1.5 text-xs sm:text-sm">
   324→            <BarChart3 className="h-3.5 w-3.5" />
   325→            <span className="hidden sm:inline">Financial Reports</span>
   326→            <span className="sm:hidden">Reports</span>
   327→          </TabsTrigger>
   328→          <TabsTrigger value="tax-rates" className="gap-1.5 text-xs sm:text-sm">
   329→            <Percent className="h-3.5 w-3.5" />
   330→            <span className="hidden sm:inline">Tax Rates</span>
   331→            <span className="sm:hidden">Tax</span>
   332→          </TabsTrigger>
   333→        </TabsList>
   334→
   335→        {/* ── Chart of Accounts Tab ─────────────────────────────────── */}
   336→        <TabsContent value="chart-of-accounts" className="mt-4 space-y-4">
   337→          {/* Account Summary KPIs */}
   338→          <motion.div
   339→            variants={containerVariants}
   340→            initial="hidden"
   341→            animate="visible"
   342→            className="grid gap-4 grid-cols-2 lg:grid-cols-4"
   343→          >
   344→            {Object.entries(ACCOUNT_TYPE_CONFIG).map(([type, config], index) => {
   345→              const accountsOfType = financeAccounts.filter(a => a.type === type)
   346→              const totalBalance = accountsOfType.reduce((sum, a) => sum + a.balance, 0)
   347→              const Icon = config.icon
   348→              return (
   349→                <motion.div key={type} variants={itemVariants}>
   350→                  <Card className="border-l-4" style={{ borderLeftColor: CHART_COLORS[index] }}>
   351→                    <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   352→                      <CardTitle className="text-sm font-medium text-muted-foreground">{config.label}</CardTitle>
   353→                      <div className="h-8 w-8 rounded-lg flex items-center justify-center" style={{ backgroundColor: `${CHART_COLORS[index]}15` }}>
   354→                        <Icon className="h-4 w-4" style={{ color: CHART_COLORS[index] }} />
   355→                      </div>
   356→                    </CardHeader>
   357→                    <CardContent>
   358→                      <div className="text-2xl font-bold">{formatCurrency(totalBalance)}</div>
   359→                      <p className="text-xs text-muted-foreground mt-1">{accountsOfType.length} account{accountsOfType.length !== 1 ? "s" : ""}</p>
   360→                    </CardContent>
   361→                  </Card>
   362→                </motion.div>
   363→              )
   364→            })}
   365→          </motion.div>
   366→
   367→          {/* Accounts Table */}
   368→          <Card>
   369→            <CardHeader className="pb-3">
   370→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   371→                <CardTitle className="text-base">Chart of Accounts</CardTitle>
   372→                <div className="flex items-center gap-2">
   373→                  <div className="relative">
   374→                    <Search className="absolute left-2.5 top-2.5 h-3.5 w-3.5 text-muted-foreground" />
   375→                    <Input
   376→                      placeholder="Search accounts..."
   377→                      className="pl-8 h-8 w-[180px] text-sm"
   378→                      value={searchQuery}
   379→                      onChange={e => setSearchQuery(e.target.value)}
   380→                    />
   381→                  </div>
   382→                  <Select value={accountTypeFilter} onValueChange={setAccountTypeFilter}>
   383→                    <SelectTrigger className="h-8 w-[140px] text-sm">
   384→                      <Filter className="h-3.5 w-3.5 mr-1" />
   385→                      <SelectValue />
   386→                    </SelectTrigger>
   387→                    <SelectContent>
   388→                      {ACCOUNT_TYPES.map(type => (
   389→                        <SelectItem key={type} value={type}>{type}</SelectItem>
   390→                      ))}
   391→                    </SelectContent>
   392→                  </Select>
   393→                  <Button
   394→                    onClick={() => setShowAddAccountDialog(true)}
   395→                    className="h-8 bg-emerald-600 hover:bg-emerald-700 text-white gap-1.5 text-xs"
   396→                  >
   397→                    <Plus className="h-3.5 w-3.5" />
   398→                    Add Account
   399→                  </Button>
   400→                </div>
   401→              </div>
   402→            </CardHeader>
   403→            <CardContent>
   404→              {filteredAccounts.length === 0 ? (
   405→                <div className="flex flex-col items-center justify-center py-16 text-center">
   406→                  <div className="h-16 w-16 rounded-full bg-muted flex items-center justify-center mb-4">
   407→                    <Landmark className="h-8 w-8 text-muted-foreground" />
   408→                  </div>
   409→                  <h3 className="font-semibold text-lg">No accounts found</h3>
   410→                  <p className="text-muted-foreground text-sm mt-1 max-w-sm">
   411→                    {searchQuery || accountTypeFilter !== "All"
   412→                      ? "Try adjusting your filters or search terms"
   413→                      : "Add your first account to get started"}
   414→                  </p>
   415→                </div>
   416→              ) : (
   417→                <ScrollArea className="max-h-[500px]">
   418→                  {/* Table header */}
   419→                  <div className="hidden md:grid md:grid-cols-[80px_1fr_120px_120px_120px_80px] gap-4 pb-3 border-b text-xs font-medium text-muted-foreground">
   420→                    <span>Code</span>
   421→                    <span>Name</span>
   422→                    <span>Type</span>
   423→                    <span>Category</span>
   424→                    <span className="text-right">Balance</span>
   425→                    <span className="text-center">Status</span>
   426→                  </div>
   427→                  <div className="space-y-0">
   428→                    <AnimatePresence>
   429→                      {filteredAccounts.map((account, index) => (
   430→                        <motion.div
   431→                          key={account.id}
   432→                          initial={{ opacity: 0, y: 8 }}
   433→                          animate={{ opacity: 1, y: 0 }}
   434→                          exit={{ opacity: 0, y: -8 }}
   435→                          transition={{ delay: index * 0.02 }}
   436→                          className="grid grid-cols-1 md:grid-cols-[80px_1fr_120px_120px_120px_80px] gap-2 md:gap-4 py-3 border-b last:border-0 hover:bg-muted/50 transition-colors items-center"
   437→                        >
   438→                          <span className="font-mono text-sm text-muted-foreground">{account.code}</span>
   439→                          <div className="flex items-center gap-2">
   440→                            <div className="h-8 w-8 rounded-lg bg-emerald-50 flex items-center justify-center shrink-0">
   441→                              {React.createElement(
   442→                                ACCOUNT_TYPE_CONFIG[account.type]?.icon || Building2,
   443→                                { className: "h-4 w-4 text-emerald-600" }
   444→                              )}
   445→                            </div>
   446→                            <span className="font-medium text-sm truncate">{account.name}</span>
   447→                          </div>
   448→                          <div>{renderTypeBadge(account.type)}</div>
   449→                          <span className="text-sm text-muted-foreground truncate">{account.category || "—"}</span>
   450→                          <span className="font-semibold text-sm text-right">{formatCurrency(account.balance)}</span>
   451→                          <div className="flex justify-center">
   452→                            {account.isActive ? (
   453→                              <Badge variant="outline" className="text-xs bg-emerald-50 text-emerald-700 border-emerald-200">
   454→                                Active
   455→                              </Badge>
   456→                            ) : (
   457→                              <Badge variant="outline" className="text-xs bg-slate-50 text-slate-500 border-slate-200">
   458→                                Inactive
   459→                              </Badge>
   460→                            )}
   461→                          </div>
   462→                        </motion.div>
   463→                      ))}
   464→                    </AnimatePresence>
   465→                  </div>
   466→                </ScrollArea>
   467→              )}
   468→            </CardContent>
   469→          </Card>
   470→        </TabsContent>
   471→
   472→        {/* ── Financial Reports Tab ─────────────────────────────────── */}
   473→        <TabsContent value="financial-reports" className="mt-4 space-y-4">
   474→          {loadingReport && !reportData ? (
   475→            <div className="flex items-center justify-center py-20">
   476→              <div className="flex flex-col items-center gap-3">
   477→                <div className="relative h-10 w-10">
   478→                  <div className="absolute inset-0 rounded-full border-4 border-muted-foreground/20" />
   479→                  <div className="absolute inset-0 rounded-full border-4 border-t-emerald-500 animate-spin" />
   480→                </div>
   481→                <p className="text-sm text-muted-foreground">Loading financial data...</p>
   482→              </div>
   483→            </div>
   484→          ) : reportData ? (
   485→            <>
   486→              {/* P&L Summary Cards */}
   487→              <motion.div
   488→                variants={containerVariants}
   489→                initial="hidden"
   490→                animate="visible"
   491→                className="grid gap-4 grid-cols-1 sm:grid-cols-3"
   492→              >
   493→                <motion.div variants={itemVariants}>
   494→                  <Card className="border-l-4 border-l-emerald-500">
   495→                    <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
