   158→   158→// ── Activity icon helper ────────────────────────────────────────────
   159→   159→function getActivityIcon(action: string) {
   160→   160→  const lower = action.toLowerCase();
   161→   161→  if (lower.includes("create") || lower.includes("add") || lower.includes("new"))
   162→   162→    return { icon: Plus, color: "text-emerald-600 dark:text-emerald-400 bg-emerald-100 dark:bg-emerald-900/40" };
   163→   163→  if (lower.includes("edit") || lower.includes("update") || lower.includes("modify"))
   164→   164→    return { icon: Edit3, color: "text-teal-600 dark:text-teal-400 bg-teal-100 dark:bg-teal-900/40" };
   165→   165→  if (lower.includes("delete") || lower.includes("remove"))
   166→   166→    return { icon: Trash2, color: "text-rose-600 dark:text-rose-400 bg-rose-100 dark:bg-rose-900/40" };
   167→   167→  if (lower.includes("approve") || lower.includes("accept") || lower.includes("complete"))
   168→   168→    return { icon: CheckCircle2, color: "text-green-600 dark:text-green-400 bg-green-100 dark:bg-green-900/40" };
   169→   169→  if (lower.includes("review") || lower.includes("view"))
   170→   170→    return { icon: Eye, color: "text-amber-600 dark:text-amber-400 bg-amber-100 dark:bg-amber-900/40" };
   171→   171→  if (lower.includes("comment") || lower.includes("reply"))
   172→   172→    return { icon: MessageSquare, color: "text-violet-600 dark:text-violet-400 bg-violet-100 dark:bg-violet-900/40" };
   173→   173→  if (lower.includes("assign") || lower.includes("member") || lower.includes("join"))
   174→   174→    return { icon: UserPlus, color: "text-cyan-600 dark:text-cyan-400 bg-cyan-100 dark:bg-cyan-900/40" };
   175→   175→  if (lower.includes("schedule") || lower.includes("calendar"))
   176→   176→    return { icon: CalendarCheck, color: "text-indigo-600 dark:text-indigo-400 bg-indigo-100 dark:bg-indigo-900/40" };
   177→   177→  if (lower.includes("publish") || lower.includes("share") || lower.includes("send"))
   178→   178→    return { icon: Share2, color: "text-sky-600 dark:text-sky-400 bg-sky-100 dark:bg-sky-900/40" };
   179→   179→  return { icon: Activity, color: "text-slate-600 dark:text-slate-400 bg-slate-100 dark:bg-slate-900/40" };
   180→   180→}

   182→   182→// ── Time formatting ─────────────────────────────────────────────────
   183→   183→function formatRelativeTime(dateStr: string): string {
   184→   184→  try {
   185→   185→    const date = new Date(dateStr);
   186→   186→    const now = new Date();
   187→   187→    const diffMs = now.getTime() - date.getTime();
   188→   188→    const diffMins = Math.floor(diffMs / 60000);
   189→   189→    const diffHours = Math.floor(diffMs / 3600000);
   190→   190→    const diffDays = Math.floor(diffMs / 86400000);
   191→   191→
   192→   192→    if (diffMins < 1) return "Just now";
   193→   193→    if (diffMins < 60) return `${diffMins}m ago`;
   194→   194→    if (diffHours < 24) return `${diffHours}h ago`;
   195→   195→    if (diffDays < 7) return `${diffDays}d ago`;
   196→   196→    return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   197→   197→  } catch {
   198→   198→    return "";
   199→   199→  }
   200→   200→}

   203→   203→function CustomTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string; color: string }>; label?: string }) {

   207→   207→      <p className="font-medium text-foreground mb-1">{label}</p>
   208→   208→      {payload.map((p, i) => (
   209→   209→        <div key={i} className="flex items-center gap-2">
   210→   210→          <div className="h-2 w-2 rounded-sm" style={{ backgroundColor: p.color }} />
   211→   211→          <span className="text-muted-foreground capitalize">{p.dataKey}:</span>
   212→   212→          <span className="font-mono font-medium text-foreground">{p.value}</span>
   213→   213→        </div>
   214→   214→      ))}
   215→   215→    </div>
   216→   216→  );

   270→   270→function TeamProductivityBar({ team, completed, pending }: { team: string; completed: number; pending: number }) {
   271→   271→  const total = completed + pending;
   272→   272→  const completedPct = total > 0 ? (completed / total) * 100 : 0;
   273→   273→  const pendingPct = total > 0 ? (pending / total) * 100 : 0;
   274→   274→
   275→   275→  return (
   276→   276→    <div className="space-y-1.5">
   277→   277→      <div className="flex items-center justify-between text-xs">
   278→   278→        <span className="font-medium text-foreground truncate">{team}</span>
   279→   279→        <span className="text-muted-foreground tabular-nums">
   280→   280→          {completed}<span className="text-muted-foreground/60">/{total}</span>
   281→   281→        </span>
   282→   282→      </div>
   283→   283→      <div className="flex h-2.5 w-full overflow-hidden rounded-full bg-muted/60">
   284→   284→        <div
   285→   285→          className="bg-gradient-to-r from-emerald-500 to-emerald-400 transition-all duration-500 rounded-l-full"
   286→   286→          style={{ width: `${completedPct}%` }}
   287→   287→        />
   288→   288→        <div
   289→   289→          className="bg-gradient-to-r from-amber-400 to-amber-300 dark:from-amber-500 dark:to-amber-400 transition-all duration-500"
   290→   290→          style={{ width: `${pendingPct}%` }}
   291→   291→        />
   292→   292→      </div>
   293→   293→    </div>
   294→   294→  );
   295→   295→}
