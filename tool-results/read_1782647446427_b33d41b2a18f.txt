    98→// ── Activity icon helper ────────────────────────────────────────────
    99→function getActivityIcon(action: string, entity?: string) {
   100→  const lower = action.toLowerCase();

   104→    return { icon: CheckCircle2, color: "text-green-600 dark:text-green-400 bg-green-100 dark:bg-green-900/40" };

   106→    return { icon: ShieldCheck, color: "text-red-600 dark:text-red-400 bg-red-100 dark:bg-red-900/40" };

   108→    return { icon: Eye, color: "text-amber-600 dark:text-amber-400 bg-amber-100 dark:bg-amber-900/40" };

   113→    return { icon: Edit3, color: "text-teal-600 dark:text-teal-400 bg-teal-100 dark:bg-teal-900/40" };

   115→    return { icon: Trash2, color: "text-rose-600 dark:text-rose-400 bg-rose-100 dark:bg-rose-900/40" };

   142→// ── Time formatting ─────────────────────────────────────────────────
   143→function formatRelativeTime(dateStr: string): string {
   144→  try {
   145→    const date = new Date(dateStr);
   146→    const now = new Date();
   147→    const diffMs = now.getTime() - date.getTime();
   148→    const diffMins = Math.floor(diffMs / 60000);
   149→    const diffHours = Math.floor(diffMs / 3600000);
   150→    const diffDays = Math.floor(diffMs / 86400000);
   151→
   152→    if (diffMins < 1) return "Just now";
   153→    if (diffMins < 60) return `${diffMins}m ago`;
   154→    if (diffHours < 24) return `${diffHours}h ago`;
   155→    if (diffDays < 7) return `${diffDays}d ago`;
   156→    return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   157→  } catch {
   158→    return "";
   159→  }
   160→}

   369→        <motion.div
   370→          className="bg-gradient-to-r from-amber-400 to-amber-300 dark:from-amber-500 dark:to-amber-400"

   373→          transition={{ duration: 0.6, ease: "easeOut", delay: 0.1 }}

   388→                backgroundColor: isCompleted ? "#10b981" : "#f59e0b30",
   389→              }}
   390→            />
   391→          );
   392→        })}
   393→      </div>
   394→    </div>
   395→  );
   396→}

   420→  const activeDisseminationTasks = disseminationTasks.filter(
   421→    (t) => t.status === "pending" || t.status === "in_progress"
   422→  ).length;
   423→
   424→  // ── Compute real KPI change percentages from data ────────────────────
   425→  const now = new Date();
   426→  const thisMonthStart = new Date(now.getFullYear(), now.getMonth(), 1);
   427→  const lastMonthStart = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   428→
   429→  const contentThisMonth = contents.filter((c) => new Date(c.createdAt) >= thisMonthStart).length;
   430→  const contentLastMonth = contents.filter((c) => {
   431→    const d = new Date(c.createdAt);
   432→    return d >= lastMonthStart && d < thisMonthStart;
   433→  }).length;
   434→  const contentChange = contentLastMonth > 0
   435→    ? Math.round(((contentThisMonth - contentLastMonth) / contentLastMonth) * 1000) / 10
   436→    : (contentThisMonth > 0 ? 100 : 0);
   437→
   438→  const postsThisMonthCalc = posts.filter((p) => {
   439→    const d = new Date(p.publishedAt || p.createdAt);
   440→    return d >= thisMonthStart;
   441→  }).length;
   442→  const postsLastMonth = posts.filter((p) => {
   443→    const d = new Date(p.publishedAt || p.createdAt);
   444→    return d >= lastMonthStart && d < thisMonthStart;
   445→  }).length;
   446→  const postsChange = postsLastMonth > 0
   447→    ? Math.round(((postsThisMonthCalc - postsLastMonth) / postsLastMonth) * 1000) / 10
   448→    : (postsThisMonthCalc > 0 ? 100 : 0);
   449→
   450→  // Build KPI cards (work-related only, 6 cards)

   452→    {
   453→      title: "Total Content",
   454→      value: dashboard.totalContent,
   455→      change: contentChange,
   456→      changeLabel: "vs last month",
   457→      icon: FileText,
   458→      gradient: "from-emerald-500 to-emerald-600",
   459→      iconColor: "text-white",
   460→    },
   461→    {
   462→      title: "Posts This Month",
   463→      value: dashboard.postsThisMonth,
   464→      change: postsChange,
   465→      changeLabel: "vs last month",
   466→      icon: Send,
   467→      gradient: "from-teal-500 to-teal-600",
   468→      iconColor: "text-white",
   469→    },
   470→    {
   471→      title: "Scheduled Posts",
   472→      value: dashboard.scheduledPosts,
   473→      icon: Clock,
   474→      gradient: "from-cyan-500 to-cyan-600",
   475→      iconColor: "text-white",
   476→      badge: "Upcoming",
   477→      badgeColor: "bg-cyan-100 text-cyan-700 dark:bg-cyan-900/40 dark:text-cyan-300",
