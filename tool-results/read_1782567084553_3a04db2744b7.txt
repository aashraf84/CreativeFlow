   327→function parseJsonField<T>(field: string | null, fallback: T): T {
   328→  if (!field) return fallback;
   329→  try {
   330→    return JSON.parse(field) as T;
   331→  } catch {
   332→    return fallback;
   333→  }
   334→}
   335→
   336→function formatNumber(num: number): string {
   337→  if (num >= 1000000) return (num / 1000000).toFixed(1) + "M";
   338→  if (num >= 1000) return (num / 1000).toFixed(1) + "K";
   339→  return num.toString();
   340→}
   341→
   342→function formatScheduleTime(dateStr: string | null): string {
   343→  if (!dateStr) return "No time set";
   344→  try {
   345→    const date = new Date(dateStr);
   346→    if (isToday(date)) return `Today at ${format(date, "h:mm a")}`;
   347→    if (isTomorrow(date)) return `Tomorrow at ${format(date, "h:mm a")}`;
   348→    return format(date, "MMM d 'at' h:mm a");
   349→  } catch {
   350→    return "Invalid date";
   351→  }
   352→}
   353→
   354→function isOverdue(task: DisseminationTask): boolean {
   355→  if (!task.dueDate) return false;
   356→  if (task.status === "completed" || task.status === "cancelled") return false;
   357→  return isPast(new Date(task.dueDate));
   358→}

   360→function getInitials(name: string): string {
   361→  return name
   362→    .split(" ")
   363→    .map((n) => n[0])
   364→    .join("")
   365→    .toUpperCase()
   366→    .slice(0, 2);
   367→}
