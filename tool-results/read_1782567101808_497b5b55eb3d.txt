   113→function getInitials(name: string): string {
   114→  return name.split(" ").filter(Boolean).map(n => n[0]).join("").toUpperCase().slice(0, 2) || "?"
   115→}
   116→
   117→function getRelativeTime(dateStr: string): string {
   118→  try {
   119→    return formatDistanceToNow(new Date(dateStr), { addSuffix: true })
   120→  } catch {
   121→    return "Unknown"
   122→  }
   123→}
   124→
   125→function getFormattedDate(dateStr: string): string {
   126→  try {
   127→    return format(new Date(dateStr), "MMM d, yyyy 'at' h:mm a")
   128→  } catch {
   129→    return "Unknown"
   130→  }
   131→}
   132→
   133→function getAvgReviewTime(approvals: Approval[]): string {
   134→  const reviewed = approvals.filter(a => a.reviewedAt && a.createdAt)
   135→  if (reviewed.length === 0) return "N/A"
   136→  const totalHours = reviewed.reduce((sum, a) => {
   137→    return sum + differenceInHours(new Date(a.reviewedAt!), new Date(a.createdAt))
   138→  }, 0)
   139→  const avg = totalHours / reviewed.length
   140→  if (avg < 1) return "< 1h"
   141→  if (avg < 24) return `${Math.round(avg)}h`
   142→  return `${(avg / 24).toFixed(1)}d`
   143→}

   475→                      {/* Reviewed */}

   502→                              <span className="text-[11px] text-muted-foreground">• {getFormattedDate(selectedApproval.reviewedAt)}</span>
   503→                            </div>
   504→                            {selectedApproval.comment && (
   505→                              <div className="mt-2 rounded-lg bg-muted/50 p-3">
   506→                                <p className="text-sm text-muted-foreground">{selectedApproval.comment}</p>
   507→                              </div>
   508→                            )}
   509→                          </div>
   510→                        </div>
   511→                      )}
   512→
   513→                      {/* Pending indicator */}
   514→                      {selectedApproval.status === "pending" && (

   524→                              {differenceInHours(new Date(), new Date(selectedApproval.createdAt))}h since request

   526→                          </div>
   527→                        </div>
   528→                      )}
   529→                    </div>
   530→                  </div>
   531→                </div>

   570→                        <p className="text-sm font-medium">

   572→                        </p>
   573→                        <Textarea
   574→                          placeholder={actionMode === "reject" ? "Provide rejection reason..." : "Describe required changes..."}
   575→                          value={comment}

   589→                            size="sm"
   590→                            className={cn(

   592→                                ? "bg-rose-500 hover:bg-rose-600 text-white"
   593→                                : "bg-cyan-500 hover:bg-cyan-600 text-white"
   594→                            )}
   595→                            onClick={() => handleApprovalAction(
   596→                              selectedApproval.id,
   597→                              actionMode === "reject" ? "rejected" : "changes_requested",

   599→                            )}
   600→                            disabled={isSubmitting}
   601→                          >
   602→                            {isSubmitting ? "Submitting..." : actionMode === "reject" ? "Reject" : "Request Changes"}
   603→                          </Button>
   604→                        </div>
   605→                      </motion.div>
   606→                    )}
   607→                  </div>
   608→                )}
   609→
   610→                {/* Already reviewed indicator */}
   611→                {selectedApproval.status !== "pending" && (
   612→                  <div className="p-4 border-t border-border/50">
   613→                    <div className={cn(
   614→                      "flex items-center justify-center gap-2 py-2 rounded-lg text-sm font-medium",
   615→                      selectedApproval.status === "approved" && "bg-emerald-50 text-emerald-600 dark:bg-emerald-950/30 dark:text-emerald-400",
   616→                      selectedApproval.status === "rejected" && "bg-rose-50 text-rose-600 dark:bg-rose-950/30 dark:text-rose-400",
   617→                      selectedApproval.status === "changes_requested" && "bg-cyan-50 text-cyan-600 dark:bg-cyan-950/30 dark:text-cyan-400",
   618→                    )}>
   619→                      {selectedApproval.status === "approved" && <CheckCircle2 className="h-4 w-4" />}
   620→                      {selectedApproval.status === "rejected" && <XCircle className="h-4 w-4" />}
   621→                      {selectedApproval.status === "changes_requested" && <RotateCcw className="h-4 w-4" />}
   622→                      {STATUS_CONFIG[selectedApproval.status]?.label}
   623→                      {selectedApproval.reviewedBy && ` by ${selectedApproval.reviewedBy.name}`}
   624→                    </div>
   625→                  </div>
   626→                )}
   627→              </Card>
   628→            </motion.div>
   629→          )}
   630→        </AnimatePresence>
   631→      </div>
   632→    </div>
   633→  )
