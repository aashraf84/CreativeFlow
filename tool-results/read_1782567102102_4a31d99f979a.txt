    61→// ── Color group helpers ──
    62→type ColorGroup = "primary" | "secondary" | "accent" | "neutral"
    63→
    64→interface ColorItem {
    65→  name: string
    66→  hex: string
    67→  group: ColorGroup
    68→}
    69→
    70→function parseColorValue(value: string): ColorItem | null {
    71→  try {
    72→    const parsed = JSON.parse(value)
    73→    if (parsed.hex && parsed.name) return parsed as ColorItem
    74→    return null
    75→  } catch {
    76→    return null
    77→  }
    78→}
    79→
    80→interface FontItem {
    81→  name: string
    82→  category: "heading" | "body" | "accent"
    83→  weights: number[]
    84→  sample: string
    85→}
    86→
    87→function parseFontValue(value: string): FontItem | null {
    88→  try {
    89→    const parsed = JSON.parse(value)
    90→    if (parsed.name) return parsed as FontItem
    91→    return null
    92→  } catch {
    93→    return null
    94→  }
    95→}
    96→
    97→interface LogoItem {
    98→  name: string
    99→  variant: "primary" | "secondary" | "icon" | "monochrome"
   100→  url: string
   101→  background: "light" | "dark"
   102→}
   103→
   104→function parseLogoValue(value: string): LogoItem | null {
   105→  try {
   106→    const parsed = JSON.parse(value)
   107→    if (parsed.name) return parsed as LogoItem
   108→    return null
   109→  } catch {
   110→    return null
   111→  }
   112→}
   113→
   114→interface VoiceItem {
   115→  description: string
   116→  dos: string[]
   117→  donts: string[]
   118→  taglines: string[]
   119→}
   120→
   121→function parseVoiceValue(value: string): VoiceItem | null {
   122→  try {
   123→    const parsed = JSON.parse(value)
   124→    if (parsed.description) return parsed as VoiceItem
   125→    return null
   126→  } catch {
   127→    return null
   128→  }
   129→}
   130→
   131→interface GuidelineItem {
   132→  title: string
   133→  description: string
   134→  documentType: string
   135→  url: string
   136→}
   137→
   138→function parseGuidelineValue(value: string): GuidelineItem | null {
   139→  try {
   140→    const parsed = JSON.parse(value)
   141→    if (parsed.title) return parsed as GuidelineItem
   142→    return null
   143→  } catch {
   144→    return null
   145→  }
   146→}
