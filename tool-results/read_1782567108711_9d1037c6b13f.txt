   145→function AddLeaveRequestDialog() {
   146→  const [open, setOpen] = useState(false)
   147→  const [loading, setLoading] = useState(false)
   148→  const { employees, leaveRequests, setLeaveRequests } = useERPStore()
   149→
   150→  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
   151→    e.preventDefault()
   152→    setLoading(true)
   153→    const form = new FormData(e.currentTarget)
   154→    const startDate = form.get("startDate") as string
   155→    const endDate = form.get("endDate") as string
   156→    const start = new Date(startDate)
   157→    const end = new Date(endDate)
   158→    const days = Math.max(1, Math.ceil((end.getTime() - start.getTime()) / (1000 * 60 * 60 * 24)) + 1)
   159→
   160→    try {
   161→      const res = await fetch("/api/leave-requests", {
   162→        method: "POST",
   163→        headers: { "Content-Type": "application/json" },
   164→        body: JSON.stringify({
   165→          employeeId: form.get("employeeId"),
   166→          type: form.get("type"),
   167→          startDate,
   168→          endDate,
   169→          days,
   170→          reason: form.get("reason"),
   171→          status: "pending",
   172→        }),
   173→      })
   174→      if (res.ok) {
   175→        const lr = await res.json()
   176→        setLeaveRequests([...leaveRequests, lr])
   177→        setOpen(false)
   178→      }
   179→    } catch {
   180→      // handle error silently
   181→    } finally {
   182→      setLoading(false)
   183→    }
   184→  }
   185→
   186→  return (
   187→    <Dialog open={open} onOpenChange={setOpen}>
   188→      <DialogTrigger asChild>
   189→        <Button size="sm" variant="outline" className="gap-2">
   190→          <UserPlus className="h-4 w-4" /> Add Leave Request
   191→        </Button>
   192→      </DialogTrigger>
   193→      <DialogContent className="sm:max-w-[450px]">
   194→        <DialogHeader>
   195→          <DialogTitle className="text-lg font-semibold">New Leave Request</DialogTitle>
   196→        </DialogHeader>
   197→        <form onSubmit={handleSubmit} className="space-y-4">
   198→          <div className="space-y-2">
   199→            <Label htmlFor="employeeId">Employee</Label>
   200→            <Select name="employeeId" required>
   201→              <SelectTrigger><SelectValue placeholder="Select employee..." /></SelectTrigger>
   202→              <SelectContent>
   203→                {employees.filter((e) => e.status === "active" || e.status === "probation").map((emp) => (
   204→                  <SelectItem key={emp.id} value={emp.id}>
   205→                    {emp.firstName} {emp.lastName}
   206→                  </SelectItem>
   207→                ))}
   208→              </SelectContent>
   209→            </Select>
   210→          </div>
   211→          <div className="space-y-2">
   212→            <Label htmlFor="type">Leave Type</Label>
   213→            <Select name="type" required>
   214→              <SelectTrigger><SelectValue placeholder="Select type..." /></SelectTrigger>
   215→              <SelectContent>
   216→                <SelectItem value="vacation">Vacation</SelectItem>
   217→                <SelectItem value="sick">Sick Leave</SelectItem>
   218→                <SelectItem value="personal">Personal</SelectItem>

   220→            </Select>
   221→          </div>
   222→          <div className="grid grid-cols-2 gap-4">
   223→            <div className="space-y-2">

   248→// ── Main Payroll View ───────────────────────────────────────────────
   249→export function PayrollView() {
   250→  const { payrollRuns, leaveRequests, employees, setLeaveRequests, setPayrollRuns } = useERPStore()
   251→  const [selectedRun, setSelectedRun] = useState<PayrollRun | null>(null)
   252→  const [activeTab, setActiveTab] = useState("payroll")
   253→  const [runningPayroll, setRunningPayroll] = useState(false)
   254→
   255→  // KPIs
   256→  const latestPayroll = payrollRuns[0]
   257→  const monthlyPayroll = latestPayroll?.totalNet || 0
   258→  const avgSalary = employees.length > 0
   259→    ? Math.round(employees.filter((e) => e.status !== "terminated").reduce((s, e) => s + (e.salary || 0), 0) / employees.filter((e) => e.status !== "terminated").length)
   260→    : 0
   261→  const totalDeductions = latestPayroll?.totalDeductions || 0
   262→  const netPay = latestPayroll?.totalNet || 0
   263→
   264→  // Pending leave requests
   265→  const pendingLeaves = leaveRequests.filter((lr) => lr.status === "pending")
   266→
   267→  // Leave balance data for chart
   268→  const leaveBalanceData = useMemo(() => {
   269→    const activeEmps = employees.filter((e) => e.status === "active" || e.status === "on_leave" || e.status === "probation")
   270→    return activeEmps.slice(0, 8).map((emp) => {
   271→      const empLeaves = leaveRequests.filter((lr) => lr.employeeId === emp.id && lr.status === "approved")
   272→      const used = empLeaves.reduce((sum, lr) => sum + lr.days, 0)
   273→      return { name: emp.firstName, used: Math.round(used), remaining: Math.max(0, 20 - Math.round(used)) }
   274→    })
   275→  }, [employees, leaveRequests])
   276→
   277→  const handleApproveLeave = async (lr: LeaveRequest) => {
   278→    try {
   279→      const res = await fetch(`/api/leave-requests/${lr.id}`, {
   280→        method: "PUT",
   281→        headers: { "Content-Type": "application/json" },
   282→        body: JSON.stringify({ status: "approved", approvedById: "admin", approvedAt: new Date().toISOString() }),
   283→      })
   284→      if (res.ok) {
   285→        const updated = await res.json()
   286→        setLeaveRequests(leaveRequests.map((l) => (l.id === lr.id ? { ...l, ...updated } : l)))
   287→      }
   288→    } catch {
   289→      // handle error silently
   290→    }
   291→  }
   292→
   293→  const handleRejectLeave = async (lr: LeaveRequest, comment?: string) => {
   294→    try {
   295→      const res = await fetch(`/api/leave-requests/${lr.id}`, {
   296→        method: "PUT",
   297→        headers: { "Content-Type": "application/json" },
   298→        body: JSON.stringify({ status: "rejected", approvedById: "admin", approvedAt: new Date().toISOString(), comment: comment || "Rejected" }),
   299→      })
   300→      if (res.ok) {
   301→        const updated = await res.json()
   302→        setLeaveRequests(leaveRequests.map((l) => (l.id === lr.id ? { ...l, ...updated } : l)))
   303→      }
   304→    } catch {
   305→      // handle error silently
   306→    }
   307→  }
   308→
   309→  const handleRunPayroll = async () => {
   310→    setRunningPayroll(true)
   311→    try {
   312→      const now = new Date()
   313→      const period = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}`
   314→      const activeEmps = employees.filter((e) => e.status === "active" || e.status === "on_leave" || e.status === "probation")
   315→      const items = activeEmps.map((emp) => {
   316→        const baseSalary = Math.round((emp.salary || 0) / 12)
   317→        const tax = Math.round(baseSalary * 0.28)
   318→        const deductions = Math.round(baseSalary * 0.15)
   319→        const netPay = baseSalary - tax - deductions
   320→        return { employeeId: emp.id, baseSalary, overtime: 0, bonuses: 0, deductions, tax, netPay }
   321→      })
   322→      const totalGross = items.reduce((s, i) => s + i.baseSalary + i.overtime + i.bonuses, 0)
   323→      const totalDed = items.reduce((s, i) => s + i.deductions + i.tax, 0)
   324→      const totalNet = items.reduce((s, i) => s + i.netPay, 0)
   325→
   326→      const res = await fetch("/api/payroll", {
   327→        method: "POST",
   328→        headers: { "Content-Type": "application/json" },
   329→        body: JSON.stringify({ period, status: "draft", totalGross, totalDeductions: totalDed, totalNet, items }),
   330→      })
   331→      if (res.ok) {
   332→        const newRun = await res.json()
   333→        setPayrollRuns([newRun, ...payrollRuns])
   334→      }
   335→    } catch {
   336→      // handle error silently
   337→    } finally {
   338→      setRunningPayroll(false)
   339→    }
   340→  }
   341→
   342→  return (
   343→    <div className="p-4 md:p-6 space-y-6 max-w-[1600px] mx-auto">
   344→      {/* Header */}
   345→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   346→        <div>
   347→          <h1 className="text-2xl font-bold tracking-tight">Payroll Management</h1>
   348→          <p className="text-sm text-muted-foreground mt-1">Manage compensation, payroll runs, and leave requests</p>
   349→        </div>
   350→        <Button
   351→          className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2"
   352→          onClick={handleRunPayroll}
   353→          disabled={runningPayroll}
   354→        >
   355→          <Play className="h-4 w-4" />
   356→          {runningPayroll ? "Running..." : "Run Payroll"}
   357→        </Button>
   358→      </div>
   359→
   360→      {/* KPI Cards */}
   361→      <motion.div
   362→        variants={containerVariants}
   363→        initial="hidden"
   364→        animate="visible"
   365→        className="grid grid-cols-2 md:grid-cols-4 gap-4"
   366→      >
   367→        {[
   368→          { label: "Monthly Payroll", value: formatCurrency(monthlyPayroll), icon: Banknote, color: "text-emerald-600", bg: "bg-emerald-50" },
   369→          { label: "Avg Salary", value: formatCurrency(avgSalary), icon: DollarSign, color: "text-teal-600", bg: "bg-teal-50" },
   370→          { label: "Total Deductions", value: formatCurrency(totalDeductions), icon: TrendingDown, color: "text-amber-600", bg: "bg-amber-50" },
   371→          { label: "Net Pay", value: formatCurrency(netPay), icon: Wallet, color: "text-cyan-600", bg: "bg-cyan-50" },
   372→        ].map((kpi) => (
   373→          <motion.div key={kpi.label} variants={itemVariants}>
   374→            <Card className="border-0 shadow-sm hover:shadow-md transition-shadow">
   375→              <CardContent className="p-4">
   376→                <div className="flex items-center justify-between">
   377→                  <div>
   378→                    <p className="text-xs text-muted-foreground font-medium">{kpi.label}</p>
   379→                    <p className="text-xl font-bold mt-1">{kpi.value}</p>
   380→                  </div>
   381→                  <div className={`h-10 w-10 rounded-xl ${kpi.bg} flex items-center justify-center`}>
   382→                    <kpi.icon className={`h-5 w-5 ${kpi.color}`} />
   383→                  </div>
   384→                </div>
   385→              </CardContent>
   386→            </Card>
   387→          </motion.div>
   388→        ))}
   389→      </motion.div>
   390→
   391→      {/* Main Tabs */}
   392→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   393→        <TabsList className="bg-muted/50">
   394→          <TabsTrigger value="payroll" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   395→            Payroll Runs
   396→          </TabsTrigger>
   397→          <TabsTrigger value="leave" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white relative">
   398→            Leave Requests
   399→            {pendingLeaves.length > 0 && (
   400→              <Badge className="ml-1.5 h-5 min-w-[20px] flex items-center justify-center text-[10px] bg-red-500 text-white px-1">
   401→                {pendingLeaves.length}
   402→              </Badge>
   403→            )}
   404→          </TabsTrigger>
   405→        </TabsList>
   406→
   407→        {/* Payroll Runs Tab */}
   408→        <TabsContent value="payroll" className="mt-4">
   409→          {selectedRun ? (
   410→            <PayrollRunDetail run={selectedRun} onClose={() => setSelectedRun(null)} />
   411→          ) : payrollRuns.length === 0 ? (
   412→            <Card className="border-dashed">
   413→              <CardContent className="flex flex-col items-center justify-center py-16">
   414→                <Banknote className="h-12 w-12 text-muted-foreground/40 mb-4" />
   415→                <p className="text-lg font-medium text-muted-foreground">No payroll runs</p>
   416→                <p className="text-sm text-muted-foreground/70 mt-1">Click &quot;Run Payroll&quot; to create your first run</p>
   417→              </CardContent>
   418→            </Card>
   419→          ) : (
   420→            <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-3">
   421→              {payrollRuns.map((run) => {
   422→                const statusCfg = STATUS_CONFIG[run.status] || STATUS_CONFIG.draft
   423→                return (
   424→                  <motion.div key={run.id} variants={itemVariants}>
   425→                    <Card
   426→                      className="border-0 shadow-sm hover:shadow-md transition-all cursor-pointer group"
   427→                      onClick={() => setSelectedRun(run)}
   428→                    >
   429→                      <CardContent className="p-4">
   430→                        <div className="flex items-center justify-between">
   431→                          <div className="flex items-center gap-4">
   432→                            <div className="h-12 w-12 rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 flex items-center justify-center text-white font-bold text-sm">
   433→                              {run.period.split("-")[1]}
   434→                            </div>
   435→                            <div>
   436→                              <div className="flex items-center gap-2">
   437→                                <h3 className="font-semibold">Period: {run.period}</h3>
   438→                                <Badge variant="outline" className={`text-[10px] px-1.5 py-0 ${statusCfg.bg} ${statusCfg.color} border`}>
   439→                                  {statusCfg.label}
   440→                                </Badge>
   441→                              </div>
   442→                              <p className="text-xs text-muted-foreground mt-0.5">
   443→                                {formatDate(run.runDate)} &middot; {run.items?.length || 0} employees
   444→                              </p>
   445→                            </div>
   446→                          </div>
   447→                          <div className="flex items-center gap-6">
   448→                            <div className="text-right hidden sm:block">
