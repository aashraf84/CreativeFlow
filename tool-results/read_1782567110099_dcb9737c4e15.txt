    25→const CATEGORIES = ["All", "Stationery", "Electronics", "Print Materials", "Merchandise", "Software"]
    26→
    27→function getStockStatus(product: Product) {
    28→  const totalQty = product.stockLevels?.reduce((s, sl) => s + sl.quantity, 0) || 0
    29→  if (totalQty === 0) return "out" as const
    30→  if (totalQty <= product.reorderLevel) return "low" as const
    31→  return "ok" as const
    32→}
    33→
    34→function StockIndicator({ status }: { status: "ok" | "low" | "out" }) {
    35→  const config = {
    36→    ok: { color: "bg-emerald-500", label: "In Stock", textColor: "text-emerald-700", bgColor: "bg-emerald-50" },
    37→    low: { color: "bg-amber-500", label: "Low Stock", textColor: "text-amber-700", bgColor: "bg-amber-50" },
    38→    out: { color: "bg-red-500", label: "Out of Stock", textColor: "text-red-700", bgColor: "bg-red-50" },
    39→  }
    40→  const c = config[status]
    41→  return (
    42→    <span className={`inline-flex items-center gap-1.5 px-2 py-0.5 rounded-full text-xs font-medium ${c.bgColor} ${c.textColor}`}>
    43→      <span className={`h-1.5 w-1.5 rounded-full ${c.color}`} />
    44→      {c.label}
    45→    </span>
    46→  )
    47→}

    49→export function ProductsView() {

    54→  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null)

    69→    const qty = p.stockLevels?.reduce((s, sl) => s + sl.quantity, 0) || 0
    70→    return sum + qty * p.costPrice
    71→  }, 0)
    72→

    75→    let result = products

    77→      result = result.filter(p => p.category === categoryFilter)
    78→    }
    79→    if (search.trim()) {
    80→      const q = search.toLowerCase()

    82→        p.name.toLowerCase().includes(q) || p.sku.toLowerCase().includes(q)
    83→      )
    84→    }

    86→  }, [products, categoryFilter, search])
    87→
    88→  // Low stock products
    89→  const lowStockProducts = useMemo(() => {

    91→      const qty = p.stockLevels?.reduce((s, sl) => s + sl.quantity, 0) || 0

   109→    const body = {
   110→      name: formData.get("name") as string,
   111→      sku: formData.get("sku") as string,
   112→      category: formData.get("category") as string,
   113→      unit: formData.get("unit") as string,
   114→      unitPrice: parseFloat(formData.get("unitPrice") as string) || 0,
   115→      costPrice: parseFloat(formData.get("costPrice") as string) || 0,
   116→      reorderLevel: parseInt(formData.get("reorderLevel") as string) || 0,
   117→    }
   118→    try {
   119→      const res = await fetch("/api/products", {
   120→        method: "POST",
   121→        headers: { "Content-Type": "application/json" },
   122→        body: JSON.stringify(body),
   123→      })
   124→      if (res.ok) {
   125→        const data = await res.json()
   126→        useERPStore.getState().setProducts([...products, data.product || data])
   127→        setShowAddProduct(false)
   128→      }
   129→    } catch {
   130→      // handle error silently
   131→    }
   132→  }, [products])
   133→
   134→  const handleStockMovement = useCallback(async (e: React.FormEvent<HTMLFormElement>) => {
   135→    e.preventDefault()
   136→    const form = e.currentTarget
   137→    const formData = new FormData(form)
   138→    const body = {
   139→      productId: formData.get("productId") as string,
   140→      warehouseId: formData.get("warehouseId") as string,
   141→      type: formData.get("type") as string,
   142→      quantity: parseInt(formData.get("quantity") as string) || 0,
   143→      reference: formData.get("reference") as string || undefined,
   144→      notes: formData.get("notes") as string || undefined,
   145→    }
   146→    try {
   147→      const res = await fetch("/api/stock-movements", {
   148→        method: "POST",
   149→        headers: { "Content-Type": "application/json" },
   150→        body: JSON.stringify(body),
   151→      })

   154→        const smRes = await fetch("/api/stock-movements")

   158→        // Refresh products for updated stock levels

   160→        const pData = await pRes.json()

   177→        </div>
   178→        <div className="flex items-center gap-2">
   179→          <Button variant="outline" size="sm" onClick={() => setShowStockMovement(true)}>

   229→                  const qty = p.stockLevels?.reduce((s, sl) => s + sl.quantity, 0) || 0

   232→                      key={p.id}

   234→                      className="cursor-pointer border-amber-300 text-amber-800 hover:bg-amber-100"
   235→                      onClick={() => setSelectedProduct(p)}
   236→                    >
   237→                      {p.name}: {qty} left (reorder at {p.reorderLevel})
   238→                    </Badge>
   239→                  )
   240→                })}
   241→              </div>
   242→            </CardContent>
   243→          </Card>
   244→        </motion.div>
   245→      )}
   246→
   247→      {/* Filters & Search */}
   248→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   249→        <Tabs value={categoryFilter} onValueChange={setCategoryFilter}>
   250→          <TabsList className="h-9">
   251→            {CATEGORIES.map(cat => (
   252→              <TabsTrigger key={cat} value={cat} className="text-xs px-3">
   253→                {cat}
   254→              </TabsTrigger>
   255→            ))}
   256→          </TabsList>
   257→        </Tabs>
   258→        <div className="flex items-center gap-2 ml-auto">
   259→          <div className="relative">
   260→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   261→            <Input
   262→              placeholder="Search name or SKU..."
   263→              value={search}
   264→              onChange={e => setSearch(e.target.value)}
   265→              className="pl-8 h-9 w-56"
   266→            />
   267→          </div>
   268→          <div className="flex items-center border rounded-md">
   269→            <Button
   270→              variant={viewMode === "grid" ? "secondary" : "ghost"}
   271→              size="sm"
   272→              className="h-9 w-9 p-0 rounded-r-none"
   273→              onClick={() => setViewMode("grid")}
   274→            >
   275→              <LayoutGrid className="h-4 w-4" />
   276→            </Button>
   277→            <Button
   278→              variant={viewMode === "list" ? "secondary" : "ghost"}
   279→              size="sm"
   280→              className="h-9 w-9 p-0 rounded-l-none"
   281→              onClick={() => setViewMode("list")}
   282→            >
   283→              <List className="h-4 w-4" />
   284→            </Button>
   285→          </div>
   286→        </div>
   287→      </div>
   288→
   289→      {/* Product Grid/List */}
   290→      {filteredProducts.length === 0 ? (
   291→        <Card>
   292→          <CardContent className="py-16 flex flex-col items-center gap-3">
   293→            <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center">
   294→              <PackageX className="h-6 w-6 text-muted-foreground" />
   295→            </div>
   296→            <p className="text-muted-foreground font-medium">No products found</p>
   297→            <p className="text-sm text-muted-foreground">Try adjusting your search or filters</p>
   298→          </CardContent>
   299→        </Card>
   300→      ) : viewMode === "grid" ? (
   301→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   302→          <AnimatePresence mode="popLayout">
   303→            {filteredProducts.map((product, idx) => (
   304→              <ProductCard
   305→                key={product.id}
   306→                product={product}
   307→                onClick={() => setSelectedProduct(product)}
   308→                index={idx}
   309→              />
   310→            ))}
   311→          </AnimatePresence>
   312→        </div>
   313→      ) : (
   314→        <Card>
   315→          <CardContent className="p-0">
   316→            <div className="divide-y">
   317→              {filteredProducts.map((product) => (
   318→                <ProductListItem
   319→                  key={product.id}
   320→                  product={product}
   321→                  onClick={() => setSelectedProduct(product)}
   322→                />
   323→              ))}
   324→            </div>
   325→          </CardContent>
   326→        </Card>
   327→      )}
   328→
   329→      {/* Product Detail Sheet */}
   330→      <Sheet open={!!selectedProduct} onOpenChange={(open) => !open && setSelectedProduct(null)}>
   331→        <SheetContent className="w-full sm:max-w-lg">
   332→          {selectedProduct && (
   333→            <>
   334→              <SheetHeader>
   335→                <SheetTitle className="text-lg">{selectedProduct.name}</SheetTitle>
   336→              </SheetHeader>
   337→              <ScrollArea className="h-[calc(100vh-100px)] mt-4">
   338→                <div className="space-y-6 pr-4">
   339→                  {/* Basic Info */}
   340→                  <div className="space-y-3">
   341→                    <div className="flex items-center gap-2">
   342→                      <Badge variant="secondary" className="font-mono text-xs">{selectedProduct.sku}</Badge>
   343→                      {selectedProduct.category && (
   344→                        <Badge variant="outline" className="text-xs">{selectedProduct.category}</Badge>
   345→                      )}
   346→                      <StockIndicator status={getStockStatus(selectedProduct)} />
   347→                    </div>
   348→                    {selectedProduct.description && (
   349→                      <p className="text-sm text-muted-foreground">{selectedProduct.description}</p>
   350→                    )}
   351→                    <div className="grid grid-cols-2 gap-3">
   352→                      <div className="bg-muted/50 rounded-lg p-3">
   353→                        <p className="text-xs text-muted-foreground">Unit Price</p>
   354→                        <p className="text-lg font-bold">${selectedProduct.unitPrice.toFixed(2)}</p>
   355→                      </div>
   356→                      <div className="bg-muted/50 rounded-lg p-3">
   357→                        <p className="text-xs text-muted-foreground">Cost Price</p>
   358→                        <p className="text-lg font-bold">${selectedProduct.costPrice.toFixed(2)}</p>
   359→                      </div>
   360→                      <div className="bg-muted/50 rounded-lg p-3">
   361→                        <p className="text-xs text-muted-foreground">Unit</p>
   362→                        <p className="text-lg font-bold">{selectedProduct.unit}</p>
   363→                      </div>
   364→                      <div className="bg-muted/50 rounded-lg p-3">
   365→                        <p className="text-xs text-muted-foreground">Reorder Level</p>
   366→                        <p className="text-lg font-bold">{selectedProduct.reorderLevel}</p>
   367→                      </div>
   368→                    </div>
   369→                  </div>
   370→
   371→                  <Separator />
   372→
   373→                  {/* Stock Levels per Warehouse */}
   374→                  <div>
   375→                    <h3 className="text-sm font-semibold mb-3 flex items-center gap-2">
   376→                      <Warehouse className="h-4 w-4 text-emerald-600" />
   377→                      Stock by Warehouse
   378→                    </h3>
   379→                    {selectedProduct.stockLevels && selectedProduct.stockLevels.length > 0 ? (
   380→                      <div className="space-y-2">
   381→                        {selectedProduct.stockLevels.map(sl => (
   382→                          <div key={sl.id} className="flex items-center justify-between p-3 rounded-lg border bg-card">
   383→                            <div>
   384→                              <p className="text-sm font-medium">{sl.warehouse.name}</p>
   385→                              <p className="text-xs text-muted-foreground">{sl.warehouse.code}</p>
   386→                            </div>
   387→                            <div className="text-right">
   388→                              <p className="font-semibold">{sl.quantity}</p>
   389→                              {sl.reservedQuantity > 0 && (
   390→                                <p className="text-xs text-amber-600">{sl.reservedQuantity} reserved</p>
   391→                              )}
   392→                            </div>
   393→                          </div>
   394→                        ))}
   395→                      </div>
   396→                    ) : (
   397→                      <p className="text-sm text-muted-foreground">No stock levels recorded</p>
   398→                    )}
   399→                    <div className="mt-3 p-3 rounded-lg bg-emerald-50 border border-emerald-200">
   400→                      <div className="flex items-center justify-between">
   401→                        <span className="text-sm font-medium text-emerald-800">Total Stock</span>
   402→                        <span className="text-lg font-bold text-emerald-800">
   403→                          {selectedProduct.stockLevels?.reduce((s, sl) => s + sl.quantity, 0) || 0}
   404→                        </span>
   405→                      </div>
   406→                    </div>
   407→                  </div>
   408→
   409→                  <Separator />
   410→
   411→                  {/* Recent Movements */}
   412→                  <div>
   413→                    <h3 className="text-sm font-semibold mb-3 flex items-center gap-2">
   414→                      <TrendingUp className="h-4 w-4 text-teal-600" />
   415→                      Recent Movements
   416→                    </h3>
   417→                    {productMovements.length > 0 ? (
   418→                      <div className="space-y-2">
   419→                        {productMovements.map(m => (
   420→                          <MovementItem key={m.id} movement={m} />
   421→                        ))}
   422→                      </div>
   423→                    ) : (
   424→                      <p className="text-sm text-muted-foreground">No recent movements</p>
   425→                    )}
   426→                  </div>
   427→                </div>
   428→              </ScrollArea>
   429→            </>
   430→          )}
   431→        </SheetContent>
   432→      </Sheet>
   433→
   434→      {/* Add Product Dialog */}
   435→      <Dialog open={showAddProduct} onOpenChange={setShowAddProduct}>
   436→        <DialogContent className="sm:max-w-md">
   437→          <DialogHeader>
   438→            <DialogTitle>Add New Product</DialogTitle>
   439→          </DialogHeader>
   440→          <form onSubmit={handleAddProduct} className="space-y-4">
   441→            <div className="grid grid-cols-2 gap-3">
   442→              <div className="space-y-2">
   443→                <Label htmlFor="name">Product Name</Label>
   444→                <Input id="name" name="name" required placeholder="e.g. A4 Paper" />
   445→              </div>
   446→              <div className="space-y-2">
   447→                <Label htmlFor="sku">SKU</Label>
   448→                <Input id="sku" name="sku" required placeholder="e.g. A4-PAP-001" />
   449→              </div>
   450→            </div>
   451→            <div className="grid grid-cols-2 gap-3">

   574→function ProductCard({ product, onClick, index }: { product: Product; onClick: () => void; index: number }) {
   575→  const totalQty = product.stockLevels?.reduce((s, sl) => s + sl.quantity, 0) || 0
   576→  const status = getStockStatus(product)
   577→  const totalValue = totalQty * product.costPrice
   578→
   579→  return (
   580→    <motion.div
   581→      initial={{ opacity: 0, y: 12 }}
   582→      animate={{ opacity: 1, y: 0 }}
   583→      exit={{ opacity: 0, scale: 0.95 }}
   584→      transition={{ duration: 0.2, delay: Math.min(index * 0.03, 0.3) }}
   585→    >
   586→      <Card
   587→        className="cursor-pointer hover:shadow-md transition-all hover:border-emerald-200 group"
   588→        onClick={onClick}
   589→      >
   590→        <CardContent className="p-4 space-y-3">
   591→          <div className="flex items-start justify-between">
   592→            <div className="min-w-0 flex-1">
   593→              <p className="font-semibold text-sm truncate group-hover:text-emerald-700 transition-colors">
   594→                {product.name}
   595→              </p>
   596→              <Badge variant="secondary" className="font-mono text-[10px] mt-1">{product.sku}</Badge>
   597→            </div>
   598→            <ChevronRight className="h-4 w-4 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity mt-0.5" />
   599→          </div>
   600→
   601→          <div className="flex items-center gap-2">
   602→            {product.category && (
   603→              <Badge variant="outline" className="text-[10px]">{product.category}</Badge>
   604→            )}
   605→            <StockIndicator status={status} />
   606→          </div>
   607→
   608→          <div className="grid grid-cols-2 gap-2">
   609→            <div>
   610→              <p className="text-[10px] text-muted-foreground">Price</p>
   611→              <p className="text-sm font-bold">${product.unitPrice.toFixed(2)}</p>
   612→            </div>
   613→            <div>
   614→              <p className="text-[10px] text-muted-foreground">Cost</p>
   615→              <p className="text-sm font-bold">${product.costPrice.toFixed(2)}</p>
   616→            </div>
   617→          </div>
   618→

   620→          {product.stockLevels && product.stockLevels.length > 0 && (
   621→            <div className="space-y-1">
   622→              <p className="text-[10px] text-muted-foreground font-medium">Stock by Warehouse</p>
   623→              {product.stockLevels.map(sl => {
   624→                const maxQty = Math.max(sl.quantity, product.reorderLevel, 1)
   625→                const pct = Math.min((sl.quantity / maxQty) * 100, 100)
   626→                return (
   627→                  <div key={sl.id} className="flex items-center gap-2">
   628→                    <span className="text-[10px] text-muted-foreground w-16 truncate">{sl.warehouse.code}</span>
   629→                    <div className="flex-1 h-1.5 bg-muted rounded-full overflow-hidden">
   630→                      <div
   631→                        className={`h-full rounded-full transition-all ${
   632→                          sl.quantity === 0 ? "bg-red-400" :
   633→                          sl.quantity <= product.reorderLevel ? "bg-amber-400" :
   634→                          "bg-emerald-400"
   635→                        }`}
   636→                        style={{ width: `${pct}%` }}
   637→                      />
   638→                    </div>
   639→                    <span className="text-[10px] font-medium w-8 text-right">{sl.quantity}</span>
   640→                  </div>
   641→                )
   642→              })}
   643→            </div>
   644→          )}
   645→
   646→          <div className="flex items-center justify-between pt-2 border-t">
   647→            <span className="text-xs text-muted-foreground">Total: {totalQty} {product.unit}</span>
   648→            <span className="text-xs font-semibold text-teal-700">${totalValue.toLocaleString()}</span>
   649→          </div>
   650→        </CardContent>
   651→      </Card>
   652→    </motion.div>
   653→  )
