   102→  {
   103→    name: "Auto-schedule approved content",
   104→    description: "When content reaches the Approve stage, automatically schedule it for publishing.",
   105→    trigger: "stage_change",

   112→    trigger: "scheduled_time",
   113→    conditions: JSON.stringify([{ key: "days_published", operator: "greater_than", value: "30" }]),
   114→    actions: JSON.stringify([{ type: "move_to_stage", stage: "archived" }]),
   115→  },
   116→];

   118→// ── Condition / Action row types ──────────────────────────────────
   119→interface ConditionRow {
   120→  key: string;
   121→  operator: string;
   122→  value: string;
   123→}
   124→
   125→interface ActionRow {
   126→  type: string;
   127→  [key: string]: string;
   128→}
   129→
   130→// ── Form state ──────────────────────────────────────────────────
   131→interface RuleForm {
   132→  name: string;
   133→  description: string;
   134→  trigger: string;
   135→  conditions: ConditionRow[];
   136→  actions: ActionRow[];
   137→  isActive: boolean;
   138→}
   139→
   140→const emptyForm: RuleForm = {
   141→  name: "",
   142→  description: "",
   143→  trigger: "stage_change",
   144→  conditions: [{ key: "", operator: "equals", value: "" }],
   145→  actions: [{ type: "move_to_stage" }],
   146→  isActive: true,
   147→};
   148→
   149→export function AutomationView() {
   150→  const { automations, setAutomations } = useERPStore();
   151→  const [showForm, setShowForm] = useState(false);
   152→  const [editingId, setEditingId] = useState<string | null>(null);
   153→  const [form, setForm] = useState<RuleForm>(emptyForm);
   154→  const [expandedId, setExpandedId] = useState<string | null>(null);
   155→  const [deleteId, setDeleteId] = useState<string | null>(null);
   156→
   157→  const activeCount = useMemo(() => automations.filter((a) => a.isActive).length, [automations]);
   158→
   159→  // ── Parse conditions / actions for display ──────────────────────────
   160→  function parseJSON<T>(str: string | null, fallback: T[]): T[] {
   161→    if (!str) return fallback;
   162→    try {
   163→      return JSON.parse(str);
   164→    } catch {
   165→      return fallback;
   166→    }
   167→  }
   168→
   169→  // ── Create / Update ──────────────────────────────────────────────
   170→  function handleSave() {
   171→    if (!form.name.trim()) return;
   172→
   173→    const rule: AutomationRule = {
   174→      id: editingId || `auto-${Date.now()}`,
   175→      name: form.name,
   176→      description: form.description || null,
   177→      trigger: form.trigger,
   178→      conditions: JSON.stringify(form.conditions),
   179→      actions: JSON.stringify(form.actions),
   180→      isActive: form.isActive,
   181→    };
   182→
   183→    if (editingId) {
   184→      setAutomations(automations.map((a) => (a.id === editingId ? rule : a)));
   185→    } else {
   186→      setAutomations([...automations, rule]);
   187→    }
   188→    closeForm();
   189→  }
   190→
   191→  // ── Edit ──────────────────────────────────────────────────────────
   192→  function handleEdit(rule: AutomationRule) {
   193→    setEditingId(rule.id);
   194→    setForm({
   195→      name: rule.name,
   196→      description: rule.description || "",
   197→      trigger: rule.trigger,
   198→      conditions: parseJSON<ConditionRow>(rule.conditions, [{ key: "", operator: "equals", value: "" }]),
   199→      actions: parseJSON<ActionRow>(rule.actions, [{ type: "move_to_stage" }]),
   200→      isActive: rule.isActive,
   201→    });
   202→    setShowForm(true);
   203→  }
   204→
   205→  // ── Delete ──────────────────────────────────────────────────────
   206→  function handleDelete() {
   207→    if (deleteId) {
   208→      setAutomations(automations.filter((a) => a.id !== deleteId));
   209→      setDeleteId(null);
   210→    }
   211→  }
   212→
   213→  // ── Toggle active ──────────────────────────────────────────────
   214→  function handleToggle(ruleId: string, active: boolean) {
   215→    setAutomations(automations.map((a) => (a.id === ruleId ? { ...a, isActive: active } : a)));
   216→  }
   217→
   218→  // ── Use template ──────────────────────────────────────────────
   219→  function handleUseTemplate(template: typeof TEMPLATES[0]) {
   220→    setEditingId(null);
   221→    setForm({
   222→      name: template.name,
   223→      description: template.description,
   224→      trigger: template.trigger,
   225→      conditions: parseJSON<ConditionRow>(template.conditions, [{ key: "", operator: "equals", value: "" }]),
   226→      actions: parseJSON<ActionRow>(template.actions, [{ type: "move_to_stage" }]),
   227→      isActive: true,
   228→    });
   229→    setShowForm(true);
   230→  }
   231→
   232→  // ── Close form ──────────────────────────────────────────────
   233→  function closeForm() {
   234→    setShowForm(false);
   235→    setEditingId(null);
   236→    setForm(emptyForm);
   237→  }
   238→
   239→  // ── Add / remove conditions ──────────────────────────────────
   240→  function addCondition() {
   241→    setForm({ ...form, conditions: [...form.conditions, { key: "", operator: "equals", value: "" }] });
   242→  }
   243→  function removeCondition(index: number) {
   244→    setForm({ ...form, conditions: form.conditions.filter((_, i) => i !== index) });
   245→  }
   246→  function updateCondition(index: number, field: keyof ConditionRow, value: string) {
   247→    const updated = [...form.conditions];
   248→    updated[index] = { ...updated[index], [field]: value };
   249→    setForm({ ...form, conditions: updated });
   250→  }
   251→
   252→  // ── Add / remove actions ─────────────────────────────────────
   253→  function addAction() {
   254→    setForm({ ...form, actions: [...form.actions, { type: "send_notification" }] });
   255→  }
   256→  function removeAction(index: number) {
   257→    setForm({ ...form, actions: form.actions.filter((_, i) => i !== index) });
   258→  }
   259→  function updateAction(index: number, field: string, value: string) {
   260→    const updated = [...form.actions];
   261→    updated[index] = { ...updated[index], [field]: value };
   262→    setForm({ ...form, actions: updated });
   263→  }
   264→
   265→  // ── Get trigger badge info ────────────────────────────────────
   266→  function getTriggerInfo(triggerValue: string) {
   267→    return TRIGGER_TYPES.find((t) => t.value === triggerValue) || TRIGGER_TYPES[0];
   268→  }
   269→
   270→  // ── Get action badge info ─────────────────────────────────────
   271→  function getActionInfo(actionValue: string) {
   272→    return ACTION_TYPES.find((a) => a.value === actionValue) || ACTION_TYPES[0];
   273→  }
   274→
   275→  return (
   276→    <div className="p-4 md:p-6 space-y-6">
   277→      {/* Header */}
   278→      <motion.div
   279→        variants={containerVariants}
   280→        initial="hidden"
   281→        animate="visible"
   282→        className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4"
   283→      >
   284→        <div>
   285→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   286→            <Workflow className="h-6 w-6 text-emerald-500" />
   287→            Automation Rules
   288→          </h1>
   289→          <p className="text-sm text-muted-foreground mt-1">
   290→            Manage workflows and automation triggers
   291→          </p>
   292→        </div>
   293→        <div className="flex items-center gap-3">
   294→          <div className="flex items-center gap-2 px-3 py-1.5 rounded-lg bg-emerald-50 dark:bg-emerald-950 border border-emerald-200 dark:border-emerald-800">
   295→            <Zap className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   296→            <span className="text-sm font-semibold text-emerald-700 dark:text-emerald-300">{activeCount} Active</span>
   297→          </div>
   298→          <Button
   299→            onClick={() => {
   300→              setEditingId(null);
   301→              setForm(emptyForm);
   302→              setShowForm(true);
   303→            }}
   304→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   305→          >
   306→            <Plus className="h-4 w-4 mr-1.5" />
   307→            Create Rule
   308→          </Button>
   309→        </div>
   310→      </motion.div>
   311→
   312→      <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-6">
   313→        {/* Rules List */}
   314→        {automations.length === 0 ? (
   315→          <motion.div variants={itemVariants}>
   316→            <Card>
   317→              <CardContent className="py-16 text-center">
   318→                <Workflow className="h-12 w-12 text-muted-foreground/40 mx-auto mb-4" />
   319→                <h3 className="text-lg font-semibold text-muted-foreground">No automation rules yet</h3>
   320→                <p className="text-sm text-muted-foreground mt-1">Create your first rule or use a template below</p>
   321→              </CardContent>
   322→            </Card>
   323→          </motion.div>
   324→        ) : (
   325→          <div className="space-y-3">
   326→            <AnimatePresence>
   327→              {automations.map((rule, index) => {
   328→                const trigger = getTriggerInfo(rule.trigger);
   329→                const actions = parseJSON<ActionRow>(rule.actions, []);
   330→                const conditions = parseJSON<ConditionRow>(rule.conditions, []);
   331→                const isExpanded = expandedId === rule.id;
   332→                const TriggerIcon = trigger.icon;
   333→
   334→                return (
   335→                  <motion.div
   336→                    key={rule.id}
   337→                    variants={itemVariants}
   338→                    initial="hidden"
   339→                    animate="visible"
   340→                    exit={{ opacity: 0, scale: 0.95 }}
   341→                    transition={{ delay: index * 0.05 }}
   342→                  >
   343→                    <Card className={cn(
   344→                      "transition-all duration-200",
   345→                      !rule.isActive && "opacity-60",
   346→                    )}>
   347→                      <CardContent className="p-5">
   348→                        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   349→                          <div className="flex-1 min-w-0 space-y-2">
   350→                            <div className="flex items-center gap-3">
   351→                              <h3 className="font-semibold text-base truncate">{rule.name}</h3>
   352→                              {rule.isActive ? (
   353→                                <Badge variant="secondary" className="bg-emerald-50 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 shrink-0">
   354→                                  Active
   355→                                </Badge>
   356→                              ) : (
   357→                                <Badge variant="secondary" className="bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400 shrink-0">
   358→                                  Inactive
   359→                                </Badge>
   360→                              )}
   361→                            </div>
   362→                            {rule.description && (
   363→                              <p className="text-sm text-muted-foreground line-clamp-2">{rule.description}</p>
   364→                            )}
   365→                            <div className="flex flex-wrap items-center gap-2">
   366→                              {/* Trigger badge */}
   367→                              <Badge variant="outline" className="gap-1.5 text-xs" style={{ borderColor: trigger.color, color: trigger.color }}>
   368→                                <TriggerIcon className="h-3 w-3" />
   369→                                {trigger.label}
   370→                              </Badge>
   371→                              {/* Action badges */}
   372→                              {actions.map((action, i) => {
   373→                                const actionInfo = getActionInfo(action.type);
   374→                                const ActionIcon = actionInfo.icon;
   375→                                return (
   376→                                  <Badge key={i} variant="outline" className="gap-1.5 text-xs" style={{ borderColor: actionInfo.color, color: actionInfo.color }}>
   377→                                    <ActionIcon className="h-3 w-3" />
   378→                                    {actionInfo.label}
   379→                                  </Badge>
   380→                                );
   381→                              })}
   382→                            </div>
   383→                          </div>
   384→
   385→                          {/* Actions */}
   386→                          <div className="flex items-center gap-2 shrink-0">
   387→                            <Switch
   388→                              checked={rule.isActive}
   389→                              onCheckedChange={(checked) => handleToggle(rule.id, checked)}
   390→                            />
   391→                            <Button
   392→                              variant="ghost"
   393→                              size="icon"
   394→                              className="h-8 w-8"
   395→                              onClick={() => setExpandedId(isExpanded ? null : rule.id)}
   396→                            >
   397→                              {isExpanded ? <ChevronUp className="h-4 w-4" /> : <ChevronDown className="h-4 w-4" />}
   398→                            </Button>
   399→                            <Button
   400→                              variant="ghost"
   401→                              size="icon"
   402→                              className="h-8 w-8 text-muted-foreground hover:text-emerald-600"
   403→                              onClick={() => handleEdit(rule)}
   404→                            >
   405→                              <Edit3 className="h-4 w-4" />
   406→                            </Button>
   407→                            <Button
   408→                              variant="ghost"
   409→                              size="icon"
   410→                              className="h-8 w-8 text-muted-foreground hover:text-red-600"
   411→                              onClick={() => setDeleteId(rule.id)}
   412→                            >
   413→                              <Trash2 className="h-4 w-4" />
   414→                            </Button>
   415→                          </div>
   416→                        </div>
   417→
   418→                        {/* Expanded details */}
   419→                        <AnimatePresence>
   420→                          {isExpanded && (
   421→                            <motion.div
   422→                              initial={{ height: 0, opacity: 0 }}
   423→                              animate={{ height: "auto", opacity: 1 }}
   424→                              exit={{ height: 0, opacity: 0 }}
   425→                              transition={{ duration: 0.2 }}
   426→                              className="overflow-hidden"
   427→                            >
   428→                              <Separator className="my-4" />
   429→                              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   430→                                {/* Conditions */}
   431→                                <div>
   432→                                  <h4 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">Conditions</h4>
   433→                                  {conditions.length > 0 && conditions.some((c) => c.key) ? (
   434→                                    <div className="space-y-1.5">
   435→                                      {conditions.map((cond, i) => (
   436→                                        cond.key ? (
   437→                                          <div key={i} className="flex items-center gap-2 text-sm bg-muted/50 rounded-md px-3 py-1.5">
   438→                                            <code className="font-mono text-emerald-600 dark:text-emerald-400">{cond.key}</code>
   439→                                            <span className="text-muted-foreground">{cond.operator}</span>
   440→                                            <code className="font-mono text-teal-600 dark:text-teal-400">{cond.value}</code>
   441→                                          </div>
   442→                                        ) : null
   443→                                      ))}
   444→                                    </div>
   445→                                  ) : (
   446→                                    <p className="text-sm text-muted-foreground italic">No conditions defined</p>
   447→                                  )}
   448→                                </div>
   449→                                {/* Actions detail */}
   450→                                <div>
   451→                                  <h4 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">Actions Detail</h4>
   452→                                  <div className="space-y-1.5">
   453→                                    {actions.map((action, i) => {
   454→                                      const actionInfo = getActionInfo(action.type);
   455→                                      const ActionIcon = actionInfo.icon;
   456→                                      return (
   457→                                        <div key={i} className="flex items-center gap-2 text-sm bg-muted/50 rounded-md px-3 py-1.5">
   458→                                          <ActionIcon className="h-3.5 w-3.5" style={{ color: actionInfo.color }} />
   459→                                          <span>{actionInfo.label}</span>
   460→                                          {action.stage && <code className="font-mono text-emerald-600 dark:text-emerald-400 ml-1">→ {action.stage}</code>}
   461→                                          {action.team && <code className="font-mono text-teal-600 dark:text-teal-400 ml-1">→ {action.team}</code>}
   462→                                          {action.message && <code className="font-mono text-cyan-600 dark:text-cyan-400 ml-1">"{action.message}"</code>}
   463→                                        </div>
   464→                                      );
   465→                                    })}
   466→                                  </div>
   467→                                </div>
   468→                              </div>
   469→                            </motion.div>
   470→                          )}
   471→                        </AnimatePresence>
   472→                      </CardContent>
   473→                    </Card>
   474→                  </motion.div>
   475→                );
   476→              })}
   477→            </AnimatePresence>
   478→          </div>
   479→        )}
   480→
   481→        {/* Automation Templates */}
   482→        <motion.div variants={itemVariants}>
   483→          <Card>
   484→            <CardHeader className="pb-2">
   485→              <CardTitle className="text-base font-semibold flex items-center gap-2">
   486→                <Sparkles className="h-4 w-4 text-emerald-500" />
   487→                Quick Templates
   488→              </CardTitle>
   489→              <CardDescription>Get started with pre-built automation rules</CardDescription>
   490→            </CardHeader>
   491→            <CardContent>
   492→              <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   493→                {TEMPLATES.map((template, i) => (
   494→                  <div
   495→                    key={i}
   496→                    className="flex items-start justify-between gap-3 rounded-lg border p-4 hover:border-emerald-300 dark:hover:border-emerald-700 hover:bg-emerald-50/50 dark:hover:bg-emerald-950/30 transition-colors"
   497→                  >
   498→                    <div className="flex-1 min-w-0">
   499→                      <h4 className="text-sm font-medium leading-tight">{template.name}</h4>
   500→                      <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{template.description}</p>
   501→                    </div>
   502→                    <Button
   503→                      variant="outline"
   504→                      size="sm"
   505→                      className="shrink-0 text-emerald-600 border-emerald-200 hover:bg-emerald-50 dark:text-emerald-400 dark:border-emerald-800 dark:hover:bg-emerald-950"
   506→                      onClick={() => handleUseTemplate(template)}
   507→                    >
   508→                      <Copy className="h-3 w-3 mr-1" />
   509→                      Use
   510→                    </Button>
   511→                  </div>
   512→                ))}
   513→              </div>
   514→            </CardContent>
   515→          </Card>
   516→        </motion.div>
   517→      </motion.div>
   518→
   519→      {/* Create/Edit Rule Dialog */}
   520→      <Dialog open={showForm} onOpenChange={(open) => { if (!open) closeForm(); }}>
   521→        <DialogContent className="sm:max-w-[600px] max-h-[90vh]">
   522→          <DialogHeader>
   523→            <DialogTitle className="flex items-center gap-2">
   524→              <Workflow className="h-5 w-5 text-emerald-500" />
   525→              {editingId ? "Edit Rule" : "Create Rule"}
   526→            </DialogTitle>
   527→            <DialogDescription>
   528→              Define triggers, conditions, and actions for your automation
   529→            </DialogDescription>
   530→          </DialogHeader>
   531→
   532→          <ScrollArea className="max-h-[60vh] pr-4">
   533→            <div className="space-y-5 py-2">
   534→              {/* Rule name */}
   535→              <div className="space-y-2">
   536→                <Label htmlFor="rule-name" className="text-sm font-medium">Rule Name</Label>
   537→                <Input
   538→                  id="rule-name"
   539→                  placeholder="e.g., Auto-assign Design stage"
   540→                  value={form.name}
   541→                  onChange={(e) => setForm({ ...form, name: e.target.value })}
   542→                />
   543→              </div>
   544→
   545→              {/* Description */}
   546→              <div className="space-y-2">
   547→                <Label htmlFor="rule-desc" className="text-sm font-medium">Description</Label>
   548→                <Textarea
   549→                  id="rule-desc"
   550→                  placeholder="What does this automation do?"
   551→                  value={form.description}
   552→                  onChange={(e) => setForm({ ...form, description: e.target.value })}
   553→                  rows={2}
   554→                />
   555→              </div>
   556→
   557→              {/* Trigger type */}
   558→              <div className="space-y-2">
   559→                <Label className="text-sm font-medium">Trigger Type</Label>
   560→                <Select value={form.trigger} onValueChange={(v) => setForm({ ...form, trigger: v })}>
   561→                  <SelectTrigger>
   562→                    <SelectValue />
   563→                  </SelectTrigger>
   564→                  <SelectContent>
   565→                    {TRIGGER_TYPES.map((t) => {
   566→                      const Icon = t.icon;
   567→                      return (
   568→                        <SelectItem key={t.value} value={t.value}>
   569→                          <div className="flex items-center gap-2">
   570→                            <Icon className="h-3.5 w-3.5" style={{ color: t.color }} />
   571→                            {t.label}
   572→                          </div>
   573→                        </SelectItem>
   574→                      );
   575→                    })}
   576→                  </SelectContent>
   577→                </Select>
   578→              </div>
   579→
   580→              <Separator />
   581→
   582→              {/* Conditions builder */}
   583→              <div className="space-y-3">
   584→                <div className="flex items-center justify-between">
   585→                  <Label className="text-sm font-medium">Conditions</Label>
   586→                  <Button variant="ghost" size="sm" onClick={addCondition} className="text-emerald-600 h-7">
   587→                    <Plus className="h-3 w-3 mr-1" /> Add
   588→                  </Button>
   589→                </div>
   590→                {form.conditions.map((cond, i) => (
   591→                  <div key={i} className="flex items-center gap-2">
   592→                    <Input
   593→                      placeholder="Key (e.g., stage)"
   594→                      value={cond.key}
   595→                      onChange={(e) => updateCondition(i, "key", e.target.value)}
   596→                      className="flex-1"
   597→                    />
   598→                    <Select value={cond.operator} onValueChange={(v) => updateCondition(i, "operator", v)}>
   599→                      <SelectTrigger className="w-[110px]">
   600→                        <SelectValue />
   601→                      </SelectTrigger>
   602→                      <SelectContent>
   603→                        <SelectItem value="equals">equals</SelectItem>
   604→                        <SelectItem value="not_equals">not equals</SelectItem>
   605→                        <SelectItem value="greater_than">greater than</SelectItem>
   606→                        <SelectItem value="less_than">less than</SelectItem>
   607→                        <SelectItem value="contains">contains</SelectItem>
   608→                      </SelectContent>
   609→                    </Select>
   610→                    <Input
   611→                      placeholder="Value"
   612→                      value={cond.value}
   613→                      onChange={(e) => updateCondition(i, "value", e.target.value)}
   614→                      className="flex-1"
   615→                    />
   616→                    {form.conditions.length > 1 && (
   617→                      <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0" onClick={() => removeCondition(i)}>
   618→                        <Trash2 className="h-3.5 w-3.5 text-red-400" />
   619→                      </Button>
   620→                    )}
   621→                  </div>
   622→                ))}
   623→              </div>
   624→
   625→              <Separator />
   626→
   627→              {/* Actions builder */}
   628→              <div className="space-y-3">
   629→                <div className="flex items-center justify-between">
   630→                  <Label className="text-sm font-medium">Actions</Label>
   631→                  <Button variant="ghost" size="sm" onClick={addAction} className="text-emerald-600 h-7">
   632→                    <Plus className="h-3 w-3 mr-1" /> Add
   633→                  </Button>
   634→                </div>
   635→                {form.actions.map((action, i) => (
   636→                  <div key={i} className="flex items-start gap-2 p-3 rounded-lg border bg-muted/30">
   637→                    <div className="flex-1 space-y-2">
   638→                      <Select value={action.type} onValueChange={(v) => updateAction(i, "type", v)}>
   639→                        <SelectTrigger>
   640→                          <SelectValue />
   641→                        </SelectTrigger>
   642→                        <SelectContent>
   643→                          {ACTION_TYPES.map((a) => {
   644→                            const Icon = a.icon;
   645→                            return (
   646→                              <SelectItem key={a.value} value={a.value}>
   647→                                <div className="flex items-center gap-2">
   648→                                  <Icon className="h-3.5 w-3.5" style={{ color: a.color }} />
   649→                                  {a.label}
   650→                                </div>
   651→                              </SelectItem>
   652→                            );
   653→                          })}
   654→                        </SelectContent>
   655→                      </Select>
   656→                      {action.type === "move_to_stage" && (
   657→                        <Input
   658→                          placeholder="Target stage (e.g., archived)"
   659→                          value={action.stage || ""}
   660→                          onChange={(e) => updateAction(i, "stage", e.target.value)}
   661→                        />
   662→                      )}
   663→                      {action.type === "assign_to_team" && (
   664→                        <Input
   665→                          placeholder="Team name (e.g., Design)"
   666→                          value={action.team || ""}
   667→                          onChange={(e) => updateAction(i, "team", e.target.value)}
   668→                        />
   669→                      )}
   670→                      {action.type === "send_notification" && (
   671→                        <Input
   672→                          placeholder="Notification message"
   673→                          value={action.message || ""}
   674→                          onChange={(e) => updateAction(i, "message", e.target.value)}
   675→                        />
   676→                      )}
   677→                    </div>
   678→                    {form.actions.length > 1 && (
   679→                      <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0 mt-1" onClick={() => removeAction(i)}>
   680→                        <Trash2 className="h-3.5 w-3.5 text-red-400" />
   681→                      </Button>
   682→                    )}
   683→                  </div>
   684→                ))}
   685→              </div>
   686→
   687→              <Separator />
   688→
   689→              {/* Active toggle */}
   690→              <div className="flex items-center justify-between">
   691→                <div>
   692→                  <Label className="text-sm font-medium">Active</Label>
   693→                  <p className="text-xs text-muted-foreground">Enable this rule to run automatically</p>
   694→                </div>
   695→                <Switch
   696→                  checked={form.isActive}
   697→                  onCheckedChange={(checked) => setForm({ ...form, isActive: checked })}
   698→                />
   699→              </div>
   700→            </div>
   701→          </ScrollArea>
   702→
   703→          <DialogFooter className="gap-2">
   704→            <Button variant="outline" onClick={closeForm}>Cancel</Button>
   705→            <Button
   706→              onClick={handleSave}
   707→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   708→              disabled={!form.name.trim()}
   709→            >
   710→              <CheckCircle2 className="h-4 w-4 mr-1.5" />
   711→              {editingId ? "Update Rule" : "Create Rule"}
   712→            </Button>
   713→          </DialogFooter>
   714→        </DialogContent>
   715→      </Dialog>
   716→
   717→      {/* Delete confirmation */}
   718→      <AlertDialog open={!!deleteId} onOpenChange={(open) => { if (!open) setDeleteId(null); }}>
   719→        <AlertDialogContent>
   720→          <AlertDialogHeader>
   721→            <AlertDialogTitle>Delete Rule</AlertDialogTitle>
   722→            <AlertDialogDescription>
   723→              This will permanently delete this automation rule. This action cannot be undone.
   724→            </AlertDialogDescription>
   725→          </AlertDialogHeader>
   726→          <AlertDialogFooter>
   727→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   728→            <AlertDialogAction onClick={handleDelete} className="bg-red-600 hover:bg-red-700 text-white">
   729→              Delete
   730→            </AlertDialogAction>
   731→          </AlertDialogFooter>
   732→        </AlertDialogContent>
   733→      </AlertDialog>
   734→    </div>
   735→  );
   736→}
