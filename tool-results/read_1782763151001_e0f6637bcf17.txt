   142→// ── Time formatting ─────────────────────────────────────────────────
   143→function formatRelativeTime(dateStr: string): string {
   144→  try {
   145→    const date = new Date(dateStr);
   146→    const now = new Date();
   147→    const diffMs = now.getTime() - date.getTime();
   148→    const diffMins = Math.floor(diffMs / 60000);
   149→    const diffHours = Math.floor(diffMs / 3600000);
   150→    const diffDays = Math.floor(diffMs / 86400000);
   151→
   152→    if (diffMins < 1) return "Just now";
   153→    if (diffMins < 60) return `${diffMins}m ago`;
   154→    if (diffHours < 24) return `${diffHours}h ago`;
   155→    if (diffDays < 7) return `${diffDays}d ago`;
   156→    return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   157→  } catch {
   158→    return "";
   159→  }
   160→}

   184→// ── KPI Card Component ──────────────────────────────────────────────
   185→interface KPICardData {
   186→  title: string;
   187→  value: number | string;
   188→  change?: number;
   189→  changeLabel?: string;
   190→  icon: React.ComponentType<{ className?: string }>;
   191→  gradient: string;
   192→  iconColor: string;
   193→  badge?: string;
   194→  badgeColor?: string;
   195→}
   196→
   197→function KPICard({ data }: { data: KPICardData }) {
   198→  const Icon = data.icon;
   199→
   200→  // ── 3D tilt on hover (subtle, springy) ──
   201→  const xMotion = useMotionValue(0);
   202→  const yMotion = useMotionValue(0);
   203→  const rotateX = useSpring(useTransform(yMotion, [-0.5, 0.5], [6, -6]), {
   204→    stiffness: 200,
   205→    damping: 18,
   206→  });
   207→  const rotateY = useSpring(useTransform(xMotion, [-0.5, 0.5], [-6, 6]), {
   208→    stiffness: 200,
   209→    damping: 18,
   210→  });
   211→
   212→  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
   213→    const rect = e.currentTarget.getBoundingClientRect();
   214→    const px = (e.clientX - rect.left) / rect.width; // 0..1
   215→    const py = (e.clientY - rect.top) / rect.height; // 0..1
   216→    xMotion.set(px - 0.5);
   217→    yMotion.set(py - 0.5);
   218→  };
   219→
   220→  const handleMouseLeave = () => {
   221→    xMotion.set(0);
   222→    yMotion.set(0);
   223→  };
   224→
   225→  const isNumeric = typeof data.value === "number";
   226→
   227→  return (
   228→    <motion.div variants={itemVariants}>
   229→      <motion.div
   230→        onMouseMove={handleMouseMove}
   231→        onMouseLeave={handleMouseLeave}
   232→        style={{ perspective: 1000 }}
   233→        whileHover={{ scale: 1.03 }}
   234→        transition={{ type: "spring", stiffness: 300, damping: 22 }}
   235→      >
   236→        <Card className="relative overflow-hidden glass border border-black/5 dark:border-white/10 shadow-macos rounded-2xl">
   237→          <motion.div
   238→            style={{ rotateX, rotateY, transformStyle: "preserve-3d" }}
   239→            className="relative"
   240→          >
   241→            <CardContent className="p-5">
   242→              <div className="flex items-start justify-between">
   243→                <div className="flex-1 min-w-0">
   244→                  <p className="text-[11px] font-medium text-[#86868b] dark:text-[#aeaeb2] uppercase tracking-wide">
   245→                    {data.title}
   246→                  </p>
   247→                  <div className="mt-2 flex items-baseline gap-2">
   248→                    <h3 className="text-3xl font-bold tracking-tight text-[#1d1d1f] dark:text-white tabular-nums">
   249→                      {isNumeric ? (
   250→                        <AnimatedCounter value={data.value as number} duration={1000} />
   251→                      ) : (
   252→                        data.value
   253→                      )}
   254→                    </h3>
   255→                    {data.badge && (
   256→                      <span className={`inline-flex items-center px-1.5 py-0.5 text-[10px] font-semibold rounded-md leading-none ${data.badgeColor}`}>
   257→                        {data.badge}
   258→                      </span>
   259→                    )}
   260→                  </div>
   261→                  {data.change !== undefined && (
   262→                    <div className="mt-1.5 flex items-center gap-1">
   263→                      <ArrowUpRight className="h-3 w-3 text-emerald-600 dark:text-emerald-400" />
   264→                      <span className={`text-xs font-medium ${data.change >= 0 ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400"}`}>
   265→                        {Math.abs(data.change)}%
   266→                      </span>
   267→                      {data.changeLabel && (
   268→                        <span className="text-[10px] text-[#86868b] dark:text-[#aeaeb2]">{data.changeLabel}</span>
   269→                      )}
   270→                    </div>
   271→                  )}
   272→                </div>
   273→                <motion.div
   274→                  className={`h-10 w-10 rounded-xl bg-gradient-to-br ${data.gradient} flex items-center justify-center shadow-md shadow-emerald-500/20 shrink-0`}
   275→                  style={{ transform: "translateZ(40px)" }}
   276→                  whileHover={{ scale: 1.1, rotate: 5 }}
   277→                  transition={{ type: "spring", stiffness: 400, damping: 15 }}
   278→                >
   279→                  <Icon className={`h-5 w-5 ${data.iconColor}`} />
   280→                </motion.div>
   281→              </div>
   282→            </CardContent>
   283→          </motion.div>
   284→          <div className="absolute bottom-0 left-0 right-0 h-px bg-gradient-to-r from-transparent via-emerald-500/25 to-transparent" />
   285→        </Card>
   286→      </motion.div>
   287→    </motion.div>
   288→  );
   289→}

   456→              }}
   457→            />
   458→          );
   459→        })}
   460→      </div>
   461→    </motion.div>
