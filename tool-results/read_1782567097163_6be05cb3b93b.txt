     1→"use client";
     2→
     3→import React, { useState, useMemo } from "react";
     4→import { useERPStore, type Content, type ContentType, type ContentStage } from "@/lib/store";
     5→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
     6→import { Badge } from "@/components/ui/badge";
     7→import { Button } from "@/components/ui/button";
     8→import { Input } from "@/components/ui/input";
     9→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    10→import { Separator } from "@/components/ui/separator";
    11→import { Checkbox } from "@/components/ui/checkbox";
    12→import {
    13→  AlertDialog,
    14→  AlertDialogAction,
    15→  AlertDialogCancel,
    16→  AlertDialogContent,
    17→  AlertDialogDescription,
    18→  AlertDialogFooter,
    19→  AlertDialogHeader,
    20→  AlertDialogTitle,
    21→} from "@/components/ui/alert-dialog";
    22→import { motion, AnimatePresence } from "framer-motion";
    23→import {
    24→  Archive,
    25→  Search,
    26→  LayoutGrid,
    27→  List,
    28→  RotateCcw,
    29→  Trash2,
    30→  FileText,
    31→  Image,
    32→  Film,
    33→  Layers,
    34→  Video,
    35→  Palette,
    36→  Type,
    37→  CalendarDays,
    38→  Users,
    39→  FolderOpen,
    40→  Database,
    41→  HardDrive,
    42→  CheckSquare,
    43→  XSquare,
    44→} from "lucide-react";
    45→import { cn } from "@/lib/utils";
    46→
    47→// ── Animation variants ──────────────────────────────────────────────
    48→const containerVariants = {
    49→  hidden: { opacity: 0 },
    50→  visible: {
    51→    opacity: 1,
    52→    transition: { staggerChildren: 0.06, delayChildren: 0.1 },
    53→  },
    54→};
    55→
    56→const itemVariants = {
    57→  hidden: { opacity: 0, y: 16, scale: 0.97 },
    58→  visible: {
    59→    opacity: 1,
    60→    y: 0,
    61→    scale: 1,
    62→    transition: { type: "spring", stiffness: 300, damping: 24 },
    63→  },
    64→};
    65→
    66→// ── Type icon mapping ──────────────────────────────────────────────
    67→const TYPE_ICONS: Record<ContentType, React.ComponentType<{ className?: string }>> = {
    68→  post: FileText,
    69→  story: Image,
    70→  reel: Film,
    71→  carousel: Layers,
    72→  video: Video,
    73→  graphic: Palette,
    74→  article: Type,
    75→};
    76→
    77→const TYPE_COLORS: Record<ContentType, string> = {
    78→  post: "#10b981",
    79→  story: "#14b8a6",
    80→  reel: "#06b6d4",
    81→  carousel: "#0d9488",
    82→  video: "#059669",
    83→  graphic: "#34d399",
    84→  article: "#2dd4bf",
    85→};
    86→
    87→// ── View mode ──────────────────────────────────────────────────────
    88→type ViewMode = "grid" | "list";
    89→
    90→export function ArchiveView() {
    91→  const { contents, setContents, campaigns, teams } = useERPStore();
    92→  const [search, setSearch] = useState("");
    93→  const [typeFilter, setTypeFilter] = useState<string>("all");
    94→  const [campaignFilter, setCampaignFilter] = useState<string>("all");
    95→  const [teamFilter, setTeamFilter] = useState<string>("all");
    96→  const [viewMode, setViewMode] = useState<ViewMode>("grid");
    97→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
    98→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null);
    99→  const [bulkAction, setBulkAction] = useState<"restore" | "delete" | null>(null);
   100→
   101→  // ── Archived contents ──────────────────────────────────────────
   102→  const archivedContents = useMemo(() => {
   103→    return contents.filter((c) => c.stage === "archived");
   104→  }, [contents]);
   105→
   106→  // ── Filtered contents ──────────────────────────────────────────
   107→  const filteredContents = useMemo(() => {
   108→    let result = archivedContents;
   109→
   110→    if (search.trim()) {
   111→      const q = search.toLowerCase();
   112→      result = result.filter(
   113→        (c) =>
   114→          c.title.toLowerCase().includes(q) ||
   115→          (c.description && c.description.toLowerCase().includes(q))
   116→      );
   117→    }
   118→
   119→    if (typeFilter !== "all") {
   120→      result = result.filter((c) => c.type === typeFilter);
   121→    }
   122→
   123→    if (campaignFilter !== "all") {
   124→      result = result.filter((c) => c.campaignId === campaignFilter);
   125→    }
   126→
   127→    if (teamFilter !== "all") {
   128→      result = result.filter((c) => c.teamId === teamFilter);
   129→    }
   130→
   131→    return result;
   132→  }, [archivedContents, search, typeFilter, campaignFilter, teamFilter]);
   133→
   134→  // ── Stats ──────────────────────────────────────────────────────
   135→  const archivedThisMonth = useMemo(() => {
   136→    const now = new Date();
   137→    const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
   138→    return archivedContents.filter((c) => new Date(c.updatedAt) >= startOfMonth).length;
   139→  }, [archivedContents]);
   140→
   141→  const storageUsed = useMemo(() => {
   142→    return `${(archivedContents.length * 2.4).toFixed(1)} MB`;
   143→  }, [archivedContents]);
   144→
   145→  // ── Toggle selection ───────────────────────────────────────────
   146→  function toggleSelect(id: string) {
   147→    setSelectedIds((prev) => {
   148→      const next = new Set(prev);
   149→      if (next.has(id)) next.delete(id);
   150→      else next.add(id);
   151→      return next;
   152→    });
   153→  }
   154→
   155→  function toggleSelectAll() {
   156→    if (selectedIds.size === filteredContents.length) {
   157→      setSelectedIds(new Set());
   158→    } else {
   159→      setSelectedIds(new Set(filteredContents.map((c) => c.id)));
   160→    }
   161→  }
   162→
   163→  // ── Restore content ────────────────────────────────────────────
   164→  function handleRestore(id: string) {
   165→    setContents(
   166→      contents.map((c) =>
   167→        c.id === id ? { ...c, stage: "idea" as ContentStage, updatedAt: new Date().toISOString() } : c
   168→      )
   169→    );
   170→    setSelectedIds((prev) => {
   171→      const next = new Set(prev);
   172→      next.delete(id);
   173→      return next;
   174→    });
   175→  }
   176→
   177→  function handleBulkRestore() {
   178→    setContents(
   179→      contents.map((c) =>
   180→        selectedIds.has(c.id) ? { ...c, stage: "idea" as ContentStage, updatedAt: new Date().toISOString() } : c
   181→      )
   182→    );
   183→    setSelectedIds(new Set());
   184→    setBulkAction(null);
   185→  }
   186→
   187→  // ── Delete content ─────────────────────────────────────────────
   188→  function handleDelete() {
   189→    if (deleteTarget) {
   190→      setContents(contents.filter((c) => c.id !== deleteTarget));
   191→      setSelectedIds((prev) => {
   192→        const next = new Set(prev);
   193→        next.delete(deleteTarget);
   194→        return next;
   195→      });
   196→      setDeleteTarget(null);
   197→    }
   198→  }
   199→
   200→  function handleBulkDelete() {
   201→    setContents(contents.filter((c) => !selectedIds.has(c.id)));
   202→    setSelectedIds(new Set());
   203→    setBulkAction(null);
   204→  }
   205→
   206→  // ── Format date ────────────────────────────────────────────────
   207→  function formatDate(dateStr: string | null) {
   208→    if (!dateStr) return "—";
   209→    return new Date(dateStr).toLocaleDateString("en-US", {
   210→      month: "short",
   211→      day: "numeric",
   212→      year: "numeric",
   213→    });
   214→  }
   215→
   216→  return (
   217→    <div className="p-4 md:p-6 space-y-6">
   218→      {/* Header */}
   219→      <motion.div
   220→        variants={containerVariants}
   221→        initial="hidden"
   222→        animate="visible"
   223→        className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4"
   224→      >
   225→        <div>
   226→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   227→            <Archive className="h-6 w-6 text-emerald-500" />
   228→            Content Archive
   229→          </h1>
   230→          <p className="text-sm text-muted-foreground mt-1">
   231→            Browse and manage your archived content
   232→          </p>
   233→        </div>
   234→        <div className="flex items-center gap-2">
   235→          <Button
   236→            variant={viewMode === "grid" ? "default" : "outline"}
   237→            size="icon"
   238→            className={cn("h-8 w-8", viewMode === "grid" && "bg-emerald-600 hover:bg-emerald-700 text-white")}
   239→            onClick={() => setViewMode("grid")}
   240→          >
   241→            <LayoutGrid className="h-4 w-4" />
   242→          </Button>
   243→          <Button
   244→            variant={viewMode === "list" ? "default" : "outline"}
   245→            size="icon"
   246→            className={cn("h-8 w-8", viewMode === "list" && "bg-emerald-600 hover:bg-emerald-700 text-white")}
   247→            onClick={() => setViewMode("list")}
   248→          >
   249→            <List className="h-4 w-4" />
   250→          </Button>
   251→        </div>
   252→      </motion.div>
   253→
   254→      <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-6">
   255→        {/* Stats Row */}
   256→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   257→          <motion.div variants={itemVariants}>
   258→            <Card>
   259→              <CardContent className="p-4 flex items-center gap-3">
   260→                <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-50 dark:bg-emerald-950">
   261→                  <Database className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   262→                </div>
   263→                <div>
   264→                  <p className="text-2xl font-bold">{archivedContents.length}</p>
   265→                  <p className="text-xs text-muted-foreground">Total Archived</p>
   266→                </div>
   267→              </CardContent>
   268→            </Card>
   269→          </motion.div>
   270→          <motion.div variants={itemVariants}>
   271→            <Card>
   272→              <CardContent className="p-4 flex items-center gap-3">
   273→                <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-teal-50 dark:bg-teal-950">
   274→                  <CalendarDays className="h-5 w-5 text-teal-600 dark:text-teal-400" />
   275→                </div>
   276→                <div>
   277→                  <p className="text-2xl font-bold">{archivedThisMonth}</p>
   278→                  <p className="text-xs text-muted-foreground">Archived This Month</p>
   279→                </div>
   280→              </CardContent>
   281→            </Card>
   282→          </motion.div>
   283→          <motion.div variants={itemVariants}>
   284→            <Card>
   285→              <CardContent className="p-4 flex items-center gap-3">
   286→                <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-cyan-50 dark:bg-cyan-950">
   287→                  <HardDrive className="h-5 w-5 text-cyan-600 dark:text-cyan-400" />
   288→                </div>
   289→                <div>
   290→                  <p className="text-2xl font-bold">{storageUsed}</p>
   291→                  <p className="text-xs text-muted-foreground">Storage Used</p>
   292→                </div>
   293→              </CardContent>
   294→            </Card>
   295→          </motion.div>
   296→        </div>
   297→
   298→        {/* Search & Filters */}
   299→        <motion.div variants={itemVariants}>
   300→          <Card>
   301→            <CardContent className="p-4">
   302→              <div className="flex flex-col md:flex-row gap-3">
   303→                <div className="relative flex-1">
   304→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   305→                  <Input
   306→                    placeholder="Search archived content..."
   307→                    value={search}
   308→                    onChange={(e) => setSearch(e.target.value)}
   309→                    className="pl-9"
   310→                  />
   311→                </div>
   312→                <Select value={typeFilter} onValueChange={setTypeFilter}>
   313→                  <SelectTrigger className="w-full md:w-[140px]">
   314→                    <SelectValue placeholder="Type" />
   315→                  </SelectTrigger>
   316→                  <SelectContent>
   317→                    <SelectItem value="all">All Types</SelectItem>
   318→                    <SelectItem value="post">Post</SelectItem>
   319→                    <SelectItem value="story">Story</SelectItem>
   320→                    <SelectItem value="reel">Reel</SelectItem>
   321→                    <SelectItem value="carousel">Carousel</SelectItem>
   322→                    <SelectItem value="video">Video</SelectItem>
   323→                    <SelectItem value="graphic">Graphic</SelectItem>
   324→                    <SelectItem value="article">Article</SelectItem>
   325→                  </SelectContent>
   326→                </Select>
   327→                <Select value={campaignFilter} onValueChange={setCampaignFilter}>
   328→                  <SelectTrigger className="w-full md:w-[160px]">
   329→                    <SelectValue placeholder="Campaign" />
   330→                  </SelectTrigger>
   331→                  <SelectContent>
   332→                    <SelectItem value="all">All Campaigns</SelectItem>
   333→                    {campaigns.map((c) => (
   334→                      <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   335→                    ))}
   336→                  </SelectContent>
   337→                </Select>
   338→                <Select value={teamFilter} onValueChange={setTeamFilter}>
   339→                  <SelectTrigger className="w-full md:w-[140px]">
   340→                    <SelectValue placeholder="Team" />
   341→                  </SelectTrigger>
   342→                  <SelectContent>
   343→                    <SelectItem value="all">All Teams</SelectItem>
   344→                    {teams.map((t) => (
   345→                      <SelectItem key={t.id} value={t.id}>{t.name}</SelectItem>
   346→                    ))}
   347→                  </SelectContent>
   348→                </Select>
   349→              </div>
   350→            </CardContent>
   351→          </Card>
   352→        </motion.div>
   353→
   354→        {/* Bulk actions bar */}
   355→        <AnimatePresence>
   356→          {selectedIds.size > 0 && (
   357→            <motion.div
   358→              initial={{ opacity: 0, y: -10 }}
   359→              animate={{ opacity: 1, y: 0 }}
   360→              exit={{ opacity: 0, y: -10 }}
   361→              className="flex items-center gap-3 p-3 rounded-lg bg-emerald-50 dark:bg-emerald-950 border border-emerald-200 dark:border-emerald-800"
   362→            >
   363→              <span className="text-sm font-medium text-emerald-700 dark:text-emerald-300">
   364→                {selectedIds.size} selected
   365→              </span>
   366→              <div className="flex-1" />
   367→              <Button
   368→                variant="outline"
   369→                size="sm"
   370→                className="text-emerald-600 border-emerald-200 hover:bg-emerald-50 dark:text-emerald-400 dark:border-emerald-800"
   371→                onClick={() => setBulkAction("restore")}
   372→              >
   373→                <RotateCcw className="h-3.5 w-3.5 mr-1.5" />
   374→                Restore Selected
   375→              </Button>
   376→              <Button
   377→                variant="outline"
   378→                size="sm"
   379→                className="text-red-600 border-red-200 hover:bg-red-50 dark:text-red-400 dark:border-red-800"
   380→                onClick={() => setBulkAction("delete")}
   381→              >
   382→                <Trash2 className="h-3.5 w-3.5 mr-1.5" />
   383→                Delete Selected
   384→              </Button>
   385→              <Button
   386→                variant="ghost"
   387→                size="sm"
   388→                onClick={() => setSelectedIds(new Set())}
   389→              >
   390→                <XSquare className="h-3.5 w-3.5 mr-1.5" />
   391→                Clear
   392→              </Button>
   393→            </motion.div>
   394→          )}
   395→        </AnimatePresence>
   396→
   397→        {/* Content list */}
   398→        {filteredContents.length === 0 ? (
   399→          <motion.div variants={itemVariants}>
   400→            <Card>
   401→              <CardContent className="py-16 text-center">
   402→                <Archive className="h-12 w-12 text-muted-foreground/40 mx-auto mb-4" />
   403→                <h3 className="text-lg font-semibold text-muted-foreground">No archived content</h3>
   404→                <p className="text-sm text-muted-foreground mt-1">
   405→                  {archivedContents.length === 0
   406→                    ? "Content that reaches the Archive stage will appear here"
   407→                    : "No results match your filters"}
   408→                </p>
   409→              </CardContent>
   410→            </Card>
   411→          </motion.div>
   412→        ) : viewMode === "grid" ? (
   413→          /* Grid View */
   414→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   415→            <AnimatePresence>
   416→              {filteredContents.map((content) => {
   417→                const TypeIcon = TYPE_ICONS[content.type];
   418→                const typeColor = TYPE_COLORS[content.type];
   419→                const isSelected = selectedIds.has(content.id);
   420→
   421→                return (
   422→                  <motion.div
   423→                    key={content.id}
   424→                    variants={itemVariants}
   425→                    initial="hidden"
   426→                    animate="visible"
   427→                    exit={{ opacity: 0, scale: 0.95 }}
   428→                    layout
   429→                  >
   430→                    <Card className={cn(
   431→                      "transition-all duration-200 relative",
   432→                      isSelected && "ring-2 ring-emerald-500 border-emerald-300 dark:border-emerald-700",
   433→                    )}>
   434→                      {/* Selection checkbox */}
   435→                      <div className="absolute top-3 left-3 z-10">
   436→                        <Checkbox
   437→                          checked={isSelected}
   438→                          onCheckedChange={() => toggleSelect(content.id)}
   439→                          className="bg-background"
   440→                        />
   441→                      </div>
   442→
   443→                      <CardContent className="p-5 pt-10">
   444→                        <div className="space-y-3">
   445→                          {/* Type badge + Title */}
   446→                          <div className="flex items-start gap-2">
   447→                            <Badge
   448→                              variant="secondary"
   449→                              className="gap-1 text-xs shrink-0 mt-0.5"
   450→                              style={{ backgroundColor: `${typeColor}15`, color: typeColor }}
   451→                            >
   452→                              <TypeIcon className="h-3 w-3" />
   453→                              {content.type}
   454→                            </Badge>
   455→                            <h3 className="font-semibold text-sm leading-tight line-clamp-2">
   456→                              {content.title}
   457→                            </h3>
   458→                          </div>
   459→
   460→                          {/* Description */}
   461→                          {content.description && (
   462→                            <p className="text-xs text-muted-foreground line-clamp-2">
   463→                              {content.description}
   464→                            </p>
   465→                          )}
   466→
   467→                          <Separator />
   468→
   469→                          {/* Dates */}
   470→                          <div className="space-y-1.5">
   471→                            <div className="flex items-center justify-between text-xs">
   472→                              <span className="text-muted-foreground">Created</span>
   473→                              <span>{formatDate(content.createdAt)}</span>
   474→                            </div>
   475→                            <div className="flex items-center justify-between text-xs">
   476→                              <span className="text-muted-foreground">Published</span>
   477→                              <span>{formatDate(content.publishedAt)}</span>
   478→                            </div>
   479→                            <div className="flex items-center justify-between text-xs">
   480→                              <span className="text-muted-foreground">Archived</span>
   481→                              <span>{formatDate(content.updatedAt)}</span>
   482→                            </div>
   483→                          </div>
   484→
   485→                          {/* Campaign & Team */}
   486→                          <div className="flex flex-wrap gap-1.5">
   487→                            {content.campaign && (
   488→                              <Badge variant="outline" className="text-xs gap-1">
   489→                                <FolderOpen className="h-3 w-3" />
   490→                                {content.campaign.name}
   491→                              </Badge>
   492→                            )}
   493→                            {content.team && (
   494→                              <Badge variant="outline" className="text-xs gap-1">
   495→                                <Users className="h-3 w-3" />
   496→                                {content.team.name}
   497→                              </Badge>
   498→                            )}
   499→                          </div>
   500→
   501→                          {/* Actions */}
   502→                          <div className="flex items-center gap-2 pt-1">
   503→                            <Button
   504→                              variant="outline"
   505→                              size="sm"
   506→                              className="flex-1 text-emerald-600 border-emerald-200 hover:bg-emerald-50 dark:text-emerald-400 dark:border-emerald-800 dark:hover:bg-emerald-950"
   507→                              onClick={() => handleRestore(content.id)}
   508→                            >
   509→                              <RotateCcw className="h-3.5 w-3.5 mr-1.5" />
   510→                              Restore
   511→                            </Button>
   512→                            <Button
   513→                              variant="outline"
   514→                              size="sm"
   515→                              className="text-red-500 border-red-200 hover:bg-red-50 dark:text-red-400 dark:border-red-800 dark:hover:bg-red-950"
   516→                              onClick={() => setDeleteTarget(content.id)}
   517→                            >
   518→                              <Trash2 className="h-3.5 w-3.5" />
   519→                            </Button>
   520→                          </div>
   521→                        </div>
   522→                      </CardContent>
   523→                    </Card>
   524→                  </motion.div>
   525→                );
   526→              })}
   527→            </AnimatePresence>
   528→          </div>
   529→        ) : (
   530→          /* List View */
   531→          <motion.div variants={itemVariants}>
   532→            <Card>
   533→              <CardContent className="p-0">
   534→                {/* Select all row */}
   535→                <div className="flex items-center gap-3 px-5 py-3 border-b bg-muted/30">
   536→                  <Checkbox
   537→                    checked={selectedIds.size === filteredContents.length && filteredContents.length > 0}
   538→                    onCheckedChange={toggleSelectAll}
   539→                  />
   540→                  <span className="text-xs font-medium text-muted-foreground flex-1">Select All</span>
   541→                  <span className="text-xs text-muted-foreground">{filteredContents.length} items</span>
   542→                </div>
   543→
   544→                <div className="divide-y">
   545→                  {filteredContents.map((content) => {
   546→                    const TypeIcon = TYPE_ICONS[content.type];
   547→                    const typeColor = TYPE_COLORS[content.type];
   548→                    const isSelected = selectedIds.has(content.id);
   549→
   550→                    return (
   551→                      <div
   552→                        key={content.id}
   553→                        className={cn(
   554→                          "flex items-center gap-4 px-5 py-4 transition-colors hover:bg-muted/30",
   555→                          isSelected && "bg-emerald-50/50 dark:bg-emerald-950/20",
   556→                        )}
   557→                      >
   558→                        <Checkbox
   559→                          checked={isSelected}
   560→                          onCheckedChange={() => toggleSelect(content.id)}
   561→                        />
   562→                        <div className="flex h-9 w-9 items-center justify-center rounded-lg shrink-0" style={{ backgroundColor: `${typeColor}15` }}>
   563→                          <TypeIcon className="h-4 w-4" style={{ color: typeColor }} />

   577→                              <span className="flex items-center gap-1">
   578→                                <FolderOpen className="h-3 w-3" /> {content.campaign.name}

   582→                              <span className="flex items-center gap-1">
   583→                                <Users className="h-3 w-3" /> {content.team.name}

   588→                        <div className="flex items-center gap-2 shrink-0">
   589→                          <Button

   599→                            variant="ghost"
   600→                            size="icon"

   602→                            onClick={() => setDeleteTarget(content.id)}

   605→                          </Button>
   606→                        </div>
   607→                      </div>
   608→                    );
   609→                  })}
   610→                </div>
   611→              </CardContent>
   612→            </Card>
   613→          </motion.div>
   614→        )}
   615→      </motion.div>
   616→
   617→      {/* Single delete confirmation */}
   618→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => { if (!open) setDeleteTarget(null); }}>
   619→        <AlertDialogContent>
   620→          <AlertDialogHeader>
   621→            <AlertDialogTitle>Delete Permanently</AlertDialogTitle>

   633→      </AlertDialog>
   634→
   635→      {/* Bulk action confirmation */}
   636→      <AlertDialog open={!!bulkAction} onOpenChange={(open) => { if (!open) setBulkAction(null); }}>

   646→            </AlertDialogDescription>

   650→            <AlertDialogAction

   653→            >
