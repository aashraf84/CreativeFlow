   142→   142→// ── Time formatting ─────────────────────────────────────────────────
   143→   143→function formatRelativeTime(dateStr: string): string {
   144→   144→  try {
   145→   145→    const date = new Date(dateStr);
   146→   146→    const now = new Date();
   147→   147→    const diffMs = now.getTime() - date.getTime();
   148→   148→    const diffMins = Math.floor(diffMs / 60000);
   149→   149→    const diffHours = Math.floor(diffMs / 3600000);
   150→   150→    const diffDays = Math.floor(diffMs / 86400000);
   151→   151→
   152→   152→    if (diffMins < 1) return "Just now";
   153→   153→    if (diffMins < 60) return `${diffMins}m ago`;
   154→   154→    if (diffHours < 24) return `${diffHours}h ago`;
   155→   155→    if (diffDays < 7) return `${diffDays}d ago`;
   156→   156→    return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   157→   157→  } catch {
   158→   158→    return "";
   159→   159→  }
   160→   160→}
