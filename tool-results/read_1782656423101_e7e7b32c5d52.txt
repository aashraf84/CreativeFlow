    63→// ── Add Employee Dialog ─────────────────────────────────────────────
    64→function AddEmployeeDialog({ departments }: { departments: { id: string; name: string }[] }) {
    65→  const [open, setOpen] = useState(false)
    66→  const [loading, setLoading] = useState(false)
    67→  const { setEmployees, employees } = useERPStore()
    68→
    69→  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    70→    e.preventDefault()
    71→    setLoading(true)
    72→    const form = new FormData(e.currentTarget)
    73→    try {
    74→      const res = await fetch("/api/employees", {
    75→        method: "POST",
    76→        headers: { "Content-Type": "application/json" },
    77→        body: JSON.stringify({
    78→          firstName: form.get("firstName"),
    79→          lastName: form.get("lastName"),
    80→          email: form.get("email"),
    81→          phone: form.get("phone"),
    82→          position: form.get("position"),
    83→          departmentId: form.get("departmentId"),
    84→          employmentType: form.get("employmentType"),
    85→          salary: form.get("salary") ? Number(form.get("salary")) : null,
    86→          status: "active",
    87→          employeeId: `EMP-${String(employees.length + 1).padStart(3, "0")}`,
    88→          hireDate: new Date().toISOString(),
    89→        }),
    90→      })
    91→      if (res.ok) {
    92→        const emp = await res.json()
    93→        setEmployees([...employees, emp])
    94→        setOpen(false)
    95→      }
    96→    } catch {
    97→      // handle error silently
    98→    } finally {
    99→      setLoading(false)
   100→    }
   101→  }
   102→
   103→  return (
   104→    <Dialog open={open} onOpenChange={setOpen}>
   105→      <DialogTrigger asChild>
   106→        <Button className="bg-gradient-to-br from-emerald-500 to-teal-600 hover:from-emerald-600 hover:to-teal-700 text-white gap-2 rounded-full shadow-md shadow-emerald-500/25">
   107→          <Plus className="h-4 w-4" /> Add Employee

   109→      </DialogTrigger>
   110→      <DialogContent className="sm:max-w-[500px] glass border border-black/5 dark:border-white/10 shadow-macos-lg rounded-2xl">
   111→        <DialogHeader>
   112→          <DialogTitle className="text-lg font-semibold">Add New Employee</DialogTitle>
   113→        </DialogHeader>
   114→        <form onSubmit={handleSubmit} className="space-y-4">
   115→          <div className="grid grid-cols-2 gap-4">
   116→            <div className="space-y-2">
   117→              <Label htmlFor="firstName">First Name</Label>
   118→              <Input id="firstName" name="firstName" required placeholder="John" />
   119→            </div>
   120→            <div className="space-y-2">
   121→              <Label htmlFor="lastName">Last Name</Label>
   122→              <Input id="lastName" name="lastName" required placeholder="Doe" />
   123→            </div>
   124→          </div>
   125→          <div className="space-y-2">
   126→            <Label htmlFor="email">Email</Label>
   127→            <Input id="email" name="email" type="email" required placeholder="john.doe@company.com" />
   128→          </div>
   129→          <div className="space-y-2">
   130→            <Label htmlFor="phone">Phone</Label>
   131→            <Input id="phone" name="phone" placeholder="+1-555-0000" />
   132→          </div>
   133→          <div className="grid grid-cols-2 gap-4">
   134→            <div className="space-y-2">
   135→              <Label htmlFor="position">Position</Label>

   141→                <SelectTrigger><SelectValue placeholder="Select..." /></SelectTrigger>
   142→                <SelectContent>
   143→                  {departments.map((d) => (
   144→                    <SelectItem key={d.id} value={d.id}>{d.name}</SelectItem>

   155→                <SelectContent>
   156→                  <SelectItem value="full-time">Full-time</SelectItem>
   157→                  <SelectItem value="part-time">Part-time</SelectItem>
   158→                  <SelectItem value="contract">Contract</SelectItem>
