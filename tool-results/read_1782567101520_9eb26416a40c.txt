   119→// ── Format helpers ──────────────────────────────────────────────────
   120→function formatFileSize(bytes: number): string {
   121→  if (bytes === 0) return "0 B"
   122→  const k = 1024
   123→  const sizes = ["B", "KB", "MB", "GB"]
   124→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   125→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + " " + sizes[i]
   126→}
   127→
   128→function formatDate(dateStr: string): string {
   129→  return new Date(dateStr).toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" })
   130→}
